Project doctrine:
- Input may intend.
- Context may remind.
- Output may propose.

- ReducerRegistry commits accepted validations through owner reducers.
- OperationLedger dedupes every side effect.
- DirtyQueue marks refresh targets.
- Context reads.
- Cards mirror truth later; raw Story Card prose does not auto-fire mechanics.
- ReleaseCheck must tell the truth.

Hard build law for the clean-core rebuild:
- No hook wrapping.
- No late repair authority.
- No fake green checks.
- No acceptance shrink.

  WL.U = {
    norm: function (value) {
      return String(value == null ? '' : value)
        .replace(/[\u2018\u2019]/g, "'")
        .replace(/[\u201C\u201D]/g, '"')
        .replace(/\s+/g, ' ')
        .trim();
    },
    lower: function (value) { return WL.U.norm(value).toLowerCase(); },
    title: function (value) {
      var raw = WL.U.norm(value).replace(/^the\s+/i, '');
      if (!raw) return '';
      var small = { a: 1, an: 1, and: 1, at: 1, by: 1, for: 1, from: 1, in: 1, of: 1, on: 1, or: 1, the: 1, to: 1, with: 1 };
      var parts = raw.split(/\s+/);
      for (var i = 0; i < parts.length; i++) {
        var w = parts[i].toLowerCase();
        parts[i] = (i > 0 && small[w]) ? w : w.charAt(0).toUpperCase() + w.slice(1);
      }
      return parts.join(' ');
    },
    slug: function (value) {
      var s = WL.U.lower(value).replace(/[^a-z0-9]+/g, '_').replace(/^_+|_+$/g, '');
      return s.slice(0, 80) || 'unknown';
    },
    clamp: function (n, min, max) {
      n = Number(n);
      if (!isFinite(n)) n = min;
      if (n < min) return min;
      if (n > max) return max;
      return n;
    },
    hash: function (value) {
      var s = String(value == null ? '' : value);
      var h = 2166136261;
      for (var i = 0; i < s.length; i++) {
        h ^= s.charCodeAt(i);
        h += (h << 1) + (h << 4) + (h << 7) + (h << 8) + (h << 24);
      }
      return ('00000000' + ((h >>> 0).toString(16))).slice(-8);
    },
    clone: function (value) {
      try { return JSON.parse(JSON.stringify(value)); } catch (e) { return value; }
    },
    capArray: function (arr, max) {
      if (!Array.isArray(arr)) return [];
      while (arr.length > max) arr.shift();
      return arr;
    },
    ensureObject: function (parent, key, factory) {
      if (!parent || typeof parent !== 'object') return factory ? factory() : {};
      if (!parent[key] || typeof parent[key] !== 'object' || Array.isArray(parent[key])) parent[key] = factory ? factory() : {};
      return parent[key];
    },
    ensureArray: function (parent, key) {
      if (!parent || typeof parent !== 'object') return [];
      if (!Array.isArray(parent[key])) parent[key] = [];
      return parent[key];
    },
    startsCommand: function (text) { return /^\s*\/\w+/.test(String(text || '')); },
    commandName: function (text) {
      var m = String(text || '').match(/^\s*\/([a-zA-Z0-9_-]+)/);
      return m ? m[1].toLowerCase() : '';
    },
    commandArgs: function (text) { return String(text || '').replace(/^\s*\/\S+\s*/, '').trim(); },
    safeString: function (value) {
      if (value == null) return '';
      if (typeof value === 'string') return WL.U.norm(value);
      if (typeof value === 'number' || typeof value === 'boolean') return String(value);
      if (Array.isArray(value)) return value.map(function (v) { return WL.U.safeString(v); }).filter(Boolean).join(', ');
      if (typeof value === 'object') {
        if (value.displayName) return WL.U.safeString(value.displayName);
        if (value.name) return WL.U.safeString(value.name);
        if (value.title) return WL.U.safeString(value.title);
        try { return JSON.stringify(value); } catch (e) { return '[object]'; }
      }
      return WL.U.norm(value);
    },
    getGlobalState: function () {
      if (typeof globalThis === 'undefined') return {};
      if (!globalThis.state || typeof globalThis.state !== 'object') globalThis.state = {};
      return globalThis.state;
    },
    getActionCount: function () {
      try {
        if (typeof info === 'object' && info && typeof info.actionCount === 'number') return info.actionCount;
      } catch (e) {}
      return 0;
    },
    splitClauses: function (text) {
      var t = WL.U.norm(text);
      if (!t) return [];
      return t.split(/(?<=[.!?])\s+|\n+|;+/).map(function (s) { return WL.U.norm(s); }).filter(Boolean).slice(0, 18);
    }
  };

  // ============================================================
  // 06. Runtime hooks and source classification
  // ============================================================
  WL.SourceGuard = {
    classify: function (text, hook) {
      var raw = String(text == null ? '' : text);
      var n = WL.U.norm(raw);
      if (WL.U.startsCommand(raw)) return { kind: 'command', hook: hook, commandName: WL.U.commandName(raw), eligibleForCanon: false, reason: 'command_input' };
      if (/^\[WorldLedger\]/i.test(n) || /\bCommand:\s*\//i.test(n) || /\bSummary:\s*PASS\b/i.test(n)) {
        return { kind: 'command_echo', hook: hook || 'unknown', commandName: '', eligibleForCanon: false, reason: 'command_or_check_output_echo' };
      }
      if (hook === 'context') return { kind: 'context', hook: hook, commandName: '', eligibleForCanon: false, reason: 'context_is_read_only' };
      if (hook === 'input') return { kind: 'input', hook: hook, commandName: '', eligibleForCanon: false, reason: 'input_intent_only' };
      return { kind: hook || 'unknown', hook: hook || 'unknown', commandName: '', eligibleForCanon: hook === 'output', reason: hook === 'output' ? 'output_can_propose' : 'unknown_source' };
    }
  };

  WL.Runtime = {
    begin: function (root, hook, text) {
      root.runtime.currentHook = hook;
      root.runtime.currentTurn = WL.U.getActionCount();
      root.runtime.hookDepth += 1;
      root.runtime.hookTrace.push({ turn: root.runtime.currentTurn, hook: hook, hash: WL.U.hash(text), phase: 'begin' });
      WL.U.capArray(root.runtime.hookTrace, WL.Constants.maxHookTrace);
    },
    end: function (root, hook, status) {
      root.runtime.lastReturnStatus = status || 'ok';
      root.runtime.hookTrace.push({ turn: root.runtime.currentTurn, hook: hook, phase: 'end', status: status || 'ok' });
      WL.U.capArray(root.runtime.hookTrace, WL.Constants.maxHookTrace);
      root.runtime.hookDepth = Math.max(0, root.runtime.hookDepth - 1);
      root.runtime.currentHook = 'none';
      WL.StateManager.enforceInvariants(root);
    },
    runPipeline: function (root, text, hook) {
      var source = WL.SourceGuard.classify(text, hook);
      var events = [];
      if (source.eligibleForCanon || hook === 'input') {
        events = hook === 'output' ? WL.EventScanner.scanOutput(text, root, source) : WL.EventScanner.scanInput(text, root, source);
      }
      var proposals = WL.ProposalBuilder.fromEvents(root, events, source);
      var validations = WL.CanonValidator.validateAll(root, proposals, source);
      var reducerResults = WL.ReducerRegistry.applyAcceptedBatch(root, validations, source);
      WL.DirtyQueue.mark(root, reducerResults);
      root.pending.lastParserHits = events;
      root.pending.lastProposals = proposals;
      root.pending.lastValidations = validations;
      root.pending.lastReducerResults = reducerResults;
      root.pending.lastPipeline = { hook: hook, sourceKind: source.kind, events: events.length, proposals: proposals.length, validations: validations.length, reducers: reducerResults.length };
      root.runtime.lastPipelineSummary = hook + ': events=' + events.length + ', proposals=' + proposals.length + ', validations=' + validations.length + ', reducers=' + reducerResults.length;
      return root.pending.lastPipeline;
    },
    handleInput: function (text) {
      var root = WL.StateManager.root();
      WL.Runtime.begin(root, 'input', text);
      var source = WL.SourceGuard.classify(text, 'input');
      root.runtime.lastInputHash = WL.U.hash(text);
      var out = text;
      if (source.kind === 'command') {
        root.runtime.lastCommandName = source.commandName;
        root.runtime.lastCommandHash = WL.U.hash(text);
        root.runtime.lastCommandTurn = root.runtime.currentTurn;
        out = WL.CommandRouter.route(text, root);
      } else {
        WL.Runtime.runPipeline(root, text, 'input');
      }
      WL.Runtime.end(root, 'input', 'ok');
      return out;
    },
    handleContext: function (text) {
      var root = WL.StateManager.root();
      WL.Runtime.begin(root, 'context', text);
      root.runtime.lastContextHash = WL.U.hash(text);
      var packet = WL.ContextSystem.build(root, text);
      WL.Runtime.end(root, 'context', 'ok');
      return packet;
    },
    handleOutput: function (text) {
      var root = WL.StateManager.root();
      WL.Runtime.begin(root, 'output', text);
      root.runtime.lastOutputHash = WL.U.hash(text);
      WL.Runtime.runPipeline(root, text, 'output');
      WL.Runtime.end(root, 'output', 'ok');
      return text;
    }
  };

  // ============================================================
  // 07. EventScanner / ProposalBuilder / CanonValidator / ReducerRegistry
  // ============================================================
  WL.EventScanner = {
    makeEvent: function (family, type, clause, source, data) {
      var c = WL.U.norm(clause);
      return {
        family: family,
        type: type,
        sourceHook: source.hook,
        sourceKind: source.kind,
        sourceHash: WL.U.hash(source.hook + '|' + c),
        localClauseHash: WL.U.hash(c),
        textPreview: c.slice(0, 220),
        data: data || {},
        confidence: data && data.confidence || 'candidate'
      };
    },
    scanInput: function (text, root, source) {
      var out = [];
      var clauses = WL.U.splitClauses(text);
      for (var i = 0; i < clauses.length; i++) {
        out.push(WL.EventScanner.makeEvent('input', 'intent_observed', clauses[i], source, { confidence: 'intent_only' }));
      }
      WL.EventScanner.record(root, out);
      return out;
    },
    scanOutput: function (text, root, source) {
      var out = [];
      var clauses = WL.U.splitClauses(text);
      if (!clauses.length && WL.U.norm(text)) clauses = [WL.U.norm(text)];
      for (var i = 0; i < clauses.length; i++) {
        var clause = clauses[i];
        out.push(WL.EventScanner.makeEvent('system', 'output_observed', clause, source, { confidence: 'confirmed_output' }));
        var inv = WL.EventScanner.detectInventoryCandidate(clause);
        if (inv) out.push(WL.EventScanner.makeEvent('inventory', inv.type, clause, source, inv));
        var quest = WL.EventScanner.detectQuestCandidate(clause);
        if (quest) out.push(WL.EventScanner.makeEvent('quest', quest.type, clause, source, quest));
      }
      WL.EventScanner.record(root, out);
      return out;
    },
    detectInventoryCandidate: function (clause) {
      var c = WL.U.norm(clause);
      var lower = WL.U.lower(c);
      if (/\b(almost|tries? to|attempts? to|fails? to|refuses? to|declines? to)\b/.test(lower)) return null;
      if (/\b(reaches? for|looks? at|sees?|notices?|wants?|needs?|promises?|objective|quest)\b/.test(lower)) return null;
      var m = c.match(/\b(?:pick(?:s)? up|grab(?:s)?|take(?:s)?|loot(?:s)?|receive(?:s)?|pocket(?:s)?|snatch(?:es)? up|scoop(?:s)? up|secure(?:s)?|lift(?:s)?|catch(?:es)?|take(?:s)? hold of)\s+(?:the\s+|a\s+|an\s+)?([^.!?;,]{2,80})/i);
      if (!m) m = c.match(/\b(?:hands?|gives?|tosses?)\s+you\s+(?:the\s+|a\s+|an\s+)?([^.!?;,]{2,80})/i);
      if (!m) return null;
      var target = WL.EventScanner.cleanCandidateTarget(m[1]);
      if (!target) return null;
      return { type: 'acquire_candidate', targetName: target, confidence: 'candidate_confirmed_phrase' };
    },
    detectQuestCandidate: function (clause) {
      var c = WL.U.norm(clause);
      var lower = WL.U.lower(c);
      if (/\b(asks? you to|offers? you|quest|task|mission|promises?)\b/.test(lower) && /\b(clear|find|bring|escort|repair|meet|deliver|kill|defeat|investigate|protect|rescue)\b/.test(lower)) {
        return { type: 'quest_candidate', titleSeed: c.slice(0, 120), confidence: 'candidate_quest_language' };
      }
      return null;
    },
    cleanCandidateTarget: function (target) {
      var t = WL.U.norm(target)
        .replace(/\b(before|while|as|when|but|and then|from)\b[\s\S]*$/i, '')
        .replace(/^hold\s+of\s+/i, '')
        .replace(/^up\s+/i, '')
        .replace(/\s+/g, ' ')
        .trim();
      if (!t || t.length < 2) return '';
      if (/\bquest\b/i.test(t)) return '';
      return WL.U.title(t);
    },
    record: function (root, events) {
      for (var i = 0; i < events.length; i++) root.diagnostics.parser.push(events[i]);
      WL.U.capArray(root.diagnostics.parser, WL.Constants.maxParserHits);
    }
  };

  WL.ProposalBuilder = {
    fromEvents: function (root, events, source) {
      var out = [];
      var seen = {};
      for (var i = 0; i < events.length; i++) {
        var ev = events[i];
        var target = ev.data && ev.data.targetName ? WL.U.slug(ev.data.targetName) : '';
        var localKey = [ev.family, ev.type, ev.sourceHash, ev.localClauseHash, target].join('|');
        if (seen[localKey]) continue;
        seen[localKey] = true;
        var proposal = {
          id: 'proposal_' + WL.U.hash(localKey + '|' + i),
          family: ev.family,
          type: ev.type,
          sourceHook: ev.sourceHook,
          sourceKind: ev.sourceKind,
          sourceHash: ev.sourceHash,
          localClauseHash: ev.localClauseHash,
          targetId: target,
          targetName: ev.data && ev.data.targetName || '',
          textPreview: ev.textPreview,
          data: ev.data || {},
          status: 'proposed',
          opKey: WL.OperationLedger.makeKey([ev.sourceHook, ev.sourceHash, ev.localClauseHash, ev.family, ev.type, target || 'none'])
        };
        out.push(proposal);
        root.diagnostics.proposals.push(proposal);
      }
      WL.U.capArray(root.diagnostics.proposals, WL.Constants.maxProposals);
      return out;
    }
  };

  WL.CanonValidator = {
    validateAll: function (root, proposals, source) {
      var out = [];
      for (var i = 0; i < proposals.length; i++) out.push(WL.CanonValidator.validate(root, proposals[i], source));
      for (var j = 0; j < out.length; j++) root.diagnostics.validations.push(out[j]);
      WL.U.capArray(root.diagnostics.validations, WL.Constants.maxValidations);
      return out;
    },
    validate: function (root, proposal, source) {
      var result = {
        id: 'validation_' + WL.U.hash(proposal.id + '|' + proposal.opKey),
        proposalId: proposal.id,
        proposal: proposal,
        family: proposal.family,
        type: proposal.type,
        accepted: false,
        status: 'REJECTED',
        reason: '',
        reducer: ''
      };
      if (!source || !source.eligibleForCanon) {
        result.status = source && source.kind === 'input' ? 'PENDING' : 'REJECTED';
        result.reason = source && source.reason || 'source_not_eligible';
        return result;
      }
      if (proposal.family === 'system' && proposal.type === 'output_observed') {
        result.accepted = true;
        result.status = 'ACCEPTED';
        result.reason = 'pipeline_observation_only';
        result.reducer = 'SystemObservationReducer';
        return result;
      }
      if (proposal.family === 'inventory' || proposal.family === 'quest') {
        result.status = 'BLOCKED';
        result.reason = proposal.family + '_owner_not_installed_until_later_build';
        return result;
      }
      result.status = 'PENDING';
      result.reason = 'no_owner_for_family_in_build2';
      return result;
    }
  };

  WL.ReducerRegistry = {
    handlers: {},
    applyAcceptedBatch: function (root, validations, source) {
      var results = [];
      for (var i = 0; i < validations.length; i++) {
        var v = validations[i];
        if (!v.accepted) {
          results.push({ status: v.status, applied: false, reason: v.reason, family: v.family, type: v.type, targetId: v.proposal && v.proposal.targetId || '' });
          continue;
        }
        var key = v.proposal.opKey;
        var ledger = WL.OperationLedger.add(root, key, v.reducer || v.family);
        if (!ledger.applied) {
          results.push({ status: 'DUPLICATE_SKIPPED', applied: false, duplicate: true, key: ledger.key, family: v.family, type: v.type });
          continue;
        }
        var handler = WL.ReducerRegistry.handlers[v.reducer];
        if (typeof handler !== 'function') {
          results.push({ status: 'NO_HANDLER', applied: false, key: ledger.key, family: v.family, type: v.type, reason: 'handler_missing' });
          continue;
        }
        var h = handler(root, v, ledger);
        results.push(h);
      }
      for (var j = 0; j < results.length; j++) root.diagnostics.reducers.push(results[j]);
      WL.U.capArray(root.diagnostics.reducers, WL.Constants.maxReducerResults);
      return results;
    }
  };

  WL.ReducerRegistry.handlers.SystemObservationReducer = function (root, validation, ledger) {
    return {
      status: 'APPLIED_DIAGNOSTIC_ONLY',
      applied: true,
      family: validation.family,
      type: validation.type,
      key: ledger.key,
      note: 'Build 2 commits only pipeline diagnostics, not gameplay truth.'
    };
  };

  WL.DirtyQueue = {
    mark: function (root, reducerResults) {
      var anyApplied = false;
      for (var i = 0; i < reducerResults.length; i++) if (reducerResults[i].applied) anyApplied = true;
      if (anyApplied) {
        root.dirty.targets.push('diagnostics');
        root.dirty.reasons.push('pipeline_observation');
        WL.U.capArray(root.dirty.targets, 40);
        WL.U.capArray(root.dirty.reasons, 40);
      }
    }
  };

  // ============================================================
  WL.OperationLedger = {
    makeKey: function (parts) { return WL.U.slug((parts || []).join('|')); },
    has: function (root, key) { return !!(root.operationLedger.keysById[WL.OperationLedger.makeKey([key])]); },
    add: function (root, key, reason) {
      var k = WL.OperationLedger.makeKey([key]);
      if (root.operationLedger.keysById[k]) {
        root.operationLedger.duplicateSkips.push({ key: k, reason: reason || 'duplicate', turn: root.runtime.currentTurn });
        WL.U.capArray(root.operationLedger.duplicateSkips, 80);
        return { applied: false, duplicate: true, key: k };
      }
      root.operationLedger.keysById[k] = { key: k, reason: reason || '', turn: root.runtime.currentTurn };
      root.operationLedger.recentKeys.push(k);
      WL.U.capArray(root.operationLedger.recentKeys, WL.Constants.maxRecentKeys);
      return { applied: true, duplicate: false, key: k };
    }
  };

  WL.Diagnostics = {

      WL.U.capArray(root.diagnostics.audit, WL.Constants.maxAuditRows);
      return rec;
    }

    summarize: function (rows) {
