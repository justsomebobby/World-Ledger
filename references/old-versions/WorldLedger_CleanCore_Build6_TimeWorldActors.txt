    classify: function (text, hook) {
      var raw = String(text == null ? '' : text);
      var n = WL.U.norm(raw);
      if (WL.U.startsCommand(raw)) return { kind: 'command', hook: hook, commandName: WL.U.commandName(raw), eligibleForCanon: false, reason: 'command_input' };

        return { kind: 'command_echo', hook: hook || 'unknown', commandName: '', eligibleForCanon: false, reason: 'command_or_check_output_echo' };
      }
      if (hook === 'context') return { kind: 'context', hook: hook, commandName: '', eligibleForCanon: false, reason: 'context_is_read_only' };
      if (hook === 'input') return { kind: 'input', hook: hook, commandName: '', eligibleForCanon: false, reason: 'input_intent_only' };
      return { kind: hook || 'unknown', hook: hook || 'unknown', commandName: '', eligibleForCanon: hook === 'output', reason: hook === 'output' ? 'output_can_propose' : 'unknown_source' };
    }
  };

      root.runtime.hookDepth = Math.max(0, root.runtime.hookDepth - 1);
      root.runtime.currentHook = 'none';
      WL.StateManager.enforceInvariants(root);

    runPipeline: function (root, text, hook) {
      var source = WL.SourceGuard.classify(text, hook);
      var events = [];
      if (source.eligibleForCanon || hook === 'input') {
        events = hook === 'output' ? WL.EventScanner.scanOutput(text, root, source) : WL.EventScanner.scanInput(text, root, source);
      }
      var proposals = WL.ProposalBuilder.fromEvents(root, events, source);
      var validations = WL.CanonValidator.validateAll(root, proposals, source);
      var reducerResults = WL.ReducerRegistry.applyAcceptedBatch(root, validations, source);
      WL.DirtyQueue.mark(root, reducerResults);
      root.pending.lastParserHits = events;
      root.pending.lastProposals = proposals;
      root.pending.lastValidations = validations;
      root.pending.lastReducerResults = reducerResults;
      root.pending.lastPipeline = { hook: hook, sourceKind: source.kind, events: events.length, proposals: proposals.length, validations: validations.length, reducers: reducerResults.length };
      root.runtime.lastPipelineSummary = hook + ': events=' + events.length + ', proposals=' + proposals.length + ', validations=' + validations.length + ', reducers=' + reducerResults.length;
      return root.pending.lastPipeline;
    },
    handleInput: function (text) {

        root.runtime.lastCommandName = source.commandName;
        root.runtime.lastCommandHash = WL.U.hash(text);
        root.runtime.lastCommandTurn = root.runtime.currentTurn;
        out = WL.CommandRouter.route(text, root);
      } else {
        WL.Runtime.runPipeline(root, text, 'input');
      }
      WL.Runtime.end(root, 'input', 'ok');
      return out;
    },

    },
    handleOutput: function (text) {
      var root = WL.StateManager.root();
      WL.Runtime.begin(root, 'output', text);
      root.runtime.lastOutputHash = WL.U.hash(text);
      WL.Runtime.runPipeline(root, text, 'output');
      WL.Runtime.end(root, 'output', 'ok');
      return text;
    }
  };

  WL.PlayerProfileSystem = {
    ensure: function (root) {
      if (!root.startup || typeof root.startup !== 'object') root.startup = { profileImported: false, profileSourceHash: '', importedFields: [], rejectedMechanics: [], warnings: [] };
      if (!root.player || typeof root.player !== 'object') root.player = WL.Schema.root().player;
      if (!Array.isArray(root.startup.importedFields)) root.startup.importedFields = [];
      if (!Array.isArray(root.startup.rejectedMechanics)) root.startup.rejectedMechanics = [];
      if (!Array.isArray(root.startup.warnings)) root.startup.warnings = [];
      return root.player;
    },
    collectPlaceholderEntries: function (stateObj) {
      var out = [];
      var ph = stateObj && stateObj.placeholders;
      if (!ph) return out;
      if (Array.isArray(ph)) {
        for (var i = 0; i < ph.length; i++) {
          var e = ph[i];
          if (e && typeof e === 'object') out.push({ label: WL.U.safeString(e.label || e.name || e.key || e.title || e.prompt || e.question || ''), value: WL.U.safeString(e.value || e.answer || e.text || e.result || e.response || '') });
        }
      } else if (typeof ph === 'object') {
        var keys = Object.keys(ph);
        for (var j = 0; j < keys.length; j++) out.push({ label: keys[j], value: WL.U.safeString(ph[keys[j]]) });
      }
      return out.filter(function (e) { return e.label && e.value; });
    },
    classifyLabel: function (label) {
      var l = WL.U.lower(label).replace(/[^a-z0-9]+/g, ' ').trim();
      var compact = l.replace(/\s+/g, '');
      if (/pronouns?/.test(l)) return 'pronouns';
      if (/(^| )(age|years old|how old|old are you)( |$)/.test(l) || /howoldareyou/.test(compact)) return 'age';
      if (/(appearance|look like|looks|describe yourself|description|physical traits|body description)/.test(l)) return 'appearance';
      if (/(personality|traits|demeanor|temperament|attitude|what are you like)/.test(l)) return 'personality';
      if (/(backstory|history|past|former life|before the story)/.test(l)) return 'backstory';
      if (/(race|species|ancestry|lineage|bloodline|creature type|what are you|form|body type)/.test(l)) return 'race';
      if (/(class|profession|job|role|origin|background|where are you from)/.test(l)) return 'origin';
      if (/(name|called|who are you|character name|player name|choose your name)/.test(l) || /whatsyourname/.test(compact)) return 'name';
      return 'notes';
    },
    hasMechanicalCheat: function (value) {
      var v = WL.U.lower(value);
      return /\blevel\s*\d{2,}\b/.test(v) || /\b\d{4,}\s*(?:gold|silver|copper)\b/.test(v) || /\b(?:own|owns|owned|have|has)\s+(?:a\s+|an\s+|the\s+)?(?:legendary|artifact|mythic|divine)\b/.test(v) || /\bcan\s+cast\s+[A-Z]?[a-z]+/.test(value) || /\bcompleted\s+(?:the\s+)?quest\b/.test(v);
    },
    addUnique: function (arr, value, max) {
      var v = WL.U.norm(value);
      if (!v || !Array.isArray(arr)) return false;
      var sv = WL.U.slug(v);
      for (var i = 0; i < arr.length; i++) if (WL.U.slug(arr[i]) === sv) return false;
      arr.push(v); WL.U.capArray(arr, max || 12); return true;
    },
    applyEntry: function (root, entry) {
      var category = WL.PlayerProfileSystem.classifyLabel(entry.label);
      var value = WL.U.norm(entry.value);
      if (!value) return;
      if (WL.PlayerProfileSystem.hasMechanicalCheat(value)) {
        root.startup.rejectedMechanics.push({ label: entry.label, value: value, reason: 'mechanical_setup_claim_not_imported' });
        WL.U.capArray(root.startup.rejectedMechanics, 20);
      }
      if (category === 'name' && !WL.PlayerProfileSystem.hasMechanicalCheat(value)) root.player.name = WL.U.title(value);
      else if (category === 'pronouns') root.player.pronouns = value;
      else if (category === 'age') { var m = value.match(/\b(\d{1,4})\b/); root.player.age = m ? m[1] : value; }
      else if (category === 'appearance') root.player.appearance = value;
      else if (category === 'personality') WL.PlayerProfileSystem.addUnique(root.player.personalityTraits, value, 12);
      else if (category === 'backstory') {
        WL.PlayerProfileSystem.addUnique(root.player.backstoryNotes, value, 12);
        if (/\bkilled\s+(?:the\s+)?king\b/i.test(value)) WL.PlayerProfileSystem.addUnique(root.player.priorEvents, 'Killed the king before the story began', 12);
      } else if (category === 'race') WL.RaceRegistry.applyChosenRace(root, value);
      else if (category === 'origin') {
        if (/class|role/i.test(entry.label)) root.player.classLabel = value;
        else if (/profession|job/i.test(entry.label)) root.player.profession = value;
        else root.player.origin = value;
      } else WL.PlayerProfileSystem.addUnique(root.player.notes, entry.label + ': ' + value, 12);
      root.startup.importedFields.push({ label: entry.label, category: category });
      WL.U.capArray(root.startup.importedFields, 30);
    },
    importFromState: function (root, stateObj) {
      WL.PlayerProfileSystem.ensure(root);
      var entries = WL.PlayerProfileSystem.collectPlaceholderEntries(stateObj || {});
      var hash = WL.U.hash(JSON.stringify(entries));
      if (!entries.length || root.startup.profileSourceHash === hash) return false;
      for (var i = 0; i < entries.length; i++) WL.PlayerProfileSystem.applyEntry(root, entries[i]);
      root.startup.profileImported = true;
      root.startup.profileSourceHash = hash;
      return true;
    }
  };

  WL.RaceRegistry = {
    normalizeRaceName: function (value) {
      var v = WL.U.norm(value).replace(/^(?:i am|i'm|a|an|the)\s+/i, '').replace(/\s+(?:race|species|form)$/i, '');
      return WL.U.title(v || 'Unknown');
    },
    scanStoryCards: function (stateObj) {
      var cards = WL.U.getStoryCards(stateObj);
      var out = [];
      for (var i = 0; i < cards.length; i++) {
        var c = cards[i] || {};
        var title = WL.U.cardTitle(c);
        var body = WL.U.cardBody(c);
        var keys = WL.U.safeString(c.keys || c.key || '');
        var hay = WL.U.lower(title + ' ' + keys + ' ' + body);
        if (title && /(race|species|ancestry|lineage|bloodline|elf|orc|human|dwarf|dragon|beast|kin|folk)/i.test(hay)) out.push({ id: c.id || ('racecard_' + WL.U.hash(title + body)), title: title, keys: keys, body: body });
      }
      return out;
    },
    resolveChosenRace: function (name) {
      var chosen = WL.RaceRegistry.normalizeRaceName(name);
      var slug = WL.U.slug(chosen);
      var cards = WL.RaceRegistry.scanStoryCards(WL.U.getGlobalState());
      for (var i = 0; i < cards.length; i++) {
        var card = cards[i];
        var hay = WL.U.lower(card.title + ' ' + card.keys);
        if (WL.U.slug(card.title) === slug || hay.indexOf(WL.U.lower(chosen)) >= 0) return { name: chosen, source: 'story_card', sourceTitle: card.title, summary: WL.RaceRegistry.summarizeCard(card.body), traits: WL.RaceRegistry.extractTraits(card.body), constraints: [], unknowns: [] };
      }
      return { name: chosen, source: 'setup_only_uncertain', sourceTitle: '', summary: 'Race/species chosen in setup; no scenario race card found yet.', traits: [], constraints: [], unknowns: ['traits_unknown_until_confirmed'] };
    },
    summarizeCard: function (body) {
      var b = WL.U.norm(body);
      if (!b) return '';
      return b.split(/(?<=[.!?])\s+/)[0].slice(0, 240);
    },
    extractTraits: function (body) {
      var low = WL.U.lower(body);
      var traits = [];
      var candidates = ['wings','horns','tail','night vision','darkvision','gills','claws','fangs','fire resistance','cold resistance','magic sensitivity'];
      for (var i = 0; i < candidates.length; i++) if (low.indexOf(candidates[i]) >= 0) traits.push(WL.U.title(candidates[i]));
      return traits.slice(0, 8);
    },
    applyChosenRace: function (root, value) {
      var rec = WL.RaceRegistry.resolveChosenRace(value);
      root.player.raceSpecies = rec.name;
      root.player.trueRace = rec.name;
      if (!root.player.currentForm || root.player.currentForm === 'unknown') root.player.currentForm = rec.name;
      root.player.raceProfile = { source: rec.source, sourceTitle: rec.sourceTitle, summary: rec.summary, traits: rec.traits || [], constraints: rec.constraints || [], unknowns: rec.unknowns || [] };
      return rec;
    }
  };

  WL.FormTransformationSystem = {
    apply: function (root, validation, ledger) {
      var p = validation.proposal;
      if (p.type === 'dream_form_candidate' || p.type === 'vision_form_candidate') return WL.FormTransformationSystem.recordNonCanon(root, p, ledger, 'dream_or_vision_not_true_form');
      if (p.type === 'illusion_form_candidate') return WL.FormTransformationSystem.setIllusion(root, p, ledger);
      if (p.type === 'disguise_form_candidate') return WL.FormTransformationSystem.setDisguise(root, p, ledger);
      if (p.type === 'temporary_form_candidate') return WL.FormTransformationSystem.setTemporary(root, p, ledger);
      if (p.type === 'permanent_form_candidate' || p.type === 'race_shift_candidate') return WL.FormTransformationSystem.setPermanent(root, p, ledger);
      if (p.type === 'body_trait_candidate') return WL.FormTransformationSystem.addBodyTrait(root, p, ledger);
      return { status: 'IGNORED', applied: false, family: validation.family, type: validation.type, reason: 'unsupported_form_event' };
    },
    record: function (root, kind, target, key, phrase) {
      var rec = { turn: root.runtime.currentTurn, kind: kind, target: WL.U.title(target || ''), key: key, phrase: phrase || '' };
      root.body.formHistory.push(rec); WL.U.capArray(root.body.formHistory, 30); return rec;
    },
    recordNonCanon: function (root, proposal, ledger, reason) {
      WL.FormTransformationSystem.record(root, reason, proposal.data.targetForm || '', ledger.key, proposal.textPreview);
      return { status: 'RECORDED_NON_CANON', applied: false, family: 'form', type: proposal.type, targetForm: proposal.data.targetForm || '', key: ledger.key };
    },
    setIllusion: function (root, proposal, ledger) {
      root.body.activeIllusion = WL.U.title(proposal.data.targetForm || '');
      WL.FormTransformationSystem.record(root, 'illusion_only', root.body.activeIllusion, ledger.key, proposal.textPreview);
      return { status: 'APPLIED', applied: true, family: 'form', type: 'illusion', targetForm: root.body.activeIllusion, key: ledger.key };
    },
    setDisguise: function (root, proposal, ledger) {
      root.body.activeDisguise = WL.U.title(proposal.data.targetForm || '');
      WL.FormTransformationSystem.record(root, 'disguise_only', root.body.activeDisguise, ledger.key, proposal.textPreview);
      return { status: 'APPLIED', applied: true, family: 'form', type: 'disguise', targetForm: root.body.activeDisguise, key: ledger.key };
    },
    setTemporary: function (root, proposal, ledger) {
      root.body.temporaryForm = WL.U.title(proposal.data.targetForm || '');
      WL.FormTransformationSystem.record(root, 'temporary_form', root.body.temporaryForm, ledger.key, proposal.textPreview);
      return { status: 'APPLIED', applied: true, family: 'form', type: 'temporary', targetForm: root.body.temporaryForm, key: ledger.key };
    },
    setPermanent: function (root, proposal, ledger) {
      var form = WL.U.title(proposal.data.targetForm || 'Unknown');
      root.player.currentForm = form;
      root.player.raceSpecies = form;
      root.player.trueRace = form;
      root.body.temporaryForm = '';
      WL.FormTransformationSystem.record(root, 'permanent_form', form, ledger.key, proposal.textPreview);
      return { status: 'APPLIED', applied: true, family: 'form', type: 'permanent', targetForm: form, key: ledger.key };
    },
    addBodyTrait: function (root, proposal, ledger) {
      var trait = WL.U.title(proposal.data.trait || proposal.data.targetForm || 'Unknown Trait');
      var id = 'trait_' + WL.U.slug(trait);
      var exists = root.body.bodyTraits.some(function (t) { return t.id === id; });
      if (!exists) root.body.bodyTraits.push({ id: id, name: trait, source: 'confirmed_body_change', key: ledger.key });
      WL.U.capArray(root.body.bodyTraits, 30);
      WL.FormTransformationSystem.record(root, 'body_trait', trait, ledger.key, proposal.textPreview);
      return { status: exists ? 'DUPLICATE_TRAIT_SKIPPED' : 'APPLIED', applied: !exists, family: 'form', type: 'body_trait', trait: trait, key: ledger.key };
    },
    summary: function (root) {
      var lines = [];
      lines.push('True Race: ' + WL.U.safeString(root.player.trueRace || root.player.raceSpecies || 'unknown'));
      lines.push('Current Form: ' + WL.U.safeString(root.player.currentForm || 'unknown'));
      lines.push('Disguise: ' + (root.body.activeDisguise || '(none)'));
      lines.push('Illusion: ' + (root.body.activeIllusion || '(none)'));
      lines.push('Temporary Form: ' + (root.body.temporaryForm || '(none)'));
      lines.push('Body Traits: ' + (root.body.bodyTraits.length ? root.body.bodyTraits.map(function (t) { return t.name; }).join(', ') : '(none)'));
      return lines;
    }
  };

  WL.AbilityCostSystem = {
    ensure: function (root) {
      if (!root.abilityCosts || typeof root.abilityCosts !== 'object') root.abilityCosts = WL.Schema.root().abilityCosts;
      ['costIds', 'recent'].forEach(function (k) { if (!Array.isArray(root.abilityCosts[k])) root.abilityCosts[k] = []; });
      ['costsById', 'historyByAbilityId', 'estimatesById', 'confirmedById', 'overchargeRisksById'].forEach(function (k) { if (!root.abilityCosts[k] || typeof root.abilityCosts[k] !== 'object') root.abilityCosts[k] = {}; });
    },
    classifyDomain: function (name) {
      var n = WL.U.lower(name);
      var out = [];
      if (/fire|flame|ember|burn|inferno|pyro/.test(n)) out.push('fire');
      if (/spell|magic|mana|fireball|ward|arcane|rune|shadow|lightning|heal/.test(n)) out.push('magic');
      if (/sword|blade|slash|duel/.test(n)) out.push('swordsmanship');
      if (/box|punch|fist|martial|kick/.test(n)) out.push('boxing');
      if (/inspect|investigat|search|perceiv|study/.test(n)) out.push('investigation');
      if (/regen|recover|heal/.test(n)) out.push('healing');
      if (!out.length) out.push('general');
      return out;
    },
    estimate: function (root, abilityName, category, detail) {
      WL.AbilityCostSystem.ensure(root);
      var name = WL.U.title(abilityName || 'Unknown Ability');
      var cat = WL.U.lower(category || 'ability');
      var id = 'cost_' + WL.U.slug(cat + '_' + name);
      var domains = WL.AbilityCostSystem.classifyDomain(name);
      var mp = 0, ep = 0, hpRisk = 0, time = 'action', activation = true;
      if (cat === 'passive') { activation = false; time = 'always-on/conditional'; }
      else if (domains.indexOf('magic') >= 0 || domains.indexOf('fire') >= 0) { mp = 10; ep = 1; }
      else if (domains.indexOf('swordsmanship') >= 0 || domains.indexOf('boxing') >= 0) { ep = 5; }
      else { ep = 2; }
      if (/(blood|life force|lifeforce|sacrifice)/i.test(name)) hpRisk = 2;
      var mastery = WL.ProgressionSystem.domainTotal(root, domains);
      if (mastery >= 50 && mp > 0) mp = Math.max(1, Math.floor(mp * 0.7));
      if (mastery >= 50 && ep > 0) ep = Math.max(1, Math.floor(ep * 0.7));
      var rec = root.abilityCosts.costsById[id] || { id: id, abilityName: name, category: cat, confidence: 'rough', history: [] };
      rec.activationCost = activation ? 'rough_estimate' : 'none';
      rec.mp = mp; rec.ep = ep; rec.hpRisk = hpRisk; rec.time = time;
      rec.domains = domains; rec.notes = activation ? ['Cost hardens through observed use, mastery, fatigue, scale, and environment.'] : ['Passives have no activation cost; related actions may still cost normal effort.'];
      rec.lastDetail = detail || '';
      rec.history.push({ turn: root.runtime.currentTurn, mp: mp, ep: ep, hpRisk: hpRisk, confidence: rec.confidence });
      WL.U.capArray(rec.history, 10);
      root.abilityCosts.costsById[id] = rec;
      if (root.abilityCosts.costIds.indexOf(id) < 0) root.abilityCosts.costIds.push(id);
      root.abilityCosts.estimatesById[id] = rec;
      root.abilityCosts.recent.push({ id: id, abilityName: name, category: cat, mp: mp, ep: ep, hpRisk: hpRisk });
      WL.U.capArray(root.abilityCosts.recent, 30);
      return rec;
    },
    recordPressure: function (root, abilityName, detail) {
      WL.AbilityCostSystem.ensure(root);
      detail = detail || {};
      var name = WL.U.title(abilityName || 'Unknown Ability');
      var category = WL.AbilityLifecycleSystem && WL.AbilityLifecycleSystem.categoryForName ? WL.AbilityLifecycleSystem.categoryForName(root, name) : 'ability';
      var rec = WL.AbilityCostSystem.estimate(root, name, category, 'observed_cost_pressure');
      var words = WL.U.lower(detail.words || detail.textPreview || '');
      var hpRisk = detail.hpRisk || 0;
      var mpRisk = detail.mpRisk || 0;
      var epRisk = detail.epRisk || 0;
      if (/blood|life force|lifeforce|sacrifice|vein|heart|flesh|body burns|burns inside|burning inside/.test(words)) hpRisk = Math.max(hpRisk, 4);
      if (/mana|mind|focus|headache|arcane|spell/.test(words)) mpRisk = Math.max(mpRisk, 3);
      if (/strain|fatigue|muscle|breath|exhaust/.test(words)) epRisk = Math.max(epRisk, 3);
      if (/overcharge|overloads?|beyond|limit|maximum|too much/.test(words)) { mpRisk = Math.max(mpRisk, 2); epRisk = Math.max(epRisk, 1); }
      rec.confidence = 'rough_with_observed_pressure';
      rec.hpRisk = Math.max(rec.hpRisk || 0, hpRisk);
      rec.mpPressureRisk = Math.max(rec.mpPressureRisk || 0, mpRisk);
      rec.epPressureRisk = Math.max(rec.epPressureRisk || 0, epRisk);
      rec.overchargeRisk = Math.max(rec.overchargeRisk || 0, hpRisk + mpRisk + epRisk);
      rec.history.push({ turn: root.runtime.currentTurn, kind: 'cost_pressure', hpRisk: hpRisk, mpRisk: mpRisk, epRisk: epRisk, phrase: detail.textPreview || '', confidence: rec.confidence });
      WL.U.capArray(rec.history, 12);
      var risk = root.abilityCosts.overchargeRisksById[rec.id] || { costId: rec.id, abilityName: name, events: [] };
      risk.events.push({ turn: root.runtime.currentTurn, hpRisk: hpRisk, mpRisk: mpRisk, epRisk: epRisk, phrase: detail.textPreview || '', key: detail.key || '' });
      WL.U.capArray(risk.events, 12);
      root.abilityCosts.overchargeRisksById[rec.id] = risk;
      root.abilityCosts.costsById[rec.id] = rec;
      root.abilityCosts.estimatesById[rec.id] = rec;
      root.abilityCosts.recent.push({ id: rec.id, abilityName: name, category: category, pressure: true, hpRisk: rec.hpRisk, mpRisk: rec.mpPressureRisk, epRisk: rec.epPressureRisk });
      WL.U.capArray(root.abilityCosts.recent, 30);
      return rec;
    }
  };

  WL.ProgressionSystem = {
    ensure: function (root) {
      var fresh = WL.Schema.root().progression;
      if (!root.progression || typeof root.progression !== 'object') root.progression = WL.U.clone(fresh);
      if (!root.progression.statXp || typeof root.progression.statXp !== 'object') root.progression.statXp = { atk: 0, def: 0, spd: 0, int: 0, lck: 0 };
      ['atk', 'def', 'spd', 'int', 'lck'].forEach(function (k) { if (typeof root.progression.statXp[k] !== 'number') root.progression.statXp[k] = 0; });
      if (!root.progression.xpLanes || typeof root.progression.xpLanes !== 'object') root.progression.xpLanes = WL.U.clone(fresh.xpLanes);
      if (!root.progression.domainXp || typeof root.progression.domainXp !== 'object') root.progression.domainXp = {};
      if (!Array.isArray(root.progression.domainIds)) root.progression.domainIds = [];
      if (!Array.isArray(root.progression.recent)) root.progression.recent = [];
      if (!root.progression.buildIdentity || typeof root.progression.buildIdentity !== 'object') root.progression.buildIdentity = { tags: [], summary: 'undeveloped', noRubberBandScaling: true };
    },
    apply: function (root, validation, ledger) {
      var p = validation.proposal;
      if (p.type !== 'xp_candidate') return { status: 'IGNORED', applied: false, family: validation.family, type: validation.type, reason: 'unsupported_progression_event' };
      WL.ProgressionSystem.ensure(root);
      var d = p.data || {};
      var amount = WL.U.clamp(d.amount || 0, 0, 999999);
      var lane = WL.U.lower(d.lane || 'ability');
      if (!amount) return { status: 'NO_XP', applied: false, family: 'progression', type: 'xp', key: ledger.key };
      if (lane === 'level') { root.progression.xp += amount; root.progression.levelXp += amount; root.progression.xpLanes.level += amount; }
      else if (lane === 'skill') { root.progression.skillXp += amount; root.progression.xpLanes.skill += amount; }
      else if (lane === 'talent') { root.progression.talentXp += amount; root.progression.xpLanes.talent += amount; }
      else if (lane === 'passive') { root.progression.passiveXp += amount; root.progression.xpLanes.passive += amount; }
      else if (lane === 'stat') { root.progression.xpLanes.stat += amount; }
      else { root.progression.abilityXp += amount; root.progression.xpLanes.ability += amount; }
      var stats = d.stats || {};
      Object.keys(stats).forEach(function (k) { if (root.progression.statXp[k] != null) root.progression.statXp[k] += WL.U.clamp(stats[k], 0, 999999); });
      var domains = d.domains || [];
      for (var i = 0; i < domains.length; i++) WL.ProgressionSystem.addDomainXp(root, domains[i], amount);
      root.progression.recent.push({ turn: root.runtime.currentTurn, lane: lane, amount: amount, domains: domains, reason: d.reason || '', key: ledger.key });
      WL.U.capArray(root.progression.recent, 80);
      WL.ProgressionSystem.updateBuildIdentity(root);
      return { status: 'APPLIED', applied: true, family: 'progression', type: 'xp', lane: lane, amount: amount, key: ledger.key };
    },
    addDomainXp: function (root, domain, amount) {
      var d = WL.U.slug(domain || 'general');
      if (!root.progression.domainXp[d]) root.progression.domainXp[d] = 0;
      root.progression.domainXp[d] += WL.U.clamp(amount, 0, 999999);
      if (root.progression.domainIds.indexOf(d) < 0) root.progression.domainIds.push(d);
      root.progression.xpLanes.domain += amount;
    },
    domainTotal: function (root, domains) {
      WL.ProgressionSystem.ensure(root);
      var total = 0;
      domains = domains || [];
      for (var i = 0; i < domains.length; i++) total += root.progression.domainXp[WL.U.slug(domains[i])] || 0;
      return total;
    },
    updateBuildIdentity: function (root) {
      var tags = [];
      var sx = root.progression.statXp;
      if (sx.spd > sx.atk && sx.spd > sx.def) tags.push('mobility-focused');
      if ((root.progression.domainXp.magic || 0) + (root.progression.domainXp.fire || 0) >= 10) tags.push('magic-leaning');
      if ((root.progression.domainXp.swordsmanship || 0) >= 8) tags.push('sword-trained');
      if ((root.progression.domainXp.boxing || 0) >= 8) tags.push('martial-trained');
      if (!tags.length) tags.push('developing');
      root.progression.buildIdentity.tags = tags;
      root.progression.buildIdentity.summary = tags.join(', ');
    },
    estimateEnemyXp: function (profile) {
      profile = profile || {};
      var level = WL.U.clamp(profile.level || 1, 1, 999999);
      var xp = level * 4;
      var type = WL.U.lower(profile.type || profile.species || '');
      if (/rat/.test(type)) xp = Math.max(1, xp - 3);
      if (/soldier|guard|human/.test(type)) xp += 8;
      if (/wolf/.test(type)) xp += 4;
      if (/inferno|fire|demon|dragon|ancient|legendary|magical/.test(type)) xp += level * 3 + 20;
      if (/tiny/.test(WL.U.lower(profile.size || ''))) xp = Math.max(1, Math.floor(xp * 0.5));
      if (/trained|veteran|elite/.test(WL.U.lower(profile.training || ''))) xp += 10;
      if (/human|sapient|genius/.test(WL.U.lower(profile.intelligence || ''))) xp += 6;
      return Math.max(1, Math.floor(xp));
    },
    compareStats: function (a, b, stat) {
      stat = stat || 'spd';
      var av = Number((a && a.stats && a.stats[stat]) || 0);
      var bv = Number((b && b.stats && b.stats[stat]) || 0);
      if (av === bv) return 'tie';
      return av > bv ? 'a' : 'b';
    },
    summary: function (root) {
      WL.ProgressionSystem.ensure(root);
      var d = Object.keys(root.progression.domainXp).map(function (k) { return k + ':' + root.progression.domainXp[k]; }).join(', ') || '(none)';
      return [
        'Level: ' + root.progression.level + ' | XP: ' + root.progression.xp + '/' + root.progression.xpToNext,
        'Lanes: ability ' + root.progression.abilityXp + ', skill ' + root.progression.skillXp + ', talent ' + root.progression.talentXp + ', passive ' + root.progression.passiveXp + ', domain ' + root.progression.xpLanes.domain,
        'Stat XP: ATK ' + root.progression.statXp.atk + ', DEF ' + root.progression.statXp.def + ', SPD ' + root.progression.statXp.spd + ', INT ' + root.progression.statXp.int + ', LCK ' + root.progression.statXp.lck,
        'Domains: ' + d,
        'Build: ' + root.progression.buildIdentity.summary
      ];
    }
  };

  WL.AbilityLifecycleSystem = {
    ensure: function (root) {
      var fresh = WL.Schema.root();
      ['abilities', 'skills', 'talents', 'passives'].forEach(function (k) { if (!root[k] || typeof root[k] !== 'object') root[k] = WL.U.clone(fresh[k]); });
      if (!Array.isArray(root.abilities.inboxIds)) root.abilities.inboxIds = [];
      if (!Array.isArray(root.abilities.candidateIds)) root.abilities.candidateIds = [];
      if (!Array.isArray(root.abilities.failedIds)) root.abilities.failedIds = [];
      if (!Array.isArray(root.abilities.denied)) root.abilities.denied = [];
      if (!Array.isArray(root.abilities.transferLog)) root.abilities.transferLog = [];
    },
    abilityId: function (name) { return 'ability_' + WL.U.slug(name || 'unknown'); },
    getOrCreateInbox: function (root, name, source, phrase) {
      WL.AbilityLifecycleSystem.ensure(root);
      var display = WL.U.title(name || 'Unknown Ability');
      var id = WL.AbilityLifecycleSystem.abilityId(display);
      var rec = root.abilities.inboxById[id];
      if (!rec) {
        rec = { id: id, name: display, status: 'inbox', evidence: [], successCount: 0, failureCount: 0, trainingCount: 0, category: 'unclassified', stable: false, origin: source || 'observed' };
        root.abilities.inboxById[id] = rec;
        root.abilities.inboxIds.push(id);
      }
      if (phrase) {
        rec.evidence.push({ turn: root.runtime.currentTurn, phrase: phrase, source: source || 'observed' });
        WL.U.capArray(rec.evidence, 12);
      }
      return rec;
    },
    apply: function (root, validation, ledger) {
      var p = validation.proposal;
      if (p.type === 'ability_evidence_candidate' || p.type === 'failed_attempt_candidate' || p.type === 'training_evidence_candidate') return WL.AbilityLifecycleSystem.recordEvidence(root, p, ledger);
      if (p.type === 'category_transfer_request') return WL.AbilityLifecycleSystem.transferCategory(root, p, ledger);
      if (p.type === 'keys_update_request') return WL.AbilityLifecycleSystem.setPlayerKeys(root, p, ledger);
      if (p.type === 'cost_pressure_candidate') return WL.AbilityLifecycleSystem.recordCostPressure(root, p, ledger);
      return { status: 'IGNORED', applied: false, family: validation.family, type: validation.type, reason: 'unsupported_ability_event' };
    },
    recordEvidence: function (root, proposal, ledger) {
      var name = proposal.data.abilityName || proposal.data.targetName || proposal.targetName || 'Unknown Ability';
      var rec = WL.AbilityLifecycleSystem.getOrCreateInbox(root, name, proposal.type, proposal.textPreview);
      if (proposal.type === 'failed_attempt_candidate') {
        rec.failureCount += 1; rec.status = 'failed_candidate'; root.abilities.failedById[rec.id] = rec; if (root.abilities.failedIds.indexOf(rec.id) < 0) root.abilities.failedIds.push(rec.id);
      } else if (proposal.type === 'training_evidence_candidate') {
        rec.trainingCount += 1; rec.status = 'candidate'; root.abilities.candidatesById[rec.id] = rec; if (root.abilities.candidateIds.indexOf(rec.id) < 0) root.abilities.candidateIds.push(rec.id);
      } else {
        rec.successCount += 1; rec.status = 'candidate'; root.abilities.candidatesById[rec.id] = rec; if (root.abilities.candidateIds.indexOf(rec.id) < 0) root.abilities.candidateIds.push(rec.id);
      }
      root.abilities.recent.push({ action: 'evidence', abilityId: rec.id, type: proposal.type, key: ledger.key });
      WL.U.capArray(root.abilities.recent, 50);
      return { status: 'APPLIED', applied: true, family: 'ability', type: 'evidence', abilityId: rec.id, key: ledger.key };
    },
    transferCategory: function (root, proposal, ledger) {
      var name = WL.U.title(proposal.data.abilityName || proposal.data.targetName || 'Unknown Ability');
      var category = WL.U.lower(proposal.data.category || 'ability');
      var rec = WL.AbilityLifecycleSystem.getOrCreateInbox(root, name, 'category_transfer_request', proposal.textPreview);
      var verdict = WL.AbilityLifecycleSystem.validateTransfer(root, rec, category);
      var log = { turn: root.runtime.currentTurn, abilityId: rec.id, name: name, category: category, status: verdict.status, reason: verdict.reason, key: ledger.key };
      root.abilities.transferLog.push(log); WL.U.capArray(root.abilities.transferLog, 60);
      if (verdict.status === 'denied') { root.abilities.denied.push(log); WL.U.capArray(root.abilities.denied, 40); return { status: 'DENIED', applied: true, family: 'ability', type: 'category_transfer', abilityId: rec.id, category: category, reason: verdict.reason, key: ledger.key }; }
      if (category === 'skill') WL.AbilityLifecycleSystem.installSkill(root, rec, verdict);
      else if (category === 'talent') WL.AbilityLifecycleSystem.installTalent(root, rec, verdict);
      else if (category === 'passive') WL.AbilityLifecycleSystem.installPassive(root, rec, verdict);
      else WL.AbilityLifecycleSystem.installAbility(root, rec, verdict);
      WL.AbilityCostSystem.estimate(root, name, category, verdict.reason);
      return { status: verdict.status === 'unstable' ? 'APPLIED_UNSTABLE' : 'APPLIED', applied: true, family: 'ability', type: 'category_transfer', abilityId: rec.id, category: category, key: ledger.key };
    },
    validateTransfer: function (root, rec, category) {
      var name = WL.U.lower(rec.name);
      if (category === 'passive') {
        if (/fireball|meteor|lightning bolt|explosion|beam|blast/.test(name)) return { status: 'denied', reason: 'active_attack_spell_is_not_a_normal_passive' };
        if (/regen|regeneration|healing factor/.test(name)) return { status: 'weak', reason: 'passive_allowed_as_weak_adaptation_until_evidence_grows' };
        if (/boxing|swordsmanship|martial|stance|discipline/.test(name)) return { status: 'stable', reason: 'discipline_style_passive_is_plausible' };
        return { status: 'weak', reason: 'passive_candidate_needs_exposure_or_conditioning' };
      }
      if (category === 'skill') {
        if (rec.successCount <= 0 && rec.trainingCount <= 0) return { status: 'unstable', reason: 'skill_created_unstable_until_success_or_training_evidence' };
        return { status: 'stable', reason: 'skill_supported_by_observed_use_or_training' };
      }
      if (category === 'talent') {
        if (/sword|box|martial|stance|style|form|duel/.test(name) || rec.trainingCount > 0) return { status: 'stable', reason: 'talent_supported_by_style_or_training' };
        return { status: 'unstable', reason: 'talent_needs_style_evidence' };
      }
      return { status: 'stable', reason: 'general_ability_recorded' };
    },
    installSkill: function (root, rec, verdict) {
      var id = 'skill_' + WL.U.slug(rec.name);
      var target = verdict.status === 'unstable' ? root.skills.unstableById : root.skills.skillsById;
      var ids = verdict.status === 'unstable' ? root.skills.unstableIds : root.skills.skillIds;
      target[id] = { id: id, name: rec.name, level: 1, xp: 0, status: verdict.status, originAbilityId: rec.id, activationName: rec.name, costId: 'cost_skill_' + WL.U.slug(rec.name), notes: [verdict.reason] };
      if (ids.indexOf(id) < 0) ids.push(id);
    },
    installTalent: function (root, rec, verdict) {
      var id = 'talent_' + WL.U.slug(rec.name);
      root.talents.talentsById[id] = { id: id, name: rec.name, level: 1, xp: 0, status: verdict.status, originAbilityId: rec.id, keys: [rec.name], notes: [verdict.reason] };
      if (root.talents.talentIds.indexOf(id) < 0) root.talents.talentIds.push(id);
      root.talents.keysByTalentId[id] = [rec.name];
    },
    installPassive: function (root, rec, verdict) {
      var id = 'passive_' + WL.U.slug(rec.name);
      var obj = { id: id, name: rec.name, level: 1, xp: 0, status: verdict.status, originAbilityId: rec.id, activationCost: 'none', keys: [rec.name], notes: [verdict.reason] };
      if (verdict.status === 'weak') { root.passives.weakCandidateById[id] = obj; if (root.passives.weakCandidateIds.indexOf(id) < 0) root.passives.weakCandidateIds.push(id); }
      else { root.passives.passivesById[id] = obj; if (root.passives.passiveIds.indexOf(id) < 0) root.passives.passiveIds.push(id); }
      root.passives.keysByPassiveId[id] = [rec.name];
    },
    installAbility: function (root, rec, verdict) {
      root.abilities.abilitiesById[rec.id] = { id: rec.id, name: rec.name, level: 1, xp: 0, status: verdict.status, stable: verdict.status === 'stable', notes: [verdict.reason], keys: root.abilities.keysByAbilityId[rec.id] || [] };
      if (root.abilities.abilityIds.indexOf(rec.id) < 0) root.abilities.abilityIds.push(rec.id);
    },
    categoryForName: function (root, name) {
      WL.AbilityLifecycleSystem.ensure(root);
      var slug = WL.U.slug(name || '');
      if (root.skills.skillsById['skill_' + slug] || root.skills.unstableById['skill_' + slug]) return 'skill';
      if (root.talents.talentsById['talent_' + slug]) return 'talent';
      if (root.passives.passivesById['passive_' + slug] || root.passives.weakCandidateById['passive_' + slug]) return 'passive';
      if (root.abilities.abilitiesById['ability_' + slug]) return 'ability';
      return 'ability';
    },
    cleanKeys: function (keys) {
      var raw = Array.isArray(keys) ? keys : String(keys || '').split(/[,;|]/);
      var out = [];
      var seen = {};
      for (var i = 0; i < raw.length; i++) {
        var k = WL.U.norm(raw[i]).replace(/^(?:keys?|triggers?)\s*[:=]\s*/i, '').replace(/^and\s+/i, '').trim();
        if (!k || k.length > 48) continue;
        if (/^(?:story card trigger|trigger|key)$/i.test(k)) continue;
        var id = WL.U.slug(k);
        if (seen[id]) continue;
        seen[id] = true;
        out.push(k);
      }
      return out.slice(0, 12);
    },
    setPlayerKeys: function (root, proposal, ledger) {
      var name = WL.U.title(proposal.data.abilityName || proposal.data.targetName || 'Unknown Ability');
      var keys = WL.AbilityLifecycleSystem.cleanKeys(proposal.data.keys || []);
      if (!keys.length) return { status: 'NO_KEYS', applied: false, family: 'ability', type: 'keys_update', abilityId: '', key: ledger.key };
      var rec = WL.AbilityLifecycleSystem.getOrCreateInbox(root, name, 'keys_update_request', proposal.textPreview);
      rec.keys = keys.slice();
      root.abilities.keysByAbilityId[rec.id] = keys.slice();
      if (root.abilities.abilitiesById[rec.id]) root.abilities.abilitiesById[rec.id].keys = keys.slice();
      var slug = WL.U.slug(name);
      var skillId = 'skill_' + slug;
      if (root.skills.skillsById[skillId]) { root.skills.skillsById[skillId].keys = keys.slice(); root.skills.keysBySkillId[skillId] = keys.slice(); }
      if (root.skills.unstableById[skillId]) { root.skills.unstableById[skillId].keys = keys.slice(); root.skills.keysBySkillId[skillId] = keys.slice(); }
      var talentId = 'talent_' + slug;
      if (root.talents.talentsById[talentId]) { root.talents.talentsById[talentId].keys = keys.slice(); root.talents.keysByTalentId[talentId] = keys.slice(); }
      var passiveId = 'passive_' + slug;
      if (root.passives.passivesById[passiveId]) { root.passives.passivesById[passiveId].keys = keys.slice(); root.passives.keysByPassiveId[passiveId] = keys.slice(); }
      if (root.passives.weakCandidateById[passiveId]) { root.passives.weakCandidateById[passiveId].keys = keys.slice(); root.passives.keysByPassiveId[passiveId] = keys.slice(); }
      root.abilities.recent.push({ action: 'keys_update', abilityId: rec.id, keys: keys.slice(), key: ledger.key });
      WL.U.capArray(root.abilities.recent, 50);
      return { status: 'APPLIED', applied: true, family: 'ability', type: 'keys_update', abilityId: rec.id, keys: keys.slice(), key: ledger.key };
    },
    recordCostPressure: function (root, proposal, ledger) {
      var name = WL.U.title(proposal.data.abilityName || proposal.data.targetName || 'Unknown Ability');
      var rec = WL.AbilityLifecycleSystem.getOrCreateInbox(root, name, 'cost_pressure_candidate', proposal.textPreview);
      rec.evidence.push({ turn: root.runtime.currentTurn, phrase: proposal.textPreview, source: 'cost_pressure' });
      WL.U.capArray(rec.evidence, 12);
      var cost = WL.AbilityCostSystem.recordPressure(root, name, { words: proposal.data.riskWords || proposal.textPreview, textPreview: proposal.textPreview, hpRisk: proposal.data.hpRisk || 0, mpRisk: proposal.data.mpRisk || 0, epRisk: proposal.data.epRisk || 0, key: ledger.key });
      return { status: 'APPLIED', applied: true, family: 'ability', type: 'cost_pressure', abilityId: rec.id, costId: cost.id, key: ledger.key };
    },
    summary: function (root) {
      WL.AbilityLifecycleSystem.ensure(root);
      var lines = [];
      lines.push('Inbox: ' + (root.abilities.inboxIds.map(function (id) { return root.abilities.inboxById[id].name + ' [' + root.abilities.inboxById[id].status + ']'; }).join(', ') || '(none)'));
      lines.push('Skills: ' + (root.skills.skillIds.map(function (id) { return root.skills.skillsById[id].name; }).join(', ') || '(none)') + ' | Unstable: ' + (root.skills.unstableIds.map(function (id) { return root.skills.unstableById[id].name; }).join(', ') || '(none)'));
      lines.push('Talents: ' + (root.talents.talentIds.map(function (id) { return root.talents.talentsById[id].name; }).join(', ') || '(none)'));
      lines.push('Passives: ' + (root.passives.passiveIds.map(function (id) { return root.passives.passivesById[id].name; }).join(', ') || '(none)') + ' | Weak: ' + (root.passives.weakCandidateIds.map(function (id) { return root.passives.weakCandidateById[id].name; }).join(', ') || '(none)'));
      lines.push('Denied transfers: ' + root.abilities.denied.length);
      return lines;
    }
  };

  WL.EventScanner = {
    makeEvent: function (family, type, clause, source, data) {
      var c = WL.U.norm(clause);
      return {
        family: family,
        type: type,
        sourceHook: source.hook,
        sourceKind: source.kind,
        sourceHash: WL.U.hash(source.hook + '|' + c),
        localClauseHash: WL.U.hash(c),
        textPreview: c.slice(0, 220),
        data: data || {},
        confidence: data && data.confidence || 'candidate'
      };
    },
    scanInput: function (text, root, source) {
      var out = [];
      var clauses = WL.U.splitClauses(text);
      for (var i = 0; i < clauses.length; i++) {
        out.push(WL.EventScanner.makeEvent('input', 'intent_observed', clauses[i], source, { confidence: 'intent_only' }));
        var abilityEvents = WL.EventScanner.detectAbilityProgressionCandidates(clauses[i], 'input');
        for (var a = 0; a < abilityEvents.length; a++) out.push(WL.EventScanner.makeEvent(abilityEvents[a].family, abilityEvents[a].type, clauses[i], source, abilityEvents[a]));
      }
      WL.EventScanner.record(root, out);
      return out;
    },
    scanOutput: function (text, root, source) {
      var out = [];
      var clauses = WL.U.splitClauses(text);
      if (!clauses.length && WL.U.norm(text)) clauses = [WL.U.norm(text)];
      for (var i = 0; i < clauses.length; i++) {
        var clause = clauses[i];
        out.push(WL.EventScanner.makeEvent('system', 'output_observed', clause, source, { confidence: 'confirmed_output' }));

        var payment = WL.EventScanner.detectCurrencyPayment(clause);
        if (payment) {
          out.push(WL.EventScanner.makeEvent('inventory', 'payment_candidate', clause, source, payment));
          out.push(WL.EventScanner.makeEvent('quest', 'reward_payment_candidate', clause, source, payment));
        }

        var quest = WL.EventScanner.detectQuestCandidate(clause);
        if (quest) out.push(WL.EventScanner.makeEvent('quest', quest.type, clause, source, quest));

        var inventoryHits = WL.EventScanner.detectInventoryCandidates(clause);
        for (var h = 0; h < inventoryHits.length; h++) {
          out.push(WL.EventScanner.makeEvent('inventory', inventoryHits[h].type, clause, source, inventoryHits[h]));
        }

        var formHits = WL.EventScanner.detectFormBodyCandidates(clause);
        for (var f = 0; f < formHits.length; f++) {
          out.push(WL.EventScanner.makeEvent('form', formHits[f].type, clause, source, formHits[f]));
        }

        var abilityEvents = WL.EventScanner.detectAbilityProgressionCandidates(clause, 'output');
        for (var a = 0; a < abilityEvents.length; a++) {
          out.push(WL.EventScanner.makeEvent(abilityEvents[a].family, abilityEvents[a].type, clause, source, abilityEvents[a]));
        }

        var timeEvents = WL.EventScanner.detectTimeTravelCandidates(clause);
        for (var t = 0; t < timeEvents.length; t++) {
          out.push(WL.EventScanner.makeEvent('time', timeEvents[t].type, clause, source, timeEvents[t]));
        }

        var placeEvents = WL.EventScanner.detectWorldPlaceCandidates(clause);
        for (var p = 0; p < placeEvents.length; p++) {
          out.push(WL.EventScanner.makeEvent('place', placeEvents[p].type, clause, source, placeEvents[p]));
        }

        var actorEvents = WL.EventScanner.detectActorSocialCandidates(clause);
        for (var ac = 0; ac < actorEvents.length; ac++) {
          out.push(WL.EventScanner.makeEvent('actor', actorEvents[ac].type, clause, source, actorEvents[ac]));
        }
      }
      WL.EventScanner.record(root, out);
      return out;
    },
    hasNegativeAcquisition: function (clause) {
      var lower = WL.U.lower(clause);
      return /\b(almost|tries? to|attempts? to|fails? to|failed to|refuses? to|declines? to|cannot|can't|couldn't|doesn't|didn't|never)\b/.test(lower) ||
        /\b(reaches? for|reached for|looks? at|sees?|saw|notices?|noticed|wants?|needs?|searches? for|promises?|objective|quest|task|mission|reward)\b/.test(lower) ||
        /\btake(?:s)?\s+(?:a\s+|the\s+)?(?:long\s+)?look\b/.test(lower) ||
        /\b(but|however)\b[\s\S]*\b(snatches?|takes?|pulls?|knocks?|steals?|away|out of reach)\b/.test(lower);
    },
    detectCurrencyPayment: function (clause) {
      var c = WL.U.norm(clause);
      var m = c.match(/\b([A-Z][A-Za-z' -]{1,50})\s+(?:pays?|paid|hands?|gives?|gave|tosses?)\s+you\s+(\d{1,6})\s+(copper|silver|gold)\b/i);
      if (!m) m = c.match(/\byou\s+(?:are\s+)?(?:paid|given)\s+(\d{1,6})\s+(copper|silver|gold)\b(?:\s+by\s+([A-Z][A-Za-z' -]{1,50}))?/i);
      if (!m) return null;
      var payer = '';
      var amount = 0;
      var currency = '';
      if (/^you\s/i.test(c)) {
        amount = Number(m[1] || 0); currency = WL.U.lower(m[2] || 'gold'); payer = WL.U.title(m[3] || 'unknown payer');
      } else {
        payer = WL.U.title(m[1] || 'unknown payer'); amount = Number(m[2] || 0); currency = WL.U.lower(m[3] || 'gold');
      }
      if (!amount || amount < 1) return null;
      return { payer: payer, amount: amount, currency: currency, confidence: 'confirmed_payment_phrase' };
    },
    detectInventoryCandidate: function (clause) {
      var list = WL.EventScanner.detectInventoryCandidates(clause);
      return list.length ? list[0] : null;
    },
    detectInventoryCandidates: function (clause) {
      var c = WL.U.norm(clause);
      var out = [];
      if (WL.EventScanner.hasNegativeAcquisition(c)) return out;
      if (WL.EventScanner.detectCurrencyPayment(c)) return out;
      var m = c.match(/\b(?:pick(?:s)? up|grab(?:s)?|take(?:s)? hold of|take(?:s)?|loot(?:s)?|receive(?:s)?|pocket(?:s)?|snatch(?:es)? up|scoop(?:s)? up|secure(?:s)?|lift(?:s)?|catch(?:es)?)\s+(?:the\s+|a\s+|an\s+)?([^.!?;,]{2,100})/i);
      if (!m) m = c.match(/\b(?:hands?|gives?|gave|tosses?)\s+you\s+(?:the\s+|a\s+|an\s+)?([^.!?;,]{2,100})/i);
      if (!m) m = c.match(/\b(?:places?|sets?|slips?)\s+(?:the\s+|a\s+|an\s+)?([^.!?;,]{2,100}?)\s+(?:in|into|onto)\s+your\s+(?:hand|hands|palm|pack|bag|inventory)\b/i);
      if (!m) m = c.match(/\byou\s+(?:are\s+)?given\s+(?:the\s+|a\s+|an\s+)?([^.!?;,]{2,100})/i);
      if (!m) return out;
      var parts = WL.EventScanner.splitCandidateTargets(m[1]);
      var seen = {};
      for (var i = 0; i < parts.length; i++) {
        var target = WL.EventScanner.cleanCandidateTarget(parts[i]);
        if (!target) continue;
        var id = 'item_' + WL.U.slug(target);
        if (seen[id]) continue;
        seen[id] = true;
        out.push({ type: 'acquire_candidate', targetName: target, targetId: id, confidence: 'confirmed_acquisition_phrase' });
      }
      return out;
    },
    splitCandidateTargets: function (target) {
      var t = WL.U.norm(target);
      if (!t) return [];
      t = t.replace(/\s+(?:by|from)\s+[A-Z][A-Za-z' -]{1,50}\s*$/i, '');
      t = t.replace(/\s+(?:before|while|as|when|but|and then|from|off|out of)\b[\s\S]*$/i, '');
      t = t.replace(/\s+(?:in|into|onto)\s+your\s+(?:hand|hands|palm|pack|bag|inventory)\b[\s\S]*$/i, '');
      var raw = t.split(/\s*,\s*|\s+and\s+(?=(?:the\s+|a\s+|an\s+)?[a-z][a-z' -]{1,60}$)/i);
      var out = [];
      for (var i = 0; i < raw.length; i++) {
        var p = WL.U.norm(raw[i]).replace(/^(?:the|a|an)\s+/i, '');
        if (p) out.push(p);
      }

      m = c.match(/^([A-Z][A-Za-z' -]{1,50})\s+asks?\s+you\s+to\s+([^.!?;]{3,180}?)(?:\s+and\s+promises?\s+(\d{1,6})\s+(copper|silver|gold))?[.!?]?$/i);
      if (!m) m = c.match(/^([A-Z][A-Za-z' -]{1,50})\s+offers?\s+(?:you\s+)?(?:a\s+)?(?:quest|task|mission)\s+to\s+([^.!?;]{3,180}?)(?:\s+for\s+(\d{1,6})\s+(copper|silver|gold))?[.!?]?$/i);
      if (m) {
        giver = WL.U.title(m[1]);
        objective = WL.EventScanner.cleanQuestObjective(m[2]);
        reward = m[3] ? { amount: Number(m[3]), currency: WL.U.lower(m[4] || 'gold'), payer: giver } : null;
        return { type: 'offer_candidate', giver: giver, objective: objective, title: WL.EventScanner.makeQuestTitle(objective), reward: reward, confidence: 'confirmed_quest_offer' };
      }
      m = c.match(/\byou\s+accept\s+([A-Z][A-Za-z' -]{1,50})'s\s+(?:quest|task|mission)\s+to\s+([^.!?;]{3,180})[.!?]?/i);
      if (!m) m = c.match(/\byou\s+accept\s+(?:the\s+)?(?:quest|task|mission)\s+to\s+([^.!?;]{3,180})/i);
      if (m) {
        if (m.length >= 3) { giver = WL.U.title(m[1]); objective = WL.EventScanner.cleanQuestObjective(m[2]); }
        else { giver = ''; objective = WL.EventScanner.cleanQuestObjective(m[1]); }
        return { type: 'accept_candidate', giver: giver, objective: objective, title: WL.EventScanner.makeQuestTitle(objective), confidence: 'confirmed_quest_acceptance' };
      }
      if (/\b(quest|task|mission)\b/.test(lower) && /\b(complete|completed|finished|failed|abandoned)\b/.test(lower)) {
        return { type: 'status_candidate', objective: WL.EventScanner.cleanQuestObjective(c), title: WL.EventScanner.makeQuestTitle(c), confidence: 'quest_status_language' };
      }
      return null;
    },
    cleanQuestObjective: function (objective) {
      var o = WL.U.norm(objective)
        .replace(/\s+and\s+promises?\s+\d{1,6}\s+(?:copper|silver|gold)\b[\s\S]*$/i, '')
        .replace(/\s+for\s+\d{1,6}\s+(?:copper|silver|gold)\b[\s\S]*$/i, '')
        .replace(/^(?:the\s+)?(?:quest|task|mission)\s+to\s+/i, '')
        .replace(/^resolve\s+/i, '')
        .replace(/\s+/g, ' ')
        .trim();
      return o || 'unspecified objective';
    },
    makeQuestTitle: function (objective) {
      var o = WL.EventScanner.cleanQuestObjective(objective);
      o = o.replace(/^(?:to\s+)?/i, '');
      return WL.U.title(o).replace(/^(?:Resolve\s+)+/i, '') || 'Untitled Quest';
    },
    record: function (root, events) {
      for (var i = 0; i < events.length; i++) root.diagnostics.parser.push(events[i]);
      WL.U.capArray(root.diagnostics.parser, WL.Constants.maxParserHits);
    }
  };

  WL.ProposalBuilder = {
    fromEvents: function (root, events, source) {
      var out = [];
      var seen = {};
      for (var i = 0; i < events.length; i++) {
        var ev = events[i];
        var target = ev.data && (ev.data.targetId || ev.data.targetName || ev.data.objective || ev.data.title || ev.data.currency || ev.type) || '';

        var localKey = [ev.family, ev.type, ev.sourceHash, ev.localClauseHash, targetSlug].join('|');
        if (seen[localKey]) continue;
        seen[localKey] = true;
        var proposal = {
          id: 'proposal_' + WL.U.hash(localKey + '|' + i),
          family: ev.family,
          type: ev.type,
          sourceHook: ev.sourceHook,
          sourceKind: ev.sourceKind,
          sourceHash: ev.sourceHash,
          localClauseHash: ev.localClauseHash,
          targetId: targetSlug,
          targetName: ev.data && ev.data.targetName || ev.data && ev.data.title || '',
          textPreview: ev.textPreview,
          data: ev.data || {},
          status: 'proposed',
          opKey: WL.OperationLedger.makeKey([ev.sourceHook, ev.sourceHash, ev.localClauseHash, ev.family, ev.type, targetSlug || 'none'])
        };
        out.push(proposal);
        root.diagnostics.proposals.push(proposal);
      }
      WL.U.capArray(root.diagnostics.proposals, WL.Constants.maxProposals);
      return out;
    }
  };

  WL.CanonValidator = {
    validateAll: function (root, proposals, source) {
      var out = [];
      for (var i = 0; i < proposals.length; i++) out.push(WL.CanonValidator.validate(root, proposals[i], source));
      for (var j = 0; j < out.length; j++) root.diagnostics.validations.push(out[j]);
      WL.U.capArray(root.diagnostics.validations, WL.Constants.maxValidations);
      return out;
    },
    validate: function (root, proposal, source) {
      var result = {
        id: 'validation_' + WL.U.hash(proposal.id + '|' + proposal.opKey),
        proposalId: proposal.id,
        proposal: proposal,
        family: proposal.family,
        type: proposal.type,
        accepted: false,
        status: 'REJECTED',
        reason: '',
        reducer: ''
      };
      var isAbilityManagementInput = source && source.kind === 'input' && proposal.family === 'ability' && (proposal.type === 'category_transfer_request' || proposal.type === 'keys_update_request');
      if ((!source || !source.eligibleForCanon) && !isAbilityManagementInput) {
        result.status = source && source.kind === 'input' ? 'PENDING' : 'REJECTED';
        result.reason = source && source.reason || 'source_not_eligible';
        return result;
      }
      if (proposal.family === 'system' && proposal.type === 'output_observed') {
        result.accepted = true;
        result.status = 'ACCEPTED';
        result.reason = 'pipeline_observation_only';
        result.reducer = 'SystemObservationReducer';
        return result;
      }
      if (proposal.family === 'inventory') {
        if (proposal.type === 'acquire_candidate' && proposal.data && proposal.data.targetName) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_output_acquisition'; result.reducer = 'InventoryEquipmentReducer'; return result;
        }
        if (proposal.type === 'payment_candidate' && proposal.data && proposal.data.amount > 0 && proposal.data.currency) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_output_payment'; result.reducer = 'InventoryEquipmentReducer'; return result;
        }
      }
      if (proposal.family === 'quest') {
        if (proposal.type === 'offer_candidate' && proposal.data && proposal.data.objective) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_quest_offer'; result.reducer = 'QuestObjectiveRewardReducer'; return result;
        }
        if (proposal.type === 'accept_candidate' && proposal.data && proposal.data.objective) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_quest_acceptance'; result.reducer = 'QuestObjectiveRewardReducer'; return result;
        }
        if (proposal.type === 'reward_payment_candidate' && proposal.data && proposal.data.amount > 0) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_reward_payment_record'; result.reducer = 'QuestObjectiveRewardReducer'; return result;
        }
      }
      if (proposal.family === 'form') {
        if (proposal.data && (proposal.data.targetForm || proposal.data.trait)) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_form_or_body_evidence'; result.reducer = 'FormTransformationReducer'; return result;
        }
      }
      if (proposal.family === 'ability') {
        if (proposal.type === 'category_transfer_request' && proposal.data && proposal.data.abilityName) {
          result.accepted = true; result.status = isAbilityManagementInput ? 'ACCEPTED_MANAGEMENT_INPUT' : 'ACCEPTED'; result.reason = 'validated_ability_category_transfer'; result.reducer = 'AbilityLifecycleReducer'; return result;
        }
        if (proposal.type === 'keys_update_request' && proposal.data && proposal.data.abilityName && proposal.data.keys && proposal.data.keys.length) {
          result.accepted = true; result.status = isAbilityManagementInput ? 'ACCEPTED_MANAGEMENT_INPUT' : 'ACCEPTED'; result.reason = 'validated_player_authored_keys_update'; result.reducer = 'AbilityLifecycleReducer'; return result;
        }
        if (proposal.type === 'cost_pressure_candidate' && source && source.eligibleForCanon && proposal.data && proposal.data.abilityName) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_overcharge_cost_pressure'; result.reducer = 'AbilityLifecycleReducer'; return result;
        }
        if (proposal.data && proposal.data.abilityName) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_ability_evidence'; result.reducer = 'AbilityLifecycleReducer'; return result;
        }
      }
      if (proposal.family === 'progression') {
        if (source && source.eligibleForCanon && proposal.type === 'xp_candidate' && proposal.data && proposal.data.amount > 0) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_output_progression'; result.reducer = 'ProgressionReducer'; return result;
        }
      }
      if (proposal.family === 'time') {
        if (source && source.eligibleForCanon && proposal.data) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_time_route_deadline_evidence'; result.reducer = 'TimeTravelReducer'; return result;
        }
      }
      if (proposal.family === 'place') {
        if (source && source.eligibleForCanon && proposal.data) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_world_place_evidence'; result.reducer = 'WorldStructureReducer'; return result;
        }
      }
      if (proposal.family === 'actor') {
        if (source && source.eligibleForCanon && proposal.data) {
          result.accepted = true; result.status = 'ACCEPTED'; result.reason = 'confirmed_actor_social_evidence'; result.reducer = 'ActorSocialReducer'; return result;
        }
      }
      result.status = 'PENDING';
      result.reason = 'no_owner_for_family_in_build6';
      return result;
    }
  };

  WL.ReducerRegistry = {
    handlers: {},
    applyAcceptedBatch: function (root, validations, source) {
      var results = [];
      for (var i = 0; i < validations.length; i++) {
        var v = validations[i];
        if (!v.accepted) {
          results.push({ status: v.status, applied: false, reason: v.reason, family: v.family, type: v.type, targetId: v.proposal && v.proposal.targetId || '' });
          continue;
        }
        var key = v.proposal.opKey;
        var ledger = WL.OperationLedger.add(root, key, v.reducer || v.family);
        if (!ledger.applied) {
          results.push({ status: 'DUPLICATE_SKIPPED', applied: false, duplicate: true, key: ledger.key, family: v.family, type: v.type });
          continue;
        }
        var handler = WL.ReducerRegistry.handlers[v.reducer];
        if (typeof handler !== 'function') {
          results.push({ status: 'NO_HANDLER', applied: false, key: ledger.key, family: v.family, type: v.type, reason: 'handler_missing' });
          continue;
        }
        var h = handler(root, v, ledger);
        results.push(h);
      }
      for (var j = 0; j < results.length; j++) root.diagnostics.reducers.push(results[j]);
      WL.U.capArray(root.diagnostics.reducers, WL.Constants.maxReducerResults);
      return results;
    }
  };

  WL.InventoryEquipmentSystem = {
    apply: function (root, validation, ledger) {
      var proposal = validation.proposal;
      if (proposal.type === 'acquire_candidate') return WL.InventoryEquipmentSystem.acquire(root, proposal, ledger);
      if (proposal.type === 'payment_candidate') return WL.InventoryEquipmentSystem.pay(root, proposal, ledger);
      return { status: 'IGNORED', applied: false, family: validation.family, type: validation.type, reason: 'unsupported_inventory_event' };
    },
    isStackable: function (name) {
      return /\b(coin|coins|arrow|arrows|bolt|bolts|ration|rations|herb|herbs|potion|potions)\b/i.test(name || '');
    },
    acquire: function (root, proposal, ledger) {
      var name = WL.U.title(proposal.data.targetName || proposal.targetName || 'Unknown Item');
      var id = proposal.data.targetId || ('item_' + WL.U.slug(name));
      if (!name || /\bquest\b/i.test(name)) return { status: 'REJECTED_BAD_ITEM_NAME', applied: false, family: 'inventory', type: 'acquire_candidate', key: ledger.key };
      var item = root.inventory.itemsById[id];
      if (!item) {
        item = {
          id: id,
          displayName: name,
          quantity: 1,
          stackable: WL.InventoryEquipmentSystem.isStackable(name),
          condition: 'unknown',
          custody: 'owned_by_player',
          equippedSlot: '',
          sourceHistory: []
        };
        root.inventory.itemsById[id] = item;
        root.inventory.ownedIds.push(id);
      } else if (item.stackable) {
        item.quantity = WL.U.clamp(Number(item.quantity || 1) + 1, 1, 999999);
      }
      item.sourceHistory.push({ key: ledger.key, sourceHash: proposal.sourceHash, phrase: proposal.textPreview });
      WL.U.capArray(item.sourceHistory, 12);
      root.inventory.log.push({ turn: root.runtime.currentTurn, action: 'acquire', itemId: id, name: name, key: ledger.key });
      WL.U.capArray(root.inventory.log, 80);
      return { status: 'APPLIED', applied: true, family: 'inventory', type: 'acquire', itemId: id, targetName: name, key: ledger.key };
    },
    pay: function (root, proposal, ledger) {
      var amount = WL.U.clamp(proposal.data.amount, 1, 999999);
      var currency = WL.U.lower(proposal.data.currency || 'gold');
      if (!root.inventory.currency[currency] && root.inventory.currency[currency] !== 0) root.inventory.currency[currency] = 0;
      root.inventory.currency[currency] += amount;
      root.inventory.log.push({ turn: root.runtime.currentTurn, action: 'payment', amount: amount, currency: currency, payer: proposal.data.payer || '', key: ledger.key });
      WL.U.capArray(root.inventory.log, 80);
      return { status: 'APPLIED', applied: true, family: 'inventory', type: 'payment', amount: amount, currency: currency, key: ledger.key };
    },
    summary: function (root) {
      var lines = [];
      for (var i = 0; i < root.inventory.ownedIds.length; i++) {
        var id = root.inventory.ownedIds[i];
        var item = root.inventory.itemsById[id];
        if (item) lines.push(item.displayName + ' x' + (item.quantity || 1));
      }
      if (!lines.length) lines.push('(none)');
      lines.push('Currency: ' + (root.inventory.currency.gold || 0) + ' gold, ' + (root.inventory.currency.silver || 0) + ' silver, ' + (root.inventory.currency.copper || 0) + ' copper');
      return lines;
    }
  };

  WL.QuestObjectiveRewardSystem = {
    apply: function (root, validation, ledger) {
      var proposal = validation.proposal;
      if (proposal.type === 'offer_candidate') return WL.QuestObjectiveRewardSystem.offer(root, proposal, ledger);
      if (proposal.type === 'accept_candidate') return WL.QuestObjectiveRewardSystem.accept(root, proposal, ledger);
      if (proposal.type === 'reward_payment_candidate') return WL.QuestObjectiveRewardSystem.recordPayment(root, proposal, ledger);
      return { status: 'IGNORED', applied: false, family: validation.family, type: validation.type, reason: 'unsupported_quest_event' };
    },
    questId: function (giver, objective) {
      return 'quest_' + WL.U.slug((giver || 'unknown') + '_' + (objective || 'objective'));
    },
    findByObjective: function (root, giver, objective) {
      var id = WL.QuestObjectiveRewardSystem.questId(giver || '', objective || '');
      if (root.quests.questsById[id]) return id;
      var targetSlug = WL.U.slug(objective || '');
      var keys = Object.keys(root.quests.questsById);
      for (var i = 0; i < keys.length; i++) {
        var q = root.quests.questsById[keys[i]];
        if (WL.U.slug(q.objective || '') === targetSlug && (!giver || !q.giver || WL.U.slug(q.giver) === WL.U.slug(giver))) return keys[i];
      }
      return '';
    },
    ensureQuest: function (root, data) {
      var giver = WL.U.title(data.giver || 'Unknown');
      var objective = WL.EventScanner.cleanQuestObjective(data.objective || data.title || 'unspecified objective');
      var id = WL.QuestObjectiveRewardSystem.findByObjective(root, giver, objective) || WL.QuestObjectiveRewardSystem.questId(giver, objective);
      var q = root.quests.questsById[id];
      if (!q) {
        q = { id: id, title: WL.EventScanner.makeQuestTitle(objective), giver: giver, objective: objective, status: 'offered', rewardsPromised: [], rewardsPaid: [], history: [] };
        root.quests.questsById[id] = q;
      }
      return q;
    },
    offer: function (root, proposal, ledger) {
      var q = WL.QuestObjectiveRewardSystem.ensureQuest(root, proposal.data);
      q.status = q.status || 'offered';
      if (root.quests.offeredIds.indexOf(q.id) < 0 && root.quests.activeIds.indexOf(q.id) < 0) root.quests.offeredIds.push(q.id);
      if (proposal.data.reward && proposal.data.reward.amount > 0) {
        var rid = 'reward_' + WL.U.hash(q.id + '|' + proposal.data.reward.amount + '|' + proposal.data.reward.currency);
        var exists = q.rewardsPromised.some(function (r) { return r.id === rid; });
        if (!exists) {
          var rec = { id: rid, questId: q.id, amount: proposal.data.reward.amount, currency: proposal.data.reward.currency, payer: proposal.data.reward.payer || q.giver, paid: false };
          q.rewardsPromised.push(rec);
          root.quests.rewardPromises.push(rec);
        }
      }
      q.history.push({ key: ledger.key, action: 'offer', phrase: proposal.textPreview });
      WL.U.capArray(q.history, 20);
      return { status: 'APPLIED', applied: true, family: 'quest', type: 'offer', questId: q.id, key: ledger.key };
    },
    accept: function (root, proposal, ledger) {
      var q = WL.QuestObjectiveRewardSystem.ensureQuest(root, proposal.data);
      q.status = 'active';
      var oi = root.quests.offeredIds.indexOf(q.id);
      if (oi >= 0) root.quests.offeredIds.splice(oi, 1);
      if (root.quests.activeIds.indexOf(q.id) < 0) root.quests.activeIds.push(q.id);
      q.history.push({ key: ledger.key, action: 'accept', phrase: proposal.textPreview });
      WL.U.capArray(q.history, 20);
      return { status: 'APPLIED', applied: true, family: 'quest', type: 'accept', questId: q.id, key: ledger.key };
    },
    recordPayment: function (root, proposal, ledger) {
      var amount = proposal.data.amount;
      var currency = proposal.data.currency;
      var payerSlug = WL.U.slug(proposal.data.payer || '');
      var keys = Object.keys(root.quests.questsById);
      var matched = '';
      for (var i = 0; i < keys.length; i++) {
        var q = root.quests.questsById[keys[i]];
        for (var j = 0; j < q.rewardsPromised.length; j++) {
          var r = q.rewardsPromised[j];
          if (!r.paid && r.amount === amount && r.currency === currency && (!payerSlug || WL.U.slug(r.payer || q.giver) === payerSlug)) {
            r.paid = true;
            r.paidKey = ledger.key;
            q.rewardsPaid.push(r.id);
            matched = q.id;
            root.quests.rewardPayments.push({ questId: q.id, rewardId: r.id, amount: amount, currency: currency, payer: proposal.data.payer || '', key: ledger.key });
            break;
          }
        }
        if (matched) break;
      }
      return { status: matched ? 'APPLIED' : 'NO_MATCHING_PROMISE_RECORDED', applied: true, family: 'quest', type: 'reward_payment_record', questId: matched, key: ledger.key };
    },
    summary: function (root) {
      var lines = [];
      var keys = Object.keys(root.quests.questsById);
      for (var i = 0; i < keys.length; i++) {
        var q = root.quests.questsById[keys[i]];
        lines.push(q.title + ' [' + q.status + '] — ' + q.objective);
      }
      return lines.length ? lines : ['(none)'];
    }
  };

  WL.TimeTravelSystem = {
    apply: function (root, validation, ledger) {
      var p = validation.proposal;
      if (p.type === 'elapsed_time_candidate') return WL.TimeTravelSystem.advance(root, p, ledger);
      if (p.type === 'route_base_candidate') return WL.TimeTravelSystem.recordRouteBase(root, p, ledger);
      if (p.type === 'route_delay_candidate') return WL.TimeTravelSystem.recordRouteDelay(root, p, ledger);
      if (p.type === 'deadline_candidate') return WL.TimeTravelSystem.recordDeadline(root, p, ledger);
      return { status: 'IGNORED', applied: false, family: 'time', type: p.type, reason: 'unsupported_time_event' };
    },
    advanceClock: function (root, minutes) {
      minutes = WL.U.clamp(minutes, 0, 5256000);
      root.time.elapsedMinutes += minutes;
      var total = ((root.time.day - 1) * 1440) + (root.time.hour * 60) + root.time.minute + minutes;
      root.time.day = Math.floor(total / 1440) + 1;
      var rem = total % 1440;
      root.time.hour = Math.floor(rem / 60);
      root.time.minute = rem % 60;
    },
    advance: function (root, proposal, ledger) {
      var minutes = WL.U.clamp(proposal.data.minutes, 1, 5256000);
      WL.TimeTravelSystem.advanceClock(root, minutes);
      root.time.recent.push({ key: ledger.key, action: 'elapsed', minutes: minutes, phrase: proposal.textPreview });
      WL.U.capArray(root.time.recent, 40);
      return { status: 'APPLIED', applied: true, family: 'time', type: 'elapsed', minutes: minutes, key: ledger.key };
    },
    recordRouteBase: function (root, proposal, ledger) {
      var data = proposal.data;
      var id = data.targetId || data.routeId || WL.EventScanner.routeId(data.from, data.to);
      var route = root.time.routesById[id];
      if (!route) {
        route = { id: id, from: data.from, to: data.to, baseMinutes: data.minutes, delays: [], history: [], aliases: [data.from + ' -> ' + data.to, data.to + ' -> ' + data.from] };
        root.time.routesById[id] = route;
        root.time.routeIds.push(id);
      } else if (!route.baseMinutes) {
        route.baseMinutes = data.minutes;
      }
      root.time.currentRouteId = id;
      route.history.push({ key: ledger.key, action: 'base', minutes: data.minutes, phrase: proposal.textPreview });
      WL.U.capArray(route.history, 20);
      return { status: 'APPLIED', applied: true, family: 'time', type: 'route_base', routeId: id, baseMinutes: route.baseMinutes, key: ledger.key };
    },
    recordRouteDelay: function (root, proposal, ledger) {
      var id = root.time.currentRouteId || 'route_unknown_current_trip';
      var route = root.time.routesById[id];
      if (!route) {
        route = { id: id, from: '', to: '', baseMinutes: 0, delays: [], history: [], aliases: [] };
        root.time.routesById[id] = route;
        root.time.routeIds.push(id);
      }
      route.delays.push({ key: ledger.key, reason: proposal.data.reason || 'Unknown', actualMinutes: proposal.data.actualMinutes || 0, phrase: proposal.textPreview });
      route.history.push({ key: ledger.key, action: 'delay', reason: proposal.data.reason || 'Unknown', actualMinutes: proposal.data.actualMinutes || 0 });
      WL.U.capArray(route.delays, 20);
      WL.U.capArray(route.history, 30);
      return { status: 'APPLIED', applied: true, family: 'time', type: 'route_delay', routeId: id, actualMinutes: proposal.data.actualMinutes || 0, key: ledger.key };
    },
    recordDeadline: function (root, proposal, ledger) {
      var label = proposal.data.label || 'Unknown Deadline';
      var id = 'deadline_' + WL.U.hash(label + '|' + proposal.data.task);
      if (!root.time.deadlines.some(function (d) { return d.id === id; })) {
        root.time.deadlines.push({ id: id, label: label, task: proposal.data.task || '', status: 'active', key: ledger.key });
      }
      WL.U.capArray(root.time.deadlines, 40);
      return { status: 'APPLIED', applied: true, family: 'time', type: 'deadline', deadlineId: id, key: ledger.key };
    },
    summary: function (root) {
      var lines = [];
      lines.push('Day ' + root.time.day + ', ' + String(root.time.hour).padStart(2, '0') + ':' + String(root.time.minute).padStart(2, '0') + ' / elapsed ' + root.time.elapsedMinutes + ' min');
      lines.push('Routes: ' + (root.time.routeIds.length ? root.time.routeIds.map(function (id) { var r = root.time.routesById[id]; return (r.from || '?') + ' <-> ' + (r.to || '?') + ' base ' + (r.baseMinutes || 0) + 'm delays ' + (r.delays || []).length; }).join('; ') : '(none)'));
      lines.push('Deadlines: ' + (root.time.deadlines.length ? root.time.deadlines.map(function (d) { return d.label + ' — ' + d.task; }).join('; ') : '(none)'));
      return lines;
    }

  WL.WorldStructureSystem = {
    apply: function (root, validation, ledger) {
      var p = validation.proposal;
      if (p.type === 'place_definition_candidate') return WL.WorldStructureSystem.definePlace(root, p, ledger);
      if (p.type === 'current_place_candidate') return WL.WorldStructureSystem.setCurrent(root, p, ledger);
      if (p.type === 'layout_fact_candidate') return WL.WorldStructureSystem.recordLayout(root, p, ledger);
      if (p.type === 'owner_relation_candidate') return WL.WorldStructureSystem.recordRelation(root, p, ledger, 'owner');
      if (p.type === 'worker_relation_candidate') return WL.WorldStructureSystem.recordRelation(root, p, ledger, 'worker');
      return { status: 'IGNORED', applied: false, family: 'place', type: p.type, reason: 'unsupported_place_event' };
    },
    ensurePlace: function (root, name, type) {
      name = WL.U.title(name || 'Unknown Place');
      var id = 'place_' + WL.U.slug(name);
      var p = root.places.placesById[id];
      if (!p) {
        p = { id: id, name: name, type: type || 'place', locationHint: '', owners: [], workers: [], layout: {}, layoutHistory: [], contradictions: [], relations: [], history: [] };
        root.places.placesById[id] = p;
        root.places.recent.push(id);
      }
      if (type && (!p.type || p.type === 'place')) p.type = type;
      WL.U.capArray(root.places.recent, 40);
      return p;
    },
    definePlace: function (root, proposal, ledger) {
      var p = WL.WorldStructureSystem.ensurePlace(root, proposal.data.placeName, proposal.data.placeType || 'place');
      if (proposal.data.locationHint && !p.locationHint) p.locationHint = proposal.data.locationHint;
      p.history.push({ key: ledger.key, action: 'define', phrase: proposal.textPreview });
      WL.U.capArray(p.history, 30);
      if (!root.places.currentId) root.places.currentId = p.id;
      return { status: 'APPLIED', applied: true, family: 'place', type: 'define', placeId: p.id, key: ledger.key };
    },
    setCurrent: function (root, proposal, ledger) {
      var p = WL.WorldStructureSystem.ensurePlace(root, proposal.data.placeName, proposal.data.placeType || 'place');
      root.places.currentId = p.id;
      p.history.push({ key: ledger.key, action: 'current', phrase: proposal.textPreview });
      WL.U.capArray(p.history, 30);
      return { status: 'APPLIED', applied: true, family: 'place', type: 'current', placeId: p.id, key: ledger.key };
    },
    recordLayout: function (root, proposal, ledger) {
      var placeId = root.places.currentId;
      if (!placeId) return { status: 'PENDING_NO_CURRENT_PLACE', applied: false, family: 'place', type: 'layout', key: ledger.key };
      var p = root.places.placesById[placeId];
      if (!p) return { status: 'PENDING_NO_CURRENT_PLACE_RECORD', applied: false, family: 'place', type: 'layout', key: ledger.key };
      var fixture = proposal.data.fixture;
      var value = proposal.data.value;
      var old = p.layout[fixture];
      if (old && old !== value) {
        var rec = { key: ledger.key, placeId: placeId, fixture: fixture, oldValue: old, newValue: value, phrase: proposal.textPreview };
        p.contradictions.push(rec);
        root.places.contradictions.push(rec);
        WL.U.capArray(p.contradictions, 20);
        WL.U.capArray(root.places.contradictions, 40);
        return { status: 'CONTRADICTION_FLAGGED', applied: true, family: 'place', type: 'layout_contradiction', placeId: placeId, key: ledger.key };
      }
      p.layout[fixture] = value;
      p.layoutHistory.push({ key: ledger.key, fixture: fixture, value: value, phrase: proposal.textPreview });
      WL.U.capArray(p.layoutHistory, 30);
      return { status: 'APPLIED', applied: true, family: 'place', type: 'layout', placeId: placeId, key: ledger.key };
    },
    recordRelation: function (root, proposal, ledger, kind) {
      var p = WL.WorldStructureSystem.ensurePlace(root, proposal.data.placeName, 'place');
      var actorName = proposal.data.actorName || '';
      var list = kind === 'owner' ? p.owners : p.workers;
      if (actorName && list.indexOf(actorName) < 0) list.push(actorName);
      var rec = { key: ledger.key, kind: kind, actorName: actorName, placeId: p.id, placeName: p.name };
      p.relations.push(rec);
      root.places.relations.push(rec);
      WL.U.capArray(p.relations, 30);
      WL.U.capArray(root.places.relations, 60);
      return { status: 'APPLIED', applied: true, family: 'place', type: kind + '_relation', placeId: p.id, key: ledger.key };
    },
    summary: function (root) {
      var ids = Object.keys(root.places.placesById);
      var lines = [];
      if (!ids.length) lines.push('(none)');
      for (var i = 0; i < ids.length; i++) {
        var p = root.places.placesById[ids[i]];
        lines.push(p.name + ' [' + p.type + ']' + (root.places.currentId === p.id ? ' <current>' : '') + ' owners: ' + (p.owners.join(', ') || '-') + ' workers: ' + (p.workers.join(', ') || '-'));
      }
      if (root.places.contradictions.length) lines.push('Contradictions: ' + root.places.contradictions.length);
      return lines;
    }
  };

      return { status: 'APPLIED', applied: true, family: 'actor', type: 'social', actorId: a.id, key: ledger.key };
    },

    mark: function (root, reducerResults) {
      var anyApplied = false;
      for (var i = 0; i < reducerResults.length; i++) if (reducerResults[i].applied) anyApplied = true;
      if (anyApplied) {
        root.dirty.targets.push('diagnostics');
        for (var d = 0; d < reducerResults.length; d++) {
          if (reducerResults[d].family === 'inventory' && root.dirty.targets.indexOf('inventory') < 0) root.dirty.targets.push('inventory');
          if (reducerResults[d].family === 'quest' && root.dirty.targets.indexOf('quests') < 0) root.dirty.targets.push('quests');
          if (reducerResults[d].family === 'form' && root.dirty.targets.indexOf('body') < 0) root.dirty.targets.push('body');
          if (reducerResults[d].family === 'form' && root.dirty.targets.indexOf('profile') < 0) root.dirty.targets.push('profile');
          if (reducerResults[d].family === 'ability' && root.dirty.targets.indexOf('abilities') < 0) root.dirty.targets.push('abilities');
          if (reducerResults[d].family === 'progression' && root.dirty.targets.indexOf('progression') < 0) root.dirty.targets.push('progression');
          if (reducerResults[d].family === 'time' && root.dirty.targets.indexOf('time') < 0) root.dirty.targets.push('time');
          if (reducerResults[d].family === 'place' && root.dirty.targets.indexOf('places') < 0) root.dirty.targets.push('places');
          if (reducerResults[d].family === 'actor' && root.dirty.targets.indexOf('actors') < 0) root.dirty.targets.push('actors');
        }
        root.dirty.reasons.push('pipeline_or_owner_update');
        WL.U.capArray(root.dirty.targets, 40);
        WL.U.capArray(root.dirty.reasons, 40);
      }
    }
  };
