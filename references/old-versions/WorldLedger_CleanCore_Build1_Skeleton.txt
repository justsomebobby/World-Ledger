  // ============================================================
  // 03. Utility helpers
  // ============================================================
  WL.U = {
    norm: function (value) {
      return String(value == null ? '' : value)
        .replace(/[\u2018\u2019]/g, "'")
        .replace(/[\u201C\u201D]/g, '"')
        .replace(/\s+/g, ' ')
        .trim();
    },
    lower: function (value) { return WL.U.norm(value).toLowerCase(); },
    title: function (value) {
      var raw = WL.U.norm(value).replace(/^the\s+/i, '');
      if (!raw) return '';
      var small = { a: 1, an: 1, and: 1, at: 1, by: 1, for: 1, from: 1, in: 1, of: 1, on: 1, or: 1, the: 1, to: 1, with: 1 };
      var parts = raw.split(/\s+/);
      for (var i = 0; i < parts.length; i++) {
        var w = parts[i].toLowerCase();
        parts[i] = (i > 0 && small[w]) ? w : w.charAt(0).toUpperCase() + w.slice(1);
      }
      return parts.join(' ');
    },
    slug: function (value) {
      var s = WL.U.lower(value).replace(/[^a-z0-9]+/g, '_').replace(/^_+|_+$/g, '');
      return s.slice(0, 80) || 'unknown';
    },
    clamp: function (n, min, max) {
      n = Number(n);
      if (!isFinite(n)) n = min;
      if (n < min) return min;
      if (n > max) return max;
      return n;
    },
    hash: function (value) {
      var s = String(value == null ? '' : value);
      var h = 2166136261;
      for (var i = 0; i < s.length; i++) {
        h ^= s.charCodeAt(i);
        h += (h << 1) + (h << 4) + (h << 7) + (h << 8) + (h << 24);
      }
      return ('00000000' + ((h >>> 0).toString(16))).slice(-8);
    },
    clone: function (value) {
      try { return JSON.parse(JSON.stringify(value)); } catch (e) { return value; }
    },
    capArray: function (arr, max) {
      if (!Array.isArray(arr)) return [];
      while (arr.length > max) arr.shift();
      return arr;
    },
    ensureObject: function (parent, key, factory) {
      if (!parent || typeof parent !== 'object') return factory ? factory() : {};
      if (!parent[key] || typeof parent[key] !== 'object' || Array.isArray(parent[key])) parent[key] = factory ? factory() : {};
      return parent[key];
    },
    ensureArray: function (parent, key) {
      if (!parent || typeof parent !== 'object') return [];
      if (!Array.isArray(parent[key])) parent[key] = [];
      return parent[key];
    },
    startsCommand: function (text) { return /^\s*\/\w+/.test(String(text || '')); },
    commandName: function (text) {
      var m = String(text || '').match(/^\s*\/([a-zA-Z0-9_-]+)/);
      return m ? m[1].toLowerCase() : '';
    },
    commandArgs: function (text) { return String(text || '').replace(/^\s*\/\S+\s*/, '').trim(); },
    safeString: function (value) {
      if (value == null) return '';
      if (typeof value === 'string') return WL.U.norm(value);
      if (typeof value === 'number' || typeof value === 'boolean') return String(value);
      if (Array.isArray(value)) return value.map(function (v) { return WL.U.safeString(v); }).filter(Boolean).join(', ');
      if (typeof value === 'object') {
        if (value.displayName) return WL.U.safeString(value.displayName);
        if (value.name) return WL.U.safeString(value.name);
        if (value.title) return WL.U.safeString(value.title);
        return JSON.stringify(value);
      }
      return WL.U.norm(value);
    },
    getGlobalState: function () {
      if (typeof globalThis === 'undefined') return {};
      if (!globalThis.state || typeof globalThis.state !== 'object') globalThis.state = {};
      return globalThis.state;
    },
    getActionCount: function () {
      try {
        if (typeof info === 'object' && info && typeof info.actionCount === 'number') return info.actionCount;
      } catch (e) {}
      return 0;
    }
  };

      root.schemaId = WL.SCHEMA_ID;
      return root;

  // ============================================================
  // 05. StateManager and quarantine
  // ============================================================
  WL.StateManager = {
    root: function () { return WL.StateManager.init(WL.U.getGlobalState()); },
    init: function (stateObj) {
      var st = stateObj && typeof stateObj === 'object' ? stateObj : WL.U.getGlobalState();
      if (!st[WL.ROOT_KEY] || typeof st[WL.ROOT_KEY] !== 'object' || Array.isArray(st[WL.ROOT_KEY])) {
        st[WL.ROOT_KEY] = WL.Schema.root();
      }
      var root = WL.Migration.migrate(st[WL.ROOT_KEY]);
      WL.StateManager.ensureBuckets(root);
      root.build.projectName = WL.PROJECT_NAME;
      root.build.buildName = WL.BUILD_NAME;
      root.build.buildVersion = WL.BUILD_VERSION;
      root.build.buildStage = WL.BUILD_STAGE;
      root.build.status = WL.STATUS;
      root.meta.lastUpdatedTurn = WL.U.getActionCount();
      st[WL.ROOT_KEY] = root;
      return root;
    },
    ensureBuckets: function (root) {
      var fresh = WL.Schema.root();
      var keys = Object.keys(fresh);
      for (var i = 0; i < keys.length; i++) {
        var k = keys[i];
        if (k === 'schemaVersion' || k === 'schemaId') continue;
        if (!root[k] || typeof root[k] !== 'object' || Array.isArray(root[k]) !== Array.isArray(fresh[k])) {
          root[k] = WL.U.clone(fresh[k]);
        }
      }
      WL.StateManager.enforceInvariants(root);
    },
    enforceInvariants: function (root) {
      root.body.maxHp = WL.U.clamp(root.body.maxHp, 1, 999999);
      root.body.hp = WL.U.clamp(root.body.hp, 0, root.body.maxHp);
      root.body.maxMp = WL.U.clamp(root.body.maxMp, 0, 999999);
      root.body.mp = WL.U.clamp(root.body.mp, 0, root.body.maxMp);
      root.body.maxEp = WL.U.clamp(root.body.maxEp, 0, 999999);
      root.body.ep = WL.U.clamp(root.body.ep, 0, root.body.maxEp);
      WL.U.capArray(root.runtime.hookTrace, WL.Constants.maxHookTrace);
      WL.U.capArray(root.operationLedger.recentKeys, WL.Constants.maxRecentKeys);
      WL.U.capArray(root.diagnostics.audit, WL.Constants.maxAuditRows);
    }
  };

  // ============================================================
  // 06. Runtime hooks and source classification
  // ============================================================
  WL.SourceGuard = {
    classify: function (text, hook) {
      var raw = String(text == null ? '' : text);
      if (WL.U.startsCommand(raw)) return { kind: 'command', hook: hook, commandName: WL.U.commandName(raw), eligibleForCanon: false };
      return { kind: hook || 'unknown', hook: hook || 'unknown', commandName: '', eligibleForCanon: hook === 'output' };
    }
  };

  WL.Runtime = {
    begin: function (root, hook, text) {
      root.runtime.currentHook = hook;
      root.runtime.currentTurn = WL.U.getActionCount();
      root.runtime.hookDepth += 1;
      root.runtime.hookTrace.push({ turn: root.runtime.currentTurn, hook: hook, hash: WL.U.hash(text), phase: 'begin' });
      WL.U.capArray(root.runtime.hookTrace, WL.Constants.maxHookTrace);
    },
    end: function (root, hook, status) {
      root.runtime.lastReturnStatus = status || 'ok';
      root.runtime.hookTrace.push({ turn: root.runtime.currentTurn, hook: hook, phase: 'end', status: status || 'ok' });
      WL.U.capArray(root.runtime.hookTrace, WL.Constants.maxHookTrace);
      root.runtime.hookDepth = Math.max(0, root.runtime.hookDepth - 1);
      root.runtime.currentHook = 'none';
      WL.StateManager.enforceInvariants(root);
    },
    handleInput: function (text) {
      var root = WL.StateManager.root();
      WL.Runtime.begin(root, 'input', text);
      var source = WL.SourceGuard.classify(text, 'input');
      root.runtime.lastInputHash = WL.U.hash(text);
      var out = text;
      if (source.kind === 'command') {
        root.runtime.lastCommandName = source.commandName;
        root.runtime.lastCommandHash = WL.U.hash(text);
        root.runtime.lastCommandTurn = root.runtime.currentTurn;
        out = WL.CommandRouter.route(text, root);
      } else {
        root.pending.lastInputIntent = { hash: root.runtime.lastInputHash, textPreview: WL.U.norm(text).slice(0, 160), note: 'input intent recorded only; no gameplay mutation in Build 1' };
      }
      WL.Runtime.end(root, 'input', 'ok');
      return out;
    },
    handleContext: function (text) {
      var root = WL.StateManager.root();
      WL.Runtime.begin(root, 'context', text);
      root.runtime.lastContextHash = WL.U.hash(text);
      var packet = WL.ContextSystem.build(root, text);
      WL.Runtime.end(root, 'context', 'ok');
      return packet;
    },
    handleOutput: function (text) {
      var root = WL.StateManager.root();
      WL.Runtime.begin(root, 'output', text);
      root.runtime.lastOutputHash = WL.U.hash(text);
      root.pending.lastOutputObservation = { hash: root.runtime.lastOutputHash, textPreview: WL.U.norm(text).slice(0, 160), note: 'output observed only; parser/reducer spine starts in next build' };
      WL.Runtime.end(root, 'output', 'ok');
      return text;
    }
  };

  // ============================================================
  // 07. OperationLedger and Diagnostics

    makeKey: function (parts) { return WL.U.slug((parts || []).join('|')); },
    has: function (root, key) { return !!(root.operationLedger.keysById[WL.OperationLedger.makeKey([key])]); },
    add: function (root, key, reason) {
      var k = WL.OperationLedger.makeKey([key]);
      if (root.operationLedger.keysById[k]) {
        root.operationLedger.duplicateSkips.push({ key: k, reason: reason || 'duplicate', turn: root.runtime.currentTurn });
        WL.U.capArray(root.operationLedger.duplicateSkips, 80);
        return { applied: false, duplicate: true, key: k };
      }
      root.operationLedger.keysById[k] = { key: k, reason: reason || '', turn: root.runtime.currentTurn };
      root.operationLedger.recentKeys.push(k);
      WL.U.capArray(root.operationLedger.recentKeys, WL.Constants.maxRecentKeys);
      return { applied: true, duplicate: false, key: k };
    }
  };

  WL.Diagnostics = {
    row: function (root, subsystem, level, message, extra) {

      root.diagnostics.audit.push(rec);
      WL.U.capArray(root.diagnostics.audit, WL.Constants.maxAuditRows);
      return rec;
    }
  };

      rows.push(WL.Assert.pass('SRC-002', typeof WL['onContext'] === 'function', 'WL.onContext active'));
      rows.push(WL.Assert.pass('SRC-003', typeof WL['onOutput'] === 'function', 'WL.onOutput active'));
      rows.push(WL.Assert.pass('SRC-004', Object.is(WL.onInput, WL.Runtime.handleInput), 'WL.onInput points directly to Runtime.handleInput'));
      rows.push(WL.Assert.pass('SRC-005', Object.is(WL.onContext, WL.Runtime.handleContext), 'WL.onContext points directly to Runtime.handleContext'));
      rows.push(WL.Assert.pass('SRC-006', Object.is(WL.onOutput, WL.Runtime.handleOutput), 'WL.onOutput points directly to Runtime.handleOutput'));
      rows.push(WL.Assert.pass('SRC-007', !WL.failedPromotionPhrasingLayer, 'no failed-promotion phrasing layer object'));
      rows.push(WL.Assert.pass('SRC-008', !WL.lateRepairAuthority, 'no late repair authority object'));
      rows.push(WL.Assert.pass('SRC-009', !WL['CardTemplate' + 'V115'], 'no stale old card-template authority in clean skeleton'));
      rows.push(WL.Assert.pass('SRC-010', !WL['V112' + 'AcceptanceMap'], 'no old acceptance patch chain'));
      return rows;
    },
    summarize: function (rows) {
      var out = { PASS: 0, WARN: 0, BLOCKED: 0, FAIL: 0 };
      for (var i = 0; i < rows.length; i++) out[rows[i].status] = (out[rows[i].status] || 0) + 1;
      return out;
    },
    render: function (commandName, rows, note) {
      var s = WL.SourceAudit.summarize(rows);
      var status = s.FAIL ? 'FAIL' : (s.BLOCKED ? 'BLOCKED' : (s.WARN ? 'WARN' : 'PASS'));
      var lines = [WL.CommandRenderer.header(commandName, status).replace(/\n$/, '')];
      lines.push('Build: ' + WL.BUILD_VERSION);
      lines.push('Summary: PASS ' + s.PASS + ' / WARN ' + s.WARN + ' / BLOCKED ' + s.BLOCKED + ' / FAIL ' + s.FAIL);
      if (note) lines.push('Note: ' + note);
      lines.push('');
      for (var i = 0; i < rows.length; i++) lines.push(rows[i].status + ': ' + rows[i].id + ' — ' + rows[i].label);
      return lines.join('\n') + '\n';
    }
  };

  // ============================================================
  // 11. Export / AI Dungeon hook binding
  // ============================================================
  WL.onInput = WL.Runtime.handleInput;
  WL.onContext = WL.Runtime.handleContext;
  WL.onOutput = WL.Runtime.handleOutput;

  if (typeof globalThis !== 'undefined') {
    globalThis.WorldLedger = WL;
  }
})();
