WorldLedgerV0.10 — Core MVP Foundation Script / Code Pass 1 ProposalBuilder + Control Map Correction
Made by: JustSomeBobby

WorldLedger is a freeform RPG truth engine for AI Dungeon. It is built to preserve
canon, consequences, state, and memory without turning AI Dungeon into a rigid command game.

Credit and direct inspiration:
- RPGUI by @RonenNess — state-first RPG/HUD presentation discipline and clean helper separation, reinterpreted for AI Dungeon as text-first state and command readouts. No HTML UI code is used.

- Inner Self by @LewdLeah — segmented long-term NPC memory and major-character persistence, rebuilt as selective reducer-owned memory rather than universal NPC simulation.

Not direct foundations for this build: LoLa, Hashtag-DnD, AID-dice-rolling, and Gnurro AIDscripts. They may remain general research references, but they are not credited here as active functional inspirations unless a future build directly borrows from them.

Design doctrine:
Input may intend. Context may remind. Output may propose. ParserHits are observations.
EventProposals are structured claims. CanonValidator decides eligibility. Reducers commit.
OperationLedger dedupes. DirtyQueue marks refresh. Context reads. Cards mirror. ReleaseCheck tells the truth.

Code Pass 1 scope:
This is the foundation build with the Build Chunk 10 ProposalBuilder lock corrected against the active Code Pass 1 control map. It keeps old scripts and previous WorldLedger/AIDRPG versions as evidence and salvage sources, not as paste-in architecture. It initializes, inspects, audits, validates, and commits only limited MVP-safe events while enforcing safe state creation, structural repair, preservation of valid truth, quarantine of unsafe legacy data, and no hook-wrapper leakage inside the Library bundle.
Deferred systems remain disabled: generated Story Cards, full economy, crafting, investigation, reputation/faction
automation, domain/labor, absorption/evolution, advanced combat, warfare, multi-protagonist, and manual mutation commands.
*/

    lower(value) { return this.norm(value).toLowerCase(); },
    title(value) {
      const raw = this.norm(value).replace(/^the\s+/i, '');
      if (!raw) return '';
      const small = { a: 1, an: 1, and: 1, at: 1, by: 1, for: 1, from: 1, in: 1, of: 1, on: 1, or: 1, the: 1, to: 1, with: 1 };
      return raw.split(/\s+/).map((w, i) => {
        const clean = w.toLowerCase();
        if (i > 0 && small[clean]) return clean;
        return clean.charAt(0).toUpperCase() + clean.slice(1);
      }).join(' ');
    },
    slug(value) {
      return this.lower(value).replace(/[^a-z0-9]+/g, '_').replace(/^_+|_+$/g, '').slice(0, 80) || 'unknown';
    },
    clamp(n, min, max) {
      n = Number(n);
      if (!Number.isFinite(n)) n = min;
      return Math.max(min, Math.min(max, n));
    },
    hash(value) {
      const s = String(value == null ? '' : value);
      let h = 2166136261;
      for (let i = 0; i < s.length; i++) {
        h ^= s.charCodeAt(i);
        h += (h << 1) + (h << 4) + (h << 7) + (h << 8) + (h << 24);
      }
      return ('00000000' + (h >>> 0).toString(16)).slice(-8);
    },
    capArray(arr, max) {
      if (!Array.isArray(arr)) return [];
      while (arr.length > max) arr.shift();
      return arr;
    },
    uniquePush(arr, value, max) {
      if (!Array.isArray(arr)) return false;
      const v = this.norm(value);
      if (!v) return false;
      const k = this.slug(v);
      for (let i = 0; i < arr.length; i++) if (this.slug(arr[i]) === k) return false;
      arr.push(v);
      this.capArray(arr, max || 20);
      return true;
    },
    ensureObj(parent, key, factory) {
      if (!parent || typeof parent !== 'object') return factory ? factory() : {};
      if (!parent[key] || typeof parent[key] !== 'object' || Array.isArray(parent[key])) parent[key] = factory ? factory() : {};
      return parent[key];
    },
    ensureArray(parent, key) {
      if (!parent || typeof parent !== 'object') return [];
      if (!Array.isArray(parent[key])) parent[key] = [];
      return parent[key];
    },
    safeJson(value) {
      try { return JSON.stringify(value); } catch (e) { return '[unserializable]'; }
    },
    unquoted(text) {
      return String(text || '').replace(/["“][\s\S]*?["”]/g, ' ').replace(/'[^']{1,240}'/g, ' ');
    },
    sentenceSplit(text) {
      const t = this.norm(text);
      if (!t) return [];
      return t.split(/(?<=[.!?])\s+|\n+/).map(s => s.trim()).filter(Boolean).slice(0, 16);
    },
    startsCommand(text) {
      return /^\s*\/\w+/.test(String(text || ''));
    },
    commandName(text) {
      const m = String(text || '').match(/^\s*\/([a-zA-Z0-9_-]+)/);
      return m ? m[1].toLowerCase() : '';
    },
    commandArgs(text) {
      return String(text || '').replace(/^\s*\/\S+\s*/, '').trim();
    },

      return lines;
    }

  WL.AuditLogger = {
    caps: Object.freeze({
      debug: 50, warnings: 50, parser: 40, proposals: 50, validations: 50, reducers: 40,
      cards: 50, context: 50, cleanup: 50, migration: 50, releasecheck: 50,
      lastTurnSummary: 24, recentProposals: 50, recentValidationResults: 50,
      debugMessages: 50, runtimeWarnings: 25, runtimeErrors: 20, duplicateAudit: 50,
      cardAudit: 50, contextAudit: 50, cleanupAudit: 50, migrationAudit: 50,
      repairAudit: 50, auditLog: 80
    }),
    safeLine(value, maxLength) {
      const max = WL.U.clamp(maxLength || 240, 24, 1000);
      let s = WL.U.norm(value);
      if (s.length > max) s = s.slice(0, max - 3) + '...';
      return s;
    },
    ensure(root) {
      if (!root || typeof root !== 'object') return null;
      if (!root.audits || typeof root.audits !== 'object' || Array.isArray(root.audits)) root.audits = {};
      const buckets = Object.keys(this.caps);
      for (let i = 0; i < buckets.length; i++) if (!Array.isArray(root.audits[buckets[i]])) root.audits[buckets[i]] = [];
      if (root.runtime) {
        if (!Array.isArray(root.runtime.runtimeWarnings)) root.runtime.runtimeWarnings = [];
        if (!Array.isArray(root.runtime.runtimeErrors)) root.runtime.runtimeErrors = [];
        if (!Array.isArray(root.runtime.hookTrace)) root.runtime.hookTrace = [];
      }
      root.audits.auditLoggerLockVersion = WL.AUDIT_LEDGER_LOCK_VERSION;
      return root.audits;
    },
    capBucket(root, bucketPath, max) {
      if (!root) return;
      const parts = String(bucketPath || '').split('.').filter(Boolean);
      let obj = root;
      for (let i = 0; i < parts.length - 1; i++) {
        obj = obj && obj[parts[i]];
        if (!obj || typeof obj !== 'object') return;
      }
      const key = parts[parts.length - 1];
      if (!obj || !key) return;
      if (!Array.isArray(obj[key])) obj[key] = [];
      WL.U.capArray(obj[key], max || this.caps[key] || 40);
    },
    capAll(root) {
      const audits = this.ensure(root);
      if (!audits) return;
      const caps = this.caps;
      Object.keys(caps).forEach(k => this.capBucket(root, 'audits.' + k, caps[k]));
      if (root.runtime) {
        this.capBucket(root, 'runtime.runtimeWarnings', WL.SCHEMA_CAPS.runtimeWarnings || 25);
        this.capBucket(root, 'runtime.runtimeErrors', WL.SCHEMA_CAPS.runtimeErrors || 20);
        this.capBucket(root, 'runtime.hookTrace', WL.SCHEMA_CAPS.hookTrace || 30);
        this.capBucket(root, 'runtime.suppressionAudit', 30);
        this.capBucket(root, 'runtime.returnAudit', 30);
      }
    },
    row(root, family, outcome, reason, extra) {
      return Object.assign({
        turn: WL.U.nowTurn(root),
        family: family || 'system',
        outcome: outcome || 'info',
        reason: this.safeLine(reason || '', 220),
        sourceEventKey: '',
        preview: '',
        notes: []
      }, extra || {});
    },
    append(root, bucket, entry, max) {
      const audits = this.ensure(root);
      if (!audits) return null;
      if (!Array.isArray(audits[bucket])) audits[bucket] = [];
      const row = Object.assign({ turn: WL.U.nowTurn(root), at: bucket }, entry || {});
      if (row.message) row.message = this.safeLine(row.message, 360);
      if (row.reason) row.reason = this.safeLine(row.reason, 360);
      if (row.preview) row.preview = this.safeLine(row.preview, 360);
      audits[bucket].push(row);
      WL.U.capArray(audits[bucket], max || this.caps[bucket] || WL.SCHEMA_CAPS.diagnostics || 40);
      return row;
    },
    debug(root, subsystem, message, data) {
      if (arguments.length === 3) { data = null; message = subsystem; subsystem = 'debug'; }
      const row = this.append(root, 'debugMessages', { subsystem: subsystem || 'debug', message: this.safeLine(message, 300), data: data || null }, this.caps.debugMessages);
      this.append(root, 'debug', { subsystem: subsystem || 'debug', message: this.safeLine(message, 300), data: data || null }, this.caps.debug);
      return row;
    },
    warn(root, subsystem, message, data) {
      if (arguments.length === 3) { data = null; message = subsystem; subsystem = 'warning'; }
      return this.append(root, 'warnings', { subsystem: subsystem || 'warning', message: this.safeLine(message, 300), data: data || null }, this.caps.warnings);
    },
    runtimeWarning(root, message, data) {
      const msg = this.safeLine(message, 300);
      const row = this.append(root, 'runtimeWarnings', { message: msg, data: data || null }, this.caps.runtimeWarnings);
      if (root && root.runtime) {
        if (!Array.isArray(root.runtime.runtimeWarnings)) root.runtime.runtimeWarnings = [];
        root.runtime.runtimeWarnings.push({ turn: WL.U.nowTurn(root), message: msg, data: data || null });
        WL.U.capArray(root.runtime.runtimeWarnings, WL.SCHEMA_CAPS.runtimeWarnings || 25);
        if (!Array.isArray(root.runtime.warnings)) root.runtime.warnings = [];
        root.runtime.warnings.push({ turn: WL.U.nowTurn(root), message: msg });
        WL.U.capArray(root.runtime.warnings, WL.SCHEMA_CAPS.runtimeWarnings || 25);
      }
      return row;
    },
    runtimeWarn(root, message, data) { return this.runtimeWarning(root, message, data); },
    runtimeError(root, a, b, c, d, e) {
      let rec;
      if (typeof a === 'object' && a !== null) rec = a;
      else rec = { hook: root && root.runtime ? root.runtime.currentHook : 'unknown', phase: 'runtime', message: a, data: b || null };
      rec = Object.assign({
        turn: WL.U.nowTurn(root), hook: rec.hook || 'unknown', phase: rec.phase || 'runtime',
        message: this.safeLine(rec.message || rec.error || 'runtime error', 300),
        safeTextReturned: !!rec.safeTextReturned, partialCommitBlocked: rec.partialCommitBlocked !== false
      }, rec);
      const row = this.append(root, 'runtimeErrors', rec, this.caps.runtimeErrors);
      if (root && root.runtime) {
        if (!Array.isArray(root.runtime.runtimeErrors)) root.runtime.runtimeErrors = [];
        root.runtime.runtimeErrors.push(rec);
        WL.U.capArray(root.runtime.runtimeErrors, WL.SCHEMA_CAPS.runtimeErrors || 20);
        if (!Array.isArray(root.runtime.errors)) root.runtime.errors = [];
        root.runtime.errors.push({ turn: WL.U.nowTurn(root), message: rec.message });
        WL.U.capArray(root.runtime.errors, WL.SCHEMA_CAPS.runtimeErrors || 20);
      }
      return row;
    },
    proposal(root, record) {
      const rec = this.row(root, (record && record.family) || 'proposal', (record && record.outcome) || 'created', (record && record.reason) || 'proposal recorded', record || {});
      this.append(root, 'recentProposals', rec, this.caps.recentProposals);
      return this.append(root, 'proposals', rec, this.caps.proposals);
    },
    validation(root, record) {
      const rec = this.row(root, (record && record.family) || 'validation', (record && (record.result || record.outcome)) || 'checked', (record && record.reason) || 'validation recorded', record || {});
      this.append(root, 'recentValidationResults', rec, this.caps.recentValidationResults);
      return this.append(root, 'validations', rec, this.caps.validations);
    },
    duplicate(root, key, reason) {
      const rec = this.row(root, 'ledger', 'duplicate_skip', reason || 'duplicate key skipped', { sourceEventKey: WL.OperationLedger ? WL.OperationLedger.normalizeKey(key) : WL.U.slug(key), preview: String(key || '').slice(0, 120) });
      return this.append(root, 'duplicateAudit', rec, this.caps.duplicateAudit);
    },
    card(root, message) { return this.append(root, 'cardAudit', { message: this.safeLine(message, 300) }, this.caps.cardAudit); },
    context(root, message) { return this.append(root, 'contextAudit', { message: this.safeLine(message, 300) }, this.caps.contextAudit); },
    cleanup(root, message) { return this.append(root, 'cleanupAudit', { message: this.safeLine(message, 300) }, this.caps.cleanupAudit); },
    migration(root, message) { return this.append(root, 'migrationAudit', { message: this.safeLine(message, 300) }, this.caps.migrationAudit); },
    release(root, releaseCheckRecord) {
      const rec = Object.assign({ turn: WL.U.nowTurn(root), result: 'WARN', passes: [], warnings: [], failures: [], blockers: [] }, releaseCheckRecord || {});
      return this.append(root, 'releasecheck', rec, this.caps.releasecheck);
    },
    turnSummary(root, summaryRecord) {
      const rec = Object.assign({
        turn: WL.U.nowTurn(root),
        inputHash: root && root.runtime ? root.runtime.lastInputHash : '',
        outputHash: root && root.runtime ? root.runtime.lastOutputHash : '',
        proposalsCreated: 0, accepted: 0, rejected: 0, duplicateSkipped: 0,
        dirtyTargets: root && root.dirty ? root.dirty.targets.slice(0, 12) : [],
        notes: []
      }, summaryRecord || {});
      return this.append(root, 'lastTurnSummary', rec, this.caps.lastTurnSummary);
    },
    repair(root, message) {
      const row = this.append(root, 'repairAudit', { message: this.safeLine(message, 300) }, this.caps.repairAudit);
      this.warn(root, 'repair', message);
      return row;
    },
    summarizeLastTurn(root, message) { return this.turnSummary(root, { notes: [this.safeLine(message, 240)], message: this.safeLine(message, 240) }); },
    inspect(root) {
      this.ensure(root);
      const rows = [];
      rows.push({ status: 'PASS', label: 'AuditLogger lock present: ' + WL.AUDIT_LEDGER_LOCK_VERSION });
      ['debugMessages','runtimeWarnings','runtimeErrors','recentProposals','recentValidationResults','duplicateAudit','lastTurnSummary','releasecheck'].forEach(k => rows.push(Array.isArray(root.audits[k]) ? { status: 'PASS', label: 'audit bucket present/capped: ' + k } : { status: 'FAIL', label: 'audit bucket missing: ' + k }));
      rows.push(typeof this.safeLine === 'function' ? { status: 'PASS', label: 'safeLine present' } : { status: 'FAIL', label: 'safeLine missing' });
      rows.push(typeof this.capAll === 'function' ? { status: 'PASS', label: 'capAll present' } : { status: 'FAIL', label: 'capAll missing' });
      return rows;
    },
    runBC7SelfTests() {
      const results = [];
      const add = (id, ok, label) => results.push({ id, ok: !!ok, label });
      const st = {}; const r = WL.StateManager.init(st);
      this.debug(r, 'test', 'debug message'); add('BC7-A01', r.audits.debugMessages.length === 1, 'debug audit records');

      this.runtimeError(r, { hook: 'output', phase: 'forced', message: 'boom', safeTextReturned: true, partialCommitBlocked: true }); add('BC7-A03', r.audits.runtimeErrors.length >= 1 && r.runtime.runtimeErrors.length >= 1, 'runtime error structured');
      this.proposal(r, { family: 'inventory', reason: 'proposal test', preview: 'test' }); add('BC7-A04', r.audits.recentProposals.length === 1, 'proposal audit records');

        if (!root.runtime || typeof root.runtime.lastAppliedOutputHash === 'undefined' || typeof root.runtime.suppressEventScanForCommandOutput === 'undefined') errors.push('runtime output/command suppression fields missing');

  WL.OperationLedger = {
    bucketAliases: Object.freeze({
      permanent: 'permanentKeys', permanentKeys: 'permanentKeys',
      recent: 'recentKeys', recentKeys: 'recentKeys',
      reward: 'rewardKeys', rewards: 'rewardKeys', rewardKeys: 'rewardKeys',
      card: 'cardKeys', cards: 'cardKeys', cardKeys: 'cardKeys',
      source: 'sourceUseKeys', sourceUse: 'sourceUseKeys', sourceUseKeys: 'sourceUseKeys'
    }),
    normalizeKey(value) {
      const key = WL.U.slug(String(value == null ? '' : value)).slice(0, 180);
      return key || 'empty_key';
    },
    makeKey(parts) {
      const key = (parts || []).map(p => this.normalizeKey(p)).filter(Boolean).join('|').slice(0, 260);
      return key || 'empty_key';
    },
    ensure(root) {
      if (!root || typeof root !== 'object') return null;
      if (!root.operationLedger || typeof root.operationLedger !== 'object' || Array.isArray(root.operationLedger)) root.operationLedger = {};
      const led = root.operationLedger;
      ['permanentKeys','rewardKeys','cardKeys','sourceUseKeys'].forEach(k => {
        if (Array.isArray(led[k])) {
          const obj = {};
          led[k].forEach(x => { obj[this.normalizeKey(x)] = { turn: WL.U.nowTurn(root), migratedFromArray: true }; });
          led[k] = obj;
        }
        if (!led[k] || typeof led[k] !== 'object') led[k] = {};
      });
      if (!Array.isArray(led.recentKeys)) led.recentKeys = [];
      if (!Array.isArray(led.duplicateSkips)) led.duplicateSkips = [];
      if (!Array.isArray(led.ledgerWarnings)) led.ledgerWarnings = [];
      led.maxRecentKeys = WL.U.clamp(led.maxRecentKeys || WL.SCHEMA_CAPS.recentLedger || 180, 20, 1000);
      led.ledgerLockVersion = WL.AUDIT_LEDGER_LOCK_VERSION;
      // Do not call compactRecent() here; compactRecent() itself needs ledger access.
      while (led.recentKeys.length > led.maxRecentKeys) led.recentKeys.shift();
      return led;
    },
    bucketName(name) { return this.bucketAliases[name] || name || 'recentKeys'; },
    bucket(root, name) {
      const led = this.ensure(root);
      const bucket = this.bucketName(name);
      if (bucket === 'recentKeys') return led.recentKeys;
      if (!led[bucket] || typeof led[bucket] !== 'object' || Array.isArray(led[bucket])) led[bucket] = {};
      return led[bucket];
    },
    has(root, bucket, key) {
      key = this.normalizeKey(key);
      const b = this.bucketName(bucket);
      if (b === 'recentKeys') return this.hasRecent(root, key);
      return !!this.bucket(root, b)[key];
    },
    hasPermanent(root, key) { return !!this.bucket(root, 'permanentKeys')[this.normalizeKey(key)]; },
    hasRecent(root, key) { return this.bucket(root, 'recentKeys').indexOf(this.normalizeKey(key)) >= 0; },
    hasReward(root, key) { return !!this.bucket(root, 'rewardKeys')[this.normalizeKey(key)]; },
    hasCard(root, key) { return !!this.bucket(root, 'cardKeys')[this.normalizeKey(key)]; },
    hasSourceUse(root, key) { return !!this.bucket(root, 'sourceUseKeys')[this.normalizeKey(key)]; },
    hasAny(root, key) {
      key = this.normalizeKey(key);
      return this.hasPermanent(root, key) || this.hasRecent(root, key) || this.hasReward(root, key) || this.hasCard(root, key) || this.hasSourceUse(root, key);
    },
    add(root, bucket, key, data) {
      key = this.normalizeKey(key);
      const b = this.bucketName(bucket);
      if (b === 'recentKeys') return this.addRecent(root, key, data);
      const obj = this.bucket(root, b);
      if (obj[key]) return false;
      obj[key] = Object.assign({ turn: WL.U.nowTurn(root), bucket: b }, (data && typeof data === 'object') ? data : { reason: data || '' });
      return true;
    },
    addPermanent(root, key, data) { return this.add(root, 'permanentKeys', key, data); },
    addRecent(root, key, data) {
      const led = this.ensure(root);
      key = this.normalizeKey(key);
      if (led.recentKeys.indexOf(key) >= 0) return false;
      led.recentKeys.push(key);
      this.compactRecent(root);
      return true;
    },
    addReward(root, key, data) { return this.add(root, 'rewardKeys', key, data); },
    addCard(root, key, data) { return this.add(root, 'cardKeys', key, data); },
    addSourceUse(root, key, data) { return this.add(root, 'sourceUseKeys', key, data); },
    auditDuplicate(root, key, reason) {
      const led = this.ensure(root);
      const rec = { turn: WL.U.nowTurn(root), key: this.normalizeKey(key), reason: WL.AuditLogger ? WL.AuditLogger.safeLine(reason || 'duplicate key skipped', 200) : String(reason || 'duplicate key skipped') };
      led.duplicateSkips.push(rec);
      led.lastDuplicate = rec;
      WL.U.capArray(led.duplicateSkips, 50);
      if (WL.AuditLogger) WL.AuditLogger.duplicate(root, rec.key, rec.reason);
      return rec;
    },
    tryAdd(root, bucketName, key, reason) {

        return { applied: false, duplicate: true, key: normalized, bucket };
      }
      this.add(root, bucket, normalized, (reason && typeof reason === 'object') ? reason : { reason: reason || '' });
      return { applied: true, duplicate: false, key: normalized, bucket };
    },
    compactRecent(root) {
      const led = this.ensure(root);

      for (let i = 0; i < led.recentKeys.length; i++) {
        const k = this.normalizeKey(led.recentKeys[i]);
        if (!seen[k]) { seen[k] = true; compact.push(k); }
      }
      while (compact.length > led.maxRecentKeys) compact.shift();
      led.recentKeys = compact;
      return compact.length;
    },
    summary(root) {

      return [

    },

    },

  WL.Runtime = {
    safeText(value) {
      if (value == null) return '';
      if (typeof value === 'string') return value;
      if (typeof value === 'number' || typeof value === 'boolean') return String(value);
      try { return String(value); } catch (e) { return ''; }
    },
    hashText(text) { return WL.U.hash(this.safeText(text)); },
    makeReturn(text) {
      const safe = this.safeText(text);
      return safe || ' ';
    },
    validateHookInput(hookName, text, rawState) {
      const h = WL.U.slug(hookName || 'unknown');
      const okHook = h === 'input' || h === 'context' || h === 'output';
      return { ok: okHook, hook: okHook ? h : 'unknown', text: this.safeText(text), rawStateOk: !rawState || typeof rawState === 'object' };
    },
    pushHookTrace(root, record) {
      if (!root || !root.runtime) return null;
      if (!Array.isArray(root.runtime.hookTrace)) root.runtime.hookTrace = [];
      const rec = Object.assign({ turn: root.runtime.currentTurn || 0, traceId: 'trace_' + WL.U.hash(String(root.runtime.currentTurn || 0) + '|' + String(root.runtime.hookTrace.length) + '|' + String(record && record.hook || 'hook')), status: 'started' }, record || {});
      root.runtime.hookTrace.push(rec);
      root.runtime.activeTraceId = rec.traceId;
      this.capRuntime(root);
      return rec;
    },
    updateTrace(root, traceId, status, notes) {
      if (!root || !root.runtime || !Array.isArray(root.runtime.hookTrace)) return;
      for (let i = root.runtime.hookTrace.length - 1; i >= 0; i--) {
        const rec = root.runtime.hookTrace[i];
        if (!traceId || rec.traceId === traceId) {
          rec.status = status || rec.status;
          rec.endedAt = WL.U.hash(String(status || '') + '|' + String(notes || '') + '|' + String(root.runtime.currentTurn || 0));
          if (notes) rec.notes = WL.U.norm(notes).slice(0, 160);
          return;
        }
      }
    },
    recordWarning(root, message, data) {
      if (!root || !root.runtime) return;
      if (!Array.isArray(root.runtime.runtimeWarnings)) root.runtime.runtimeWarnings = [];
      const row = { turn: WL.U.nowTurn(root), message: WL.U.norm(message), data: data || null };
      root.runtime.runtimeWarnings.push(row);
      if (!Array.isArray(root.runtime.warnings)) root.runtime.warnings = [];
      root.runtime.warnings.push(row);
      this.capRuntime(root);
      try { WL.AuditLogger.append(root, 'runtimeWarnings', { message: row.message, data: data || null }, 30); } catch (e) {}
    },
    recordError(root, hookName, phase, message, safeTextReturned, partialCommitBlocked) {
      if (!root || !root.runtime) return;
      if (!Array.isArray(root.runtime.runtimeErrors)) root.runtime.runtimeErrors = [];
      const row = { turn: WL.U.nowTurn(root), hook: hookName || 'unknown', phase: phase || 'unknown', message: WL.U.norm(message), safeTextReturned: !!safeTextReturned, partialCommitBlocked: partialCommitBlocked !== false };
      root.runtime.runtimeErrors.push(row);
      if (!Array.isArray(root.runtime.errors)) root.runtime.errors = [];
      root.runtime.errors.push(row);
      this.capRuntime(root);
      try { WL.AuditLogger.append(root, 'runtimeWarnings', { message: 'ERROR ' + row.hook + '/' + row.phase + ': ' + row.message }, 30); } catch (e) {}
    },
    capRuntime(root) {
      if (!root || !root.runtime) return;
      WL.U.capArray(root.runtime.hookTrace || (root.runtime.hookTrace = []), WL.SCHEMA_CAPS.hookTrace);
      WL.U.capArray(root.runtime.runtimeWarnings || (root.runtime.runtimeWarnings = []), WL.SCHEMA_CAPS.runtimeWarnings);
      WL.U.capArray(root.runtime.runtimeErrors || (root.runtime.runtimeErrors = []), WL.SCHEMA_CAPS.runtimeErrors);
      WL.U.capArray(root.runtime.warnings || (root.runtime.warnings = []), WL.SCHEMA_CAPS.runtimeWarnings);
      WL.U.capArray(root.runtime.errors || (root.runtime.errors = []), WL.SCHEMA_CAPS.runtimeErrors);
      WL.U.capArray(root.runtime.suppressionAudit || (root.runtime.suppressionAudit = []), 30);
      WL.U.capArray(root.runtime.returnAudit || (root.runtime.returnAudit = []), 30);
    },
    guardReentry(root, hookName) {
      const depth = Number(root && root.runtime && root.runtime.hookDepth || 0);
      if (depth > 0) {
        root.runtime.reentryBlocked = true;
        root.runtime.mutationBlocked = true;
        this.recordWarning(root, 'Hook reentry detected in ' + hookName + '; gameplay mutation blocked for safety.', { depth });
        return false;
      }
      return true;
    },
    beginHook(root, hookName, text) {
      const validation = this.validateHookInput(hookName, text, globalThis.state);
      const hook = validation.hook;
      const safe = validation.text;
      if (!root || !root.runtime) return { hook, text: safe, hash: this.hashText(safe), traceId: '' };
      this.guardReentry(root, hook);
      root.runtime.hookDepth = WL.U.clamp((root.runtime.hookDepth || 0) + 1, 0, 5);
      root.runtime.currentHook = hook;
      root.runtime.runtimeLockVersion = WL.RUNTIME_HOOK_CONTRACT_VERSION;
      root.startup.firstHookSeen = true;
      const actionCount = WL.U.getInfoActionCount();
      if (typeof actionCount === 'number') root.runtime.currentTurn = actionCount;
      else if (typeof root.runtime.currentTurn !== 'number') root.runtime.currentTurn = root.meta.currentTurn || 0;
      root.meta.currentTurn = root.runtime.currentTurn;
      const hash = this.hashText(safe);

  WL.Sanitizers = {
    stripUnsafeTail(value) {
      return WL.U.norm(value)
        .replace(/["“”]+/g, '')
        .replace(/\s+(?:to|by|while|when|as|because|before|after|until|with|without|near|from|inside|outside|towards?|against|for)\b[\s\S]*$/i, '')
        .replace(/[.,;:!?]+$/g, '')
        .trim();
    },
    item(value) {
      let v = WL.U.norm(value).replace(/^(?:the|a|an|some|your|my|his|her|their)\s+/i, '');
      v = v.replace(/\s+(?:from|near|on|in|inside|at|beside|under|over|within|by|with|using|for|toward|towards|against)\b[\s\S]*$/i, '');
      v = v.replace(/[.,;:!?]+$/g, '').trim();
      const l = v.toLowerCase();
      if (!l || l.length > 70) return '';
      if (/^(?:idea|concept|memory|rumor|dream|nothing|anything|something|it|them|this|that)$/i.test(l)) return '';
      if (/\b(?:idea of|concept of|thought of|rumor of|dream of)\b/i.test(l)) return '';
      return WL.U.title(v);
    },
    ability(value) {
      const raw = WL.U.norm(value);
      const m = raw.match(/\b([A-Z][A-Za-z'’-]{1,24}(?:\s+[A-Z][A-Za-z'’-]{1,24}){0,3}\s+(?:Ward|Slash|Strike|Bolt|Sense|Sight|Step|Stance|Form|Art|Barrier|Shield|Control|Pulse|Veil|Blade|Guard|Guarding|Flame|Spark|Flow|Grip))\b/);
      let v = m ? m[1] : raw;
      v = this.stripUnsafeTail(v).replace(/\bGuarding$/i, 'Guard');
      if (!/[A-Z]/.test(v)) return '';
      if (/\b(?:travel|smithy|door|key|inspect|open|road|street|inventory|quest|gold|copper)\b/i.test(v)) return '';
      if (v.split(/\s+/).length > 5) return '';
      return WL.U.title(v);
    },
    actor(value) {
      let v = WL.U.norm(value).replace(/^(?:the|a|an)\s+/i, '');
      v = v.replace(/[.,;:!?]+$/g, '').trim();
      if (!/^[A-Z][A-Za-z'’-]{1,30}(?:\s+[A-Z][A-Za-z'’-]{1,30}){0,2}$/.test(v)) return '';
      if (/^(?:The|You|Your|He|She|They|It|Guard|Veteran Guard|Innkeeper|Merchant)$/i.test(v)) return '';
      return WL.U.title(v);
    },
    place(value) {
      let v = WL.U.norm(value);
      v = v.replace(/\s+(?:while|as|where|when|with|without|near|beside|under|over)\b[\s\S]*$/i, '').trim();
      v = v.replace(/[.,;:!?]+$/g, '').trim();
      if (!v || v.length > 90) return '';
      if (/^(?:road|street|door|room|common room|wall|floor|body|hand|market lane|the road)$/i.test(v)) return '';
      const m = v.match(/\b(?:enter|arrive at|reach|inside)\s+(?:the\s+)?([A-Z][A-Za-z'’ -]{2,70})/i);
      if (m) v = m[1];
      return WL.U.title(v.replace(/^the\s+/i, ''));
    },
    quest(value) {
      let v = WL.U.norm(value).replace(/^to\s+/i, '');
      v = v.replace(/["“”]+/g, '').replace(/[.;]+$/g, '').trim();
      if (!v || v.length < 3 || v.length > 120) return '';
      return v.charAt(0).toUpperCase() + v.slice(1);
    },
    reward(value) {
      const v = WL.U.norm(value).match(/(\d{1,6})\s*(copper|silver|gold)\b/i);
      if (!v) return null;
      return { amount: Number(v[1]), coin: v[2].toLowerCase() };
    },
    cardTrigger(value) {
      const v = WL.U.norm(value);
      if (!v || /^(?:worldledger|aidrpg|system|settings|debug|street|door|room)$/i.test(v)) return '';
      return v.slice(0, 80);
    }
  };

  WL.CommandOutputFormatLock = {
    statusLabels: Object.freeze({ ACTIVE: 'ACTIVE', DEFERRED: 'DEFERRED', UNKNOWN: 'UNKNOWN', ERROR: 'ERROR' }),
    noInstallPattern: /\b(?:paste\s+.*(?:input|context|output)|install\s+instructions?|wrapper\s+tabs?|put\s+this\s+in\s+(?:input|context|output)|globalThis\.WorldLedger\.onInput\s*\()\b/i,
    ensure(root) {
      if (!root.commandFormat || typeof root.commandFormat !== 'object' || Array.isArray(root.commandFormat)) {
        const fresh = WL.Schema.root().commandFormat;
        root.commandFormat = fresh;
      }
      const cf = root.commandFormat;
      cf.formatVersion = WL.COMMAND_FORMAT_LOCK_VERSION;
      cf.locked = true;
      cf.prefix = '[WorldLedger]';
      if (!Array.isArray(cf.warnings)) cf.warnings = [];
      if (!Array.isArray(cf.rules)) cf.rules = WL.Schema.root().commandFormat.rules.slice();

      const lines = String(raw == null ? '' : raw).replace(/\r\n/g, '\n').split('\n');
      if (lines[0] && /^\[WorldLedger\]/.test(lines[0])) lines.shift();
      return lines.join('\n').trim();

      const first = String(raw || '').split(/\n/)[0] || '';
      const m = first.match(/^\[WorldLedger\]\s*(.*)$/);
      const title = WL.U.norm(m ? m[1] : '');
      if (title) return title;
      return '/' + WL.U.norm(name || 'command');
    },
    enforce(root, name, raw, status) {
      const cf = this.ensure(root);
      name = WL.U.slug(name || 'command');
      status = this.statusLabels[status] || status || 'ACTIVE';
      let body = this.stripOldHeader(raw);
      if (!body) body = 'No details available.';
      if (this.noInstallPattern.test(body)) {
        cf.warnings.push({ turn: WL.U.nowTurn(root), command: name, issue: 'hook_wrapper_text_removed' });
        WL.U.capArray(cf.warnings, 20);
        body = body.replace(this.noInstallPattern, '[removed hook-wrapper snippet]').trim() || 'Output contained blocked hook-wrapper wording and was removed.';
      }
      const title = this.titleFromRaw(name, raw).replace(/^\//, '');
      const out = [
        '[WorldLedger] /' + name + ' — ' + title,
        'Status: ' + status,
        'Mode: inspect-only; command output is non-canon and suppressed from story parsing.',
        'Format: ' + WL.COMMAND_FORMAT_LOCK_VERSION,
        '',
        body
      ].join('\n').replace(/\n{4,}/g, '\n\n\n').trim() + '\n';
      cf.lastCommandRendered = name;
      cf.lastStatus = status;
      cf.lastLength = out.length;
      return out || '[WorldLedger] /' + name + ' — Empty command\nStatus: ERROR\nMode: inspect-only; command output is non-canon.\nFormat: ' + WL.COMMAND_FORMAT_LOCK_VERSION + '\n\nEmpty command output repaired.\n';
    },
    inspect(root) {
      const cf = this.ensure(root);
      const rows = [];
      rows.push(cf.formatVersion === WL.COMMAND_FORMAT_LOCK_VERSION ? { status: 'PASS', label: 'command format version locked: ' + WL.COMMAND_FORMAT_LOCK_VERSION } : { status: 'FAIL', label: 'command format version mismatch' });
      rows.push(cf.locked === true ? { status: 'PASS', label: 'command format locked' } : { status: 'FAIL', label: 'command format unlocked' });
      rows.push(root.features && root.features.commandFormatLockEnabled === true ? { status: 'PASS', label: 'command format feature enabled' } : { status: 'FAIL', label: 'command format feature disabled' });
      rows.push(WL.MVP_ACTIVE_COMMANDS.indexOf('commandformat') >= 0 ? { status: 'PASS', label: '/commandformat registered' } : { status: 'FAIL', label: '/commandformat missing' });
      rows.push(this.noInstallPattern.test(WL.SourceText || '') ? { status: 'FAIL', label: 'source contains hook-wrapper snippets' } : { status: 'PASS', label: 'no hook-wrapper snippets detected by format lock' });
      const sample = this.enforce(root, 'format_sample', '[WorldLedger] Format Sample\nSample body.', 'ACTIVE');
      rows.push(/^\[WorldLedger\] \/format_sample/.test(sample) ? { status: 'PASS', label: 'formatted sample uses locked header' } : { status: 'FAIL', label: 'formatted sample header failed' });
      rows.push(/Status: ACTIVE/.test(sample) ? { status: 'PASS', label: 'formatted sample includes status' } : { status: 'FAIL', label: 'formatted sample missing status' });
      rows.push(/non-canon/.test(sample) ? { status: 'PASS', label: 'formatted sample includes non-canon warning' } : { status: 'FAIL', label: 'formatted sample missing non-canon warning' });
      return rows;
    },
    render(root) {
      const cf = this.ensure(root);
      const rows = this.inspect(root);
      const fail = rows.filter(r => r.status === 'FAIL').length;
      const lines = [
        'Command Output Format Lock: ' + WL.COMMAND_FORMAT_LOCK_VERSION + (fail ? ' / FAILURES: ' + fail : ' / locked'),
        'Prefix: ' + cf.prefix,
        'Mode line: ' + cf.modeLine,
        'Last command rendered: ' + (cf.lastCommandRendered || 'none'),
        'Rules:'
      ];
      (cf.rules || []).forEach(r => lines.push('- ' + r));
      lines.push('Audit:');
      rows.forEach(r => lines.push(r.status + ': ' + r.label));
      return lines;
    }
  };

  // ============================================================
  // 10. EventScannerLite Parser Shell
  // ============================================================
  WL.EventScannerLite = {
    lockVersion: 'worldledger.eventscanner-lock.v9',
    allowedSources: Object.freeze({ input: true, output: true }),
    disallowedSources: Object.freeze({
      command_output: true, context: true, story_card: true, protected_card: true, generated_card: true,
      debug: true, audit: true, help: true, releasecheck: true, migration_log: true, old_log: true,
      archive: true, unknown_noncanon: true, command: true
    }),
    familyOrder: Object.freeze(['inventory','ability','body','quest','reward','actor','place','time','progression']),
    unsafeSource(kind, text) {
      const sourceInfo = this.makeSourceInfo(kind || 'unknown_noncanon', null, text || '');
      const allow = this.shouldScanSource(sourceInfo, null);
      return !allow.allowed;
    },
    makeSourceInfo(sourceHook, root, text) {
      const source = WL.U.slug(sourceHook || 'unknown').replace(/_/g, '_');
      const raw = WL.U.norm(text || '');
      return {
        sourceHook: source,
        turn: WL.U.nowTurn(root),
        actionCount: WL.U.nowTurn(root),
        textHash: WL.U.hash(raw),
        isInput: source === 'input',
        isOutput: source === 'output',
        isCommandOutput: this.isCommandOutput(raw, root),
        isDiagnosticText: this.isAidRpgDiagnosticText(raw),
        isContextLike: source === 'context',
        isCardLike: /card/.test(source),
        sourceLength: raw.length
      };
    },
    shouldScanSource(sourceInfo, root) {
      sourceInfo = sourceInfo || {};
      const hook = sourceInfo.sourceHook || sourceInfo.kind || 'unknown_noncanon';
      if (this.disallowedSources[hook]) return { allowed: false, reason: 'disallowed source: ' + hook };
      if (!this.allowedSources[hook]) return { allowed: false, reason: 'source not allowlisted: ' + hook };
      if (root && root.runtime && root.runtime.suppressEventScanForCommandOutput) return { allowed: false, reason: 'runtime command-output suppression active' };
      if (sourceInfo.isCommandOutput) return { allowed: false, reason: 'command output text' };
      if (sourceInfo.isDiagnosticText) return { allowed: false, reason: 'diagnostic text' };
      return { allowed: true, reason: 'allowlisted ' + hook };
    },
    scanInput(text, root) {
      // Input creates intent/attempt hits only. It never emits completed success hits.
      return this.scanText(text, 'input', root);
    },
    scanOutput(a, b) {
      // Backwards compatible: scanOutput(root, output) or scanOutput(output, root).
      let root, text;
      if (a && typeof a === 'object' && !Array.isArray(a)) { root = a; text = b; }
      else { text = a; root = b; }
      return this.scanText(text, 'output', root);
    },
    scanText(text, sourceHook, root) {
      root = root || WL.StateManager.root();
      const sourceInfo = this.makeSourceInfo(sourceHook, root, text);
      const allow = this.shouldScanSource(sourceInfo, root);
      if (!allow.allowed) {
        const rejected = this.rejectNonCanonSource(allow.reason, sourceInfo);
        if (sourceInfo.sourceHook === 'output') root.pending.lastParserHits = [];
        if (sourceInfo.sourceHook === 'input') root.pending.lastInputParserHits = [];
        this.auditScan(root, { sourceHook: sourceInfo.sourceHook, allowed: false, reason: allow.reason, hitCount: 0, textHash: sourceInfo.textHash });
        return rejected;
      }
      const windows = this.splitIntoWindows(text).slice(0, this.maxWindows(root));
      const hits = [];
      for (let i = 0; i < windows.length && hits.length < this.maxHits(root, sourceInfo.sourceHook); i++) {
        const w = windows[i];
        const flags = this.detectSafetyFlags(w.text);
        const familyHits = this.detectFamilyHints(w.text, flags, sourceInfo.sourceHook, sourceInfo, i);
        for (let j = 0; j < familyHits.length && hits.length < this.maxHits(root, sourceInfo.sourceHook); j++) hits.push(this.makeHit(familyHits[j]));
      }
      const capped = this.capHits(hits, this.maxHits(root, sourceInfo.sourceHook));
      if (sourceInfo.sourceHook === 'output') root.pending.lastParserHits = capped;
      if (sourceInfo.sourceHook === 'input') root.pending.lastInputParserHits = capped;
      this.auditScan(root, { sourceHook: sourceInfo.sourceHook, allowed: true, windows: windows.length, hitCount: capped.length, families: capped.map(h => h.family).join(','), textHash: sourceInfo.textHash });
      return capped;
    },
    maxWindows(root) { return WL.U.clamp(root && root.features && root.features.maxParserWindows || 16, 4, 32); },
    maxHits(root, sourceHook) {
      return WL.U.clamp(root && root.features && (sourceHook === 'input' ? root.features.maxInputParserHits : root.features.maxOutputParserHits) || (sourceHook === 'input' ? 12 : 20), 4, 40);
    },
    splitIntoWindows(text) {
      const raw = WL.U.norm(text || '');
      if (!raw) return [];
      const parts = raw
        .replace(/([.!?;])\s+/g, '$1|')
        .split('|')
        .map((x, i) => ({ index: i, text: WL.U.norm(x).slice(0, 360) }))
        .filter(w => w.text.length > 0);
      return parts.slice(0, 32);
    },
    detectSafetyFlags(windowText) {
      const raw = WL.U.norm(windowText || '');
      const unq = WL.U.unquoted(raw);
      const quotedSpeech = raw !== unq || /["“”]/.test(raw);
      const dreamOrVision = /\b(?:dream|dreamed|dreamt|nightmare|vision|hallucination|imagined|imaginary|illusion)\b/i.test(raw);
      const rumorOrClaim = /\b(?:rumor|rumour|people say|they say|it is said|heard that|supposedly|allegedly|claims?|claimed|legend says)\b/i.test(raw);
      const hypothetical = /\b(?:could|would|might|may|if you wanted|if you choose|possible|perhaps|maybe|almost|nearly)\b/i.test(raw);
      const futureIntent = /\b(?:will|going to|plans? to|promises? to|tomorrow|later|soon|eventually|afterward|afterwards)\b/i.test(raw);
      const failedOrRevoked = /\b(?:fails?|failed|nothing happens|no visible effect|does not work|doesn't work|cannot|can't|unable|blocked|interrupted|stops you|snatches? .* away|no item is gained|no ability is gained)\b/i.test(raw);
      const negatedOrRefused = /\b(?:do not|don't|did not|didn't|decide not|refuse|refuses|declines?|avoid|leave it|let it go|instead of taking|not take|not pick up)\b/i.test(raw);
      const pretendOrFalse = /\b(?:pretend|joke|make believe|only mimicking|not real|fake|as a joke)\b/i.test(raw);
      const abstractOrIdea = /\b(?:idea of|concept of|thought of|memory of|symbol of|image of|dream of)\b/i.test(raw);
      const visibleOnly = /\b(?:see|saw|notice|visible|lies|rests|sits|on the shelf|nearby|in the distance|could take|available)\b/i.test(raw) && !/\b(?:pick up|take|grab|collect|accept|hands you|gives you|places .* in your hand)\b/i.test(raw);
      const nonliteral = /\b(?:learning experience|time flies|felt like years|eternity|forever|in your mind|as if)\b/i.test(raw);
      const commandText = this.isCommandOutput(raw, null) || WL.U.startsCommand(raw);
      const diagnosticText = this.isAidRpgDiagnosticText(raw);
      return {
        rawPreview: raw.slice(0, 180),
        localClausePreview: raw.slice(0, 180),
        quotedSpeech, dreamOrVision, rumorOrClaim, hypothetical, futureIntent, failedOrRevoked,
        negatedOrRefused, pretendOrFalse, abstractOrIdea, visibleOnly, nonliteral, commandText, diagnosticText,
        // Compatibility aliases for existing ProposalBuilder/CanonValidator shell.
        quote: quotedSpeech, dream: dreamOrVision, rumor: rumorOrClaim, pretend: pretendOrFalse,
        denied: failedOrRevoked || negatedOrRefused, visibleOnly: visibleOnly
      };
    },
    completedAllowed(flags, sourceHook) {
      if (sourceHook !== 'output') return false;
      if (!flags) return sourceHook === 'output';
      if (flags.quotedSpeech || flags.dreamOrVision || flags.rumorOrClaim || flags.hypothetical || flags.futureIntent || flags.failedOrRevoked || flags.negatedOrRefused || flags.pretendOrFalse || flags.abstractOrIdea || flags.commandText || flags.diagnosticText) return false;
      return true;
    },
    baseHit(family, type, name, windowText, flags, sourceHook, sourceInfo, extra) {
      const safeCompleted = this.completedAllowed(flags, sourceHook);
      const truthStatus = flags.rumorOrClaim ? 'rumored' : (flags.dreamOrVision ? 'dream' : (flags.quotedSpeech ? 'quoted' : (safeCompleted ? 'candidate_confirmed_by_output' : 'intent_or_unsafe_candidate')));
      return Object.assign({
        parserHitVersion: WL.EVENT_SCANNER_LOCK_VERSION,
        family, type,
        name: WL.U.norm(name || ''),
        sourceText: WL.U.norm(windowText || '').slice(0, 240),
        localClausePreview: WL.U.norm(windowText || '').slice(0, 180),
        flags: Object.assign({}, flags || {}),
        sourceHook,
        sourceInfo: Object.assign({}, sourceInfo || {}),
        confidence: safeCompleted ? 0.72 : 0.35,
        completedAction: safeCompleted,
        attemptOnly: sourceHook === 'input' || !safeCompleted,
        truthStatus,
        parserOnly: true,
        canCommitDirectly: false
      }, extra || {});
    },
    detectFamilyHints(windowText, flags, sourceHook, sourceInfo, windowIndex) {
      const hits = [];
      const w = WL.U.norm(windowText || '');
      if (!w || flags.commandText || flags.diagnosticText) return hits;
      const completed = this.completedAllowed(flags, sourceHook);
      const sourceAttempt = sourceHook === 'input';
      let m;

      // Inventory: visible, attempt, handoff, and pickup are only hits. They never become owned items here.
      if ((m = w.match(/\b(?:hands you|gives you|passes you|places|puts)\s+(?:the\s+|a\s+|an\s+)?([a-z][a-z' -]{2,60}?)(?:\s+(?:in your hand|into your hand|to you))\b/i))) {
        hits.push(this.baseHit('inventory', completed ? 'item.acquire' : 'item.handoff_candidate', m[1], w, flags, sourceHook, sourceInfo, { mode: 'handoff', windowIndex, completedAction: completed, attemptOnly: !completed }));
      }
      const pickupRe = /\b(?:you\s+)?(?:pick up|take|grab|retrieve|pocket|collect|accept)\s+(?:the\s+|a\s+|an\s+|some\s+)?([a-z][a-z' -]{2,60}?)(?:\s+(?:from|near|on|in|and|with|before|while)\b|[.!?]|$)/gi;
      while ((m = pickupRe.exec(w)) && hits.length < 8) {
        hits.push(this.baseHit('inventory', completed ? 'item.acquire' : 'item.take_attempt', m[1], w, flags, sourceHook, sourceInfo, { mode: 'pickup', windowIndex, completedAction: completed, attemptOnly: !completed || sourceAttempt }));
      }
      if ((m = w.match(/\b(?:you see|you notice|there is|there are|lies|rests|sits)\s+(?:the\s+|a\s+|an\s+|some\s+)?([a-z][a-z' -]{2,60}?)(?:\s+(?:on|near|beside|under|inside|at)\b|[.!?]|$)/i))) {
        hits.push(this.baseHit('inventory', 'item.visible', m[1], w, flags, sourceHook, sourceInfo, { mode: 'visible', windowIndex, completedAction: false, attemptOnly: false, truthStatus: 'visible_reference', confidence: 0.55 }));
      }

      // Ability: attempts/failures become evidence hits only, never stable ability truth.
      if ((m = w.match(/\b([A-Z][A-Za-z'’-]{1,24}(?:\s+[A-Z][A-Za-z'’-]{1,24}){0,3}\s+(?:Ward|Slash|Strike|Bolt|Sense|Sight|Step|Stance|Form|Art|Barrier|Shield|Control|Pulse|Veil|Blade|Guard))\b/)) && /\b(?:train|practice|attempt|try|focus|channel|use|learn|unlock|nothing happens|no visible effect)\b/i.test(w)) {
        const abilityType = /\b(?:learn|learned|unlock|unlocked|gain|gained)\b/i.test(w) ? 'ability.stabilize_claim' : 'ability.evidence';
        hits.push(this.baseHit('ability', abilityType, m[1], w, flags, sourceHook, sourceInfo, { windowIndex, attemptedInInput: sourceHook === 'input', completedAction: false, attemptOnly: true, confidence: flags.failedOrRevoked ? 0.7 : 0.55 }));
      }

      // Body/death: direct harm gets candidate hits; rumors/dreams/quotes are flagged.
      if (/\b(?:exhausted|fatigued|tired|worn out|out of breath|muscles burn)\b/i.test(w)) hits.push(this.baseHit('body', 'body.fatigue', 'fatigue', w, flags, sourceHook, sourceInfo, { windowIndex, severity: 1 }));
      if (/\b(?:wolf bites|bitten|stabbed|deep cut|bleeding|blood runs|burned|poisoned|arrow hits|sword cuts|breaks your arm|crushes your leg)\b/i.test(w)) hits.push(this.baseHit('body', 'body.injury', 'injury', w, flags, sourceHook, sourceInfo, { windowIndex, severity: 2 }));
      if (/\b(?:you died|you die|you are dead|kills you|killed you|slain)\b/i.test(w)) hits.push(this.baseHit('body', 'body.death_claim', 'death', w, flags, sourceHook, sourceInfo, { windowIndex, severity: 5, completedAction: false, attemptOnly: true }));

      // Quest/reward split: promise and payment are separate families/types.
      if ((m = w.match(/\b(?:quest|task|job|mission)\s+(?:to\s+)?([^.!?;]{3,100})/i))) hits.push(this.baseHit('quest', 'quest.offer', m[1], w, flags, sourceHook, sourceInfo, { windowIndex }));
      if ((m = w.match(/\b(?:accept|agree to|take on|undertake)\s+(?:the\s+)?(?:quest|task|job|mission)\s+(?:to\s+)?([^.!?;]{3,100})/i))) hits.push(this.baseHit('quest', 'quest.accept', m[1], w, flags, sourceHook, sourceInfo, { windowIndex }));
      if ((m = w.match(/\b(?:complete|completed|finish|finished)\s+(?:the\s+)?(?:quest|task|job|mission)\s+(?:to\s+)?([^.!?;]{3,100})/i))) hits.push(this.baseHit('quest', 'quest.complete', m[1], w, flags, sourceHook, sourceInfo, { windowIndex }));
      if ((m = w.match(/\b(?:promises?|vows?|offers?)\s+(?:you\s+)?(\d{1,6}\s*(?:copper|silver|gold))\b/i))) hits.push(this.baseHit('quest', 'reward.promise', m[1], w, flags, sourceHook, sourceInfo, { windowIndex, reward: WL.Sanitizers.reward(m[1]), completedAction: false, attemptOnly: true, truthStatus: 'promise_candidate' }));
      if ((m = w.match(/\b(?:pays?|paid|hands|gives)\s+(?:you\s+)?(?:(?:the\s+)?promised\s+)?(\d{1,6}\s*(?:copper|silver|gold))\b/i))) hits.push(this.baseHit('inventory', 'currency.gain', m[1], w, flags, sourceHook, sourceInfo, { windowIndex, reward: WL.Sanitizers.reward(m[1]) }));

      // Actors: named direct action/speech is a presence candidate; rumor keeps truthStatus rumored.
      const actorRe = /\b([A-Z][A-Za-z'’-]{1,30}(?:\s+[A-Z][A-Za-z'’-]{1,30}){0,2})\s+(?:says|asks|replies|nods|hands|gives|joins|follows|attacks|teaches|warns|arrives)\b/g;
      while ((m = actorRe.exec(w)) && hits.length < 10) hits.push(this.baseHit('actor', 'actor.present', m[1], w, flags, sourceHook, sourceInfo, { windowIndex, companion: /\b(?:joins|follows)\b/i.test(m[0]) }));

      // Place: input is travel intent; output strong arrival can be completed candidate; weak mention is reference only.
      if ((m = w.match(/\b(?:enter|entered|step inside|walk into|move into|head into|arrive at|arrive in|reach|return to|go inside)\s+(?:the\s+)?([A-Z][A-Za-z'’ -]{2,80})/i))) {
        const strong = /\b(?:arrive|arrived|entered|step inside|walk into|go inside|reach|return to)\b/i.test(w) && sourceHook === 'output';
        hits.push(this.baseHit('place', strong && completed ? 'place.current' : 'place.intent', m[1], w, flags, sourceHook, sourceInfo, { windowIndex, strongArrival: strong && completed, weakReference: false, completedAction: strong && completed, attemptOnly: !(strong && completed) }));
      } else if ((m = w.match(/\b([A-Z][A-Za-z'’ -]{2,80})\s+(?:is\s+)?visible\s+from\s+([A-Z][A-Za-z'’ -]{2,80})/i)) || /\b(?:visible from|can see|could see|nearby|in the distance|mentions|talks about)\b/i.test(w)) {
        const name = m && m[1] ? m[1] : 'place reference';
        hits.push(this.baseHit('place', 'place.reference', name, w, flags, sourceHook, sourceInfo, { windowIndex, weakReference: true, completedAction: false, attemptOnly: false, truthStatus: 'reference_only' }));
      }

      if ((m = w.match(/\b(\d{1,2}|one|two|three|four|five|six|seven|eight|nine|ten|twenty)\s+(minutes?|hours?|days?|weeks?|months?)\s+pass(?:es|ed)?\b/i))) hits.push(this.baseHit('time', 'time.pass', m[0], w, flags, sourceHook, sourceInfo, { windowIndex, amountText: m[1], unit: m[2].toLowerCase() }));

      return hits;
    },

  WL.ProposalBuilder = {
    lockVersion: 'worldledger.proposalbuilder-lock.v10',
    maxProposals(root) {
      return WL.U.clamp(root && root.features && root.features.maxProposalsPerHook || 20, 4, 40);
    },
    fromHits(root, hits, sourceKind) {
      root = root || WL.StateManager.root();
      const source = WL.U.slug(sourceKind || 'output') || 'output';
      const input = Array.isArray(hits) ? hits : [];
      const proposals = [];
      const seen = {};
      const max = this.maxProposals(root);
      for (let i = 0; i < input.length && proposals.length < max; i++) {
        const p = this.fromHit(root, input[i], source, i);
        if (!p) continue;
        const sig = this.signature(p);
        if (seen[sig]) continue;
        seen[sig] = true;
        proposals.push(p);
      }
      root.pending.lastProposals = proposals;
      this.audit(root, { sourceKind: source, hitCount: input.length, proposalCount: proposals.length, types: proposals.map(p => p.type).join(',') });
      return proposals;
    },
    fromHit(root, hit, sourceKind, index) {
      if (!hit || typeof hit !== 'object') return null;
      if (hit.family === 'parser' || hit.type === 'parser.source_rejected') return null;
      if (hit.canCommitDirectly === true) return null;
      const normalized = this.normalizeHit(hit, sourceKind, index);
      if (!normalized) return null;
      const target = this.bindTarget(normalized);
      const type = this.normalizeType(normalized, target);
      const owner = this.ownerFor(type, normalized.family);
      const sourceHash = normalized.sourceInfo && normalized.sourceInfo.textHash || WL.U.hash(normalized.sourceText || normalized.localClausePreview || '');
      const evidence = this.buildEvidence(normalized, target, sourceHash);
      const riskFlags = this.riskFlags(normalized, target);
      const opKey = this.makeOperationKey(root, normalized, type, target, sourceHash);
      const id = 'prop_' + WL.U.hash([type, target.id || target.raw || 'none', sourceHash, normalized.localClausePreview, normalized.sourceHook].join('|'));
      return {
        proposalVersion: WL.PROPOSAL_BUILDER_LOCK_VERSION,
        id,
        type,
        family: normalized.family,
        effectOwner: owner,
        reducerHint: owner,
        sourceKind: normalized.sourceHook || sourceKind || 'output',
        sourceText: normalized.sourceText || '',
        localClausePreview: normalized.localClausePreview || '',
        sourceInfo: normalized.sourceInfo || {},
        flags: Object.assign({}, normalized.flags || {}),
        target,
        evidence,
        riskFlags,
        opKey,
        sourceUseKey: WL.OperationLedger.makeKey(['source', normalized.sourceHook || sourceKind, sourceHash, type, target.id || target.raw || 'none']),
        dedupeKey: WL.OperationLedger.makeKey(['dedupe', type, target.id || target.raw || 'none', sourceHash, WL.U.hash(normalized.localClausePreview || '')]),
        validationReadiness: this.validationReadiness(normalized, riskFlags),
        commitPrecondition: 'CanonValidator must accept this proposal and OperationLedger must allow its opKey.',
        proposalOnly: true,
        canCommitDirectly: false,
        data: this.dataPayload(normalized)
      };
    },
    normalizeHit(hit, sourceKind, index) {
      const sourceHook = WL.U.slug(hit.sourceHook || sourceKind || 'output') || 'output';
      const sourceText = WL.U.norm(hit.sourceText || hit.localClausePreview || '').slice(0, 240);
      const localClausePreview = WL.U.norm(hit.localClausePreview || sourceText).slice(0, 180);
      const flags = Object.assign({}, hit.flags || {});
      if (flags.quotedSpeech && typeof flags.quote === 'undefined') flags.quote = true;
      if (flags.dreamOrVision && typeof flags.dream === 'undefined') flags.dream = true;
      if (flags.rumorOrClaim && typeof flags.rumor === 'undefined') flags.rumor = true;
      if (flags.pretendOrFalse && typeof flags.pretend === 'undefined') flags.pretend = true;
      if ((flags.failedOrRevoked || flags.negatedOrRefused) && typeof flags.denied === 'undefined') flags.denied = true;
      return {
        parserHitId: hit.id || 'hit_' + WL.U.hash([sourceHook, hit.type, hit.name, localClausePreview, index].join('|')),
        parserHitVersion: hit.parserHitVersion || '',
        family: hit.family || 'unknown',
        type: hit.type || (hit.family || 'unknown') + '.hint',
        name: WL.U.norm(hit.name || ''),
        sourceText,
        localClausePreview,
        flags,
        sourceHook,
        sourceInfo: Object.assign({}, hit.sourceInfo || {}),
        confidence: WL.U.clamp(hit.confidence == null ? 0.25 : hit.confidence, 0, 1),
        completedAction: !!hit.completedAction,
        attemptOnly: !!hit.attemptOnly,
        truthStatus: hit.truthStatus || 'unknown_candidate',
        parserOnly: hit.parserOnly !== false,
        canCommitDirectly: false,
        raw: hit
      };
    },
    normalizeType(hit, target) {
      let t = hit.type || 'unknown.hint';
      if (hit.sourceHook === 'input' && hit.completedAction !== true) {
        if (hit.family === 'inventory' && /^item\./.test(t)) t = 'item.intent';
        else if (hit.family === 'place' && /^place\./.test(t)) t = 'place.intent';
        else if (hit.family === 'quest' && /^quest\./.test(t)) t = 'quest.intent';
        else if (hit.family === 'actor' && /^actor\./.test(t)) t = 'actor.intent';
        else if (hit.family === 'progression') t = 'progression.intent';
        else if (hit.family === 'ability' && t !== 'ability.evidence') t = 'ability.intent';
      }
      if (!target || !target.display) {
        if (hit.family === 'body' && /^body\./.test(t)) return t;
        if (hit.family === 'time' && /^time\./.test(t)) return t;
      }
      return t;
    },
    bindTarget(hit) {
      const binding = WL.Resolver.buildTargetBinding(hit);
      binding.canonicalName = binding.display || '';
      binding.sourceName = hit.name || binding.raw || '';
      binding.kind = hit.family || binding.family || 'unknown';
      binding.safeForCommit = !!binding.display || hit.family === 'body' || hit.family === 'time';
      binding.bindingVersion = WL.PROPOSAL_BUILDER_LOCK_VERSION;
      return binding;
    },
    ownerFor(type, family) {
      if (/^item\.|^currency\./.test(type) || family === 'inventory') return 'InventoryReducerLite';
      if (/^ability\./.test(type) || family === 'ability') return 'AbilityReducerLite';
      if (/^body\./.test(type) || family === 'body') return 'BodyReducer';
      if (/^quest\.|^reward\./.test(type) || family === 'quest' || family === 'reward') return 'QuestReducerLite';
      if (/^actor\./.test(type) || family === 'actor') return 'ActorReducerLite';
      if (/^place\./.test(type) || family === 'place') return 'PlaceReducerLite';
      if (/^time\./.test(type) || family === 'time') return 'TimeReducerLite';
      if (/^progression\./.test(type) || family === 'progression') return 'ProgressionReducerLiteFuture';
      return 'NoActiveOwner';
    },
    buildEvidence(hit, target, sourceHash) {
      return {
        parserHitId: hit.parserHitId,
        parserHitVersion: hit.parserHitVersion,
        sourceHook: hit.sourceHook,
        sourceHash,
        localClausePreview: hit.localClausePreview,
        confidence: hit.confidence,
        completedAction: hit.completedAction,
        attemptOnly: hit.attemptOnly,
        truthStatus: hit.truthStatus,
        targetDisplay: target && target.display || '',
        targetId: target && target.id || ''
      };
    },
    riskFlags(hit, binding) {
      const f = hit.flags || {};
      const flags = [];
      const add = v => { if (flags.indexOf(v) < 0) flags.push(v); };
      if (!binding || !binding.display) add('ambiguous_target');
      if (f.quotedSpeech || f.quote) add('quote_present');
      if (f.dreamOrVision || f.dream) add('dream');
      if (f.rumorOrClaim || f.rumor) add('rumor');
      if (f.pretendOrFalse || f.pretend) add('pretend');
      if (f.failedOrRevoked || f.negatedOrRefused || f.denied) add('denied');
      if (f.visibleOnly) add('visible_only');
      if (f.hypothetical) add('hypothetical');
      if (f.futureIntent) add('future_intent');
      if (f.abstractOrIdea) add('abstract_or_idea');
      if (f.nonliteral) add('nonliteral');
      if (f.commandText) add('command_text');
      if (f.diagnosticText) add('diagnostic_text');
      if (hit.sourceHook === 'input') add('unconfirmed_input');
      if (hit.attemptOnly) add('attempt_only');
      if (hit.completedAction !== true) add('not_confirmed_completed');
      return flags;
    },
    validationReadiness(hit, riskFlags) {
      if (riskFlags.indexOf('command_text') >= 0 || riskFlags.indexOf('diagnostic_text') >= 0) return 'reject_source';
      if (riskFlags.indexOf('dream') >= 0 || riskFlags.indexOf('quote_present') >= 0 || riskFlags.indexOf('rumor') >= 0 || riskFlags.indexOf('pretend') >= 0) return 'unsafe_needs_reject_or_pending';
      if (riskFlags.indexOf('unconfirmed_input') >= 0 || riskFlags.indexOf('attempt_only') >= 0 || riskFlags.indexOf('not_confirmed_completed') >= 0) return 'intent_or_pending_only';
      if (hit.completedAction && hit.confidence >= 0.6) return 'validator_ready';
      return 'low_confidence_pending';
    },
    makeOperationKey(root, hit, type, target, sourceHash) {
      return WL.OperationLedger.makeKey(['proposal', hit.sourceHook || 'output', WL.U.nowTurn(root), sourceHash || '', type, target && (target.id || target.display || target.raw) || 'none', WL.U.hash(hit.localClausePreview || '')]);
    },
    signature(p) {
      return WL.OperationLedger.makeKey([p.type, p.target && (p.target.id || p.target.display || p.target.raw), p.sourceKind, p.evidence && p.evidence.sourceHash, WL.U.hash(p.localClausePreview || '')]);
    },
    dataPayload(hit) {
      const raw = Object.assign({}, hit.raw || {});
      delete raw.flags;
      delete raw.sourceInfo;
      return Object.assign({}, raw, { parserHitId: hit.parserHitId, truthStatus: hit.truthStatus, completedAction: hit.completedAction, attemptOnly: hit.attemptOnly, confidence: hit.confidence });
    },
    audit(root, summary) {
      if (!root) return;
      const row = Object.assign({ subsystem: 'ProposalBuilder', lockVersion: WL.PROPOSAL_BUILDER_LOCK_VERSION, turn: WL.U.nowTurn(root) }, summary || {});
      if (!Array.isArray(root.audits.proposalAudit)) root.audits.proposalAudit = [];
      root.audits.proposalAudit.push(row);
      WL.U.capArray(root.audits.proposalAudit, WL.SCHEMA_CAPS.proposals);
      WL.AuditLogger.append(root, 'proposals', { subsystem: 'ProposalBuilder', message: 'built proposals=' + (row.proposalCount || 0), detail: row }, 40);
    },
    inspect(root) {
      const rows = [];
      rows.push(this.lockVersion === WL.PROPOSAL_BUILDER_LOCK_VERSION ? { status: 'PASS', label: 'ProposalBuilder lock present: ' + WL.PROPOSAL_BUILDER_LOCK_VERSION } : { status: 'FAIL', label: 'ProposalBuilder lock mismatch' });
      rows.push(typeof this.fromHit === 'function' && typeof this.bindTarget === 'function' && typeof this.makeOperationKey === 'function' ? { status: 'PASS', label: 'required proposal builder functions present' } : { status: 'FAIL', label: 'required proposal builder functions missing' });
      rows.push(root.features && root.features.proposalBuilderLockEnabled === true ? { status: 'PASS', label: 'proposal builder lock feature enabled' } : { status: 'FAIL', label: 'proposal builder lock feature disabled' });
      rows.push(this.maxProposals(root) <= 40 ? { status: 'PASS', label: 'proposal cap configured' } : { status: 'FAIL', label: 'proposal cap unsafe' });
      const testRoot = WL.StateManager.init({});
      const hit = WL.EventScannerLite.scanOutput('You pick up the iron key.', testRoot)[0];
      const p = this.fromHits(testRoot, [hit], 'output')[0];
      rows.push(p && p.proposalOnly && p.canCommitDirectly === false ? { status: 'PASS', label: 'EventProposal schema marks proposal-only/no direct commit' } : { status: 'FAIL', label: 'EventProposal schema unsafe' });
      rows.push(p && p.opKey && p.sourceUseKey && p.dedupeKey ? { status: 'PASS', label: 'operation/source/dedupe keys present' } : { status: 'FAIL', label: 'proposal keys missing' });
      return rows;
    },

  // ============================================================
  // 12. CanonValidator Shell
  // ============================================================
  WL.CanonValidator = {
    validateAll(root, proposals) {
      const out = [];
      for (let i = 0; i < proposals.length; i++) out.push(this.validate(root, proposals[i]));
      root.pending.lastValidations = out;
      WL.AuditLogger.append(root, 'validations', { message: 'validated=' + out.length, results: out.map(v => v.result).join(',') }, 30);
      return out;
    },
    validate(root, p) {
      const f = p.flags || {};
      const denied = p.riskFlags.indexOf('denied') >= 0;
      const fake = p.riskFlags.indexOf('dream') >= 0 || p.riskFlags.indexOf('pretend') >= 0 || p.riskFlags.indexOf('quote_present') >= 0;
      const rumor = p.riskFlags.indexOf('rumor') >= 0;
      const noTarget = !p.target || !p.target.display;
      let result = 'pending', reason = 'shell default pending';

      if (p.sourceKind === 'command' || WL.EventScannerLite.unsafeSource(p.sourceKind, p.sourceText)) return this.row(p, 'reject', 'unsafe/non-canon source');
      if (noTarget && !/^body\.|^time\./.test(p.type)) return this.row(p, 'reject', 'ambiguous or unsafe target');

      switch (p.type) {
        case 'item.acquire':
          if (fake || rumor || denied || /\b(?:idea of|concept of|not take|decide not|snatches? it away)\b/i.test(p.sourceText)) result = 'reject', reason = 'unsafe item acquisition source';
          else result = 'accept', reason = 'confirmed pickup/handoff';
          break;
        case 'item.visible':
          result = 'pending'; reason = 'visible scene item only, not owned'; break;
        case 'currency.gain':
          if (fake || rumor || denied) result = 'reject', reason = 'unsafe currency source';
          else result = 'accept', reason = 'explicit payment';
          break;
        case 'ability.evidence':
          if (fake) result = 'hard_deny', reason = 'fake/pretend/quoted ability attempt';
          else if (denied || /\b(?:fails?|nothing happens|no visible effect|cannot|can't|lack technique)\b/i.test(p.sourceText)) result = 'soft_fail', reason = 'failed training evidence';
          else result = 'accept', reason = 'ability training evidence';
          break;
        case 'ability.stabilize_claim':
          result = 'pending'; reason = 'output-only ability claim cannot stabilize in Code Pass 1'; break;
        case 'body.fatigue':
          if (fake || rumor) result = 'reject', reason = 'non-canon fatigue source';
          else result = 'accept', reason = 'confirmed fatigue/exertion';
          break;
        case 'body.injury':
          if (fake || rumor) result = 'reject', reason = 'injury blocked by quote/dream/rumor';
          else result = 'accept', reason = 'confirmed injury language';
          break;
        case 'body.death_claim':
          if (fake || rumor) result = 'reject', reason = 'death claim from rumor/dream/quote';
          else result = 'pending', reason = 'death requires stronger future combat/body gate';
          break;
        case 'quest.offer':
          if (fake || rumor) result = 'reject', reason = 'unsafe quest source';
          else result = 'accept', reason = 'quest offer detected';
          break;
        case 'quest.accept':
          if (fake || rumor || denied) result = 'reject', reason = 'unsafe quest acceptance';
          else result = 'accept', reason = 'quest acceptance detected';
          break;
        case 'quest.complete':
          if (fake || rumor || denied) result = 'reject', reason = 'unsafe quest completion';
          else result = 'accept', reason = 'quest completion detected';
          break;
        case 'reward.promise':
          if (fake || rumor) result = 'reject', reason = 'unsafe reward promise';
          else result = 'accept', reason = 'promise tracked, not paid';
          break;
        case 'actor.present':
          if (fake || rumor) result = 'reject', reason = 'unsafe actor presence';
          else result = 'accept', reason = 'actor directly present/acting';
          break;
        case 'place.current':
          if (fake || rumor || denied || p.data.weakReference) result = 'reject', reason = 'weak/unsafe place source';
          else result = 'accept', reason = 'strong arrival/current scene';
          break;
        case 'place.reference':
          result = 'pending'; reason = 'weak place reference only'; break;
        case 'time.pass':
          if (fake || rumor || /\byears?\b/i.test(p.sourceText) || /\bpracticing narration\b/i.test(p.sourceText)) result = 'reject', reason = 'unsafe/nonliteral time passage';
          else result = 'accept', reason = 'literal short time advance';
          break;
        default:
          result = 'pending'; reason = 'no active Code Pass 1 rule';
      }
      return this.row(p, result, reason);
    },
    row(p, result, reason) {
      return { proposalId: p.id, type: p.type, family: p.family, result, reason, target: p.target, opKey: p.opKey, sourceText: p.sourceText, data: p.data };
    }
  };

  // 13. ReducerRegistry and Limited Reducers
  // ============================================================
  WL.ReducerRegistry = {
    applyAcceptedBatch(root, validations) {
      const results = [];
      for (let i = 0; i < validations.length; i++) {
        const v = validations[i];
        if (['accept','soft_fail','hard_deny'].indexOf(v.result) < 0) { results.push({ type: v.type, status: 'skipped', reason: v.result }); continue; }
        const led = WL.OperationLedger.tryAdd(root, 'recentKeys', v.opKey, { type: v.type, result: v.result });
        if (!led.applied) { results.push({ type: v.type, status: 'duplicate_skip', reason: v.reason }); continue; }
        let r;
        if (v.family === 'inventory') r = WL.InventoryReducer.apply(root, v);
        else if (v.family === 'ability') r = WL.AbilityReducer.apply(root, v);
        else if (v.family === 'body') r = WL.BodyReducer.apply(root, v);
        else if (v.family === 'quest') r = WL.QuestReducerLite.apply(root, v);
        else if (v.family === 'actor') r = WL.ActorReducerLite.apply(root, v);
        else if (v.family === 'place') r = WL.PlaceReducerLite.apply(root, v);
        else if (v.family === 'time') r = WL.TimeReducerLite.apply(root, v);
        else r = { type: v.type, status: 'no_owner', reason: 'No active reducer owner.' };
        results.push(r);
      }
      root.pending.lastReducerResults = results;
      WL.AuditLogger.append(root, 'reducers', { message: 'reducer results=' + results.length, statuses: results.map(r => r.status).join(',') }, 30);
      return results;
    }
  };

  WL.InventoryReducer = {
    apply(root, v) {
      if (v.type === 'item.acquire') return this.acquire(root, v);
      if (v.type === 'item.visible') return this.visible(root, v);
      if (v.type === 'currency.gain') return this.currency(root, v);
      return { type: v.type, status: 'ignored', reason: 'Inventory reducer has no handler.' };
    },
    itemId(name) { return 'item_' + WL.U.slug(name); },
    acquire(root, v) {
      const name = v.target.display;
      const id = this.itemId(name);
      if (!root.inventory.itemsById[id]) root.inventory.itemsById[id] = { id, name, qty: 1, holder: 'player', owner: 'player', availability: 'available', createdTurn: WL.U.nowTurn(root) };
      else root.inventory.itemsById[id].qty = WL.U.clamp((root.inventory.itemsById[id].qty || 1) + 1, 1, 9999);
      if (root.inventory.ownedIds.indexOf(id) < 0) root.inventory.ownedIds.push(id);
      WL.DirtyQueue.add(root, 'inventory', 'item acquired: ' + name);
      return { type: v.type, status: 'committed', target: name };
    },
    visible(root, v) {
      const name = v.target.display;
      WL.U.uniquePush(root.inventory.visibleSceneItems, name, 20);
      WL.DirtyQueue.add(root, 'inventory', 'visible item: ' + name);
      return { type: v.type, status: 'pending_visible_only', target: name };
    },
    currency(root, v) {
      const reward = v.data && v.data.reward || WL.Sanitizers.reward(v.target.raw || v.target.display || '');
      if (!reward) return { type: v.type, status: 'rejected', reason: 'No currency parsed.' };
      const payKey = WL.OperationLedger.makeKey(['reward', v.sourceText, reward.amount, reward.coin]);
      const led = WL.OperationLedger.tryAdd(root, 'rewardKeys', payKey, { reward });
      if (!led.applied) return { type: v.type, status: 'duplicate_skip', reason: 'Reward payment already applied.' };
      root.inventory.currency[reward.coin] = WL.U.clamp((root.inventory.currency[reward.coin] || 0) + reward.amount, 0, 999999999);
      root.quests.rewardPayments.push({ turn: WL.U.nowTurn(root), amount: reward.amount, coin: reward.coin, source: v.sourceText });
      WL.U.capArray(root.quests.rewardPayments, 40);
      WL.DirtyQueue.add(root, 'inventory', 'currency gain');
      return { type: v.type, status: 'committed', target: reward.amount + ' ' + reward.coin };
    }
  };

  WL.AbilityReducer = {
    apply(root, v) {
      if (v.type === 'ability.evidence') return this.evidence(root, v);
      if (v.type === 'ability.stabilize_claim') return { type: v.type, status: 'pending', reason: v.reason };
      return { type: v.type, status: 'ignored' };
    },
    evidence(root, v) {
      const name = v.target.display;
      const id = 'ability_' + WL.U.slug(name);
      if (v.result === 'hard_deny') {
        root.abilities.denied.push({ turn: WL.U.nowTurn(root), name, reason: v.reason });
        WL.U.capArray(root.abilities.denied, 30);
        return { type: v.type, status: 'hard_denied', target: name };
      }
      let rec = root.abilities.candidatesById[id];
      if (!rec) {
        rec = { id, name, stage: 'candidate', attempts: 0, failures: 0, successes: 0, evidence: [], createdTurn: WL.U.nowTurn(root), lastTurn: WL.U.nowTurn(root) };
        root.abilities.candidatesById[id] = rec;
        if (root.abilities.candidateIds.indexOf(id) < 0) root.abilities.candidateIds.push(id);
      }
      rec.attempts += 1;
      rec.lastTurn = WL.U.nowTurn(root);
      if (v.result === 'soft_fail') rec.failures += 1;
      else rec.successes += 1;
      rec.evidence.push({ turn: WL.U.nowTurn(root), result: v.result, source: v.sourceText.slice(0, 120) });
      WL.U.capArray(rec.evidence, 12);
      WL.DirtyQueue.add(root, 'abilities', 'ability evidence: ' + name);
      return { type: v.type, status: v.result === 'soft_fail' ? 'soft_fail_committed' : 'committed', target: name };
    }
  };

  WL.BodyReducer = {
    apply(root, v) {
      if (v.type === 'body.fatigue') {
        root.body.ep = WL.U.clamp(root.body.ep - 5, 0, root.body.maxEp);
        WL.U.uniquePush(root.body.conditions, 'fatigued', 20);
        WL.DirtyQueue.add(root, 'body', 'fatigue');

  WL.TimeReducerLite = {
    textNum(value) {
      const m = { one:1, two:2, three:3, four:4, five:5, six:6, seven:7, eight:8, nine:9, ten:10, twenty:20 };
      const l = WL.U.lower(value);
      return m[l] || Number(value) || 0;
    },
    apply(root, v) {
      const n = this.textNum(v.data && v.data.amountText);
      const unit = String(v.data && v.data.unit || 'minutes');
      if (!n) return { type: v.type, status: 'rejected', reason: 'No amount.' };
      let minutes = n;
      if (/hour/.test(unit)) minutes = n * 60;
      if (/day/.test(unit)) minutes = n * 1440;
      root.time.minute += minutes;
      while (root.time.minute >= 60) { root.time.minute -= 60; root.time.hour += 1; }
      while (root.time.hour >= 24) { root.time.hour -= 24; root.time.day += 1; }
      root.time.notes.push({ turn: WL.U.nowTurn(root), text: v.sourceText });
      WL.U.capArray(root.time.notes, 20);
      WL.DirtyQueue.add(root, 'time', 'time pass');
      return { type: v.type, status: 'committed', target: n + ' ' + unit };
    }

      out.push(this.warn('Code Pass 1 is foundation-limited; advanced systems are intentionally deferred.'));
      root.audits.releasecheck = out.map(x => ({ turn: WL.U.nowTurn(root), text: x }));
      return out;
    },
    stateAudit(root) {
      const out = [];
      out.push(root && typeof root === 'object' ? this.pass('root exists') : this.fail('root missing'));
      const buckets = WL.ROOT_BUCKETS_ALL || ['meta','features','layout','runtime','startup','player','body','progression','inventory','abilities','quests','actors','places','time','cards','operationLedger','audits','migration','quarantine','dirty','cleanup','pending'];
      for (let i = 0; i < buckets.length; i++) out.push(root[buckets[i]] ? this.pass('bucket ' + buckets[i]) : this.fail('missing bucket ' + buckets[i]));
      if (WL.StateSchemaLock) WL.StateSchemaLock.inspect(root).forEach(r => out.push(r.status + ': ' + r.label));
      if (WL.CommandOutputFormatLock) WL.CommandOutputFormatLock.inspect(root).forEach(r => out.push(r.status + ': ' + r.label));
      const b = root.body;
      out.push(b.hp >= 0 && b.hp <= b.maxHp ? this.pass('HP invariant') : this.fail('HP invariant broken'));
      out.push(b.mp >= 0 && b.mp <= b.maxMp ? this.pass('MP invariant') : this.fail('MP invariant broken'));
      out.push(b.ep >= 0 && b.ep <= b.maxEp ? this.pass('EP invariant') : this.fail('EP invariant broken'));
      return out;
    }
  };

  // ============================================================
  // 17. Hook Orchestration
  // ============================================================
  WL.processOutput = function (root, output) {
    const safe = WL.Runtime.safeText(output);
    const outputHash = WL.Runtime.hashText(safe);
    if (WL.Runtime.shouldSuppressEventScan(root, safe)) {
      WL.AuditLogger.summarizeLastTurn(root, 'Output suppressed by Runtime; command/audit text was not scanned as story.');
      WL.Runtime.clearCommandSuppression(root);
      return safe;
    }
    if (WL.Runtime.isDuplicateOutput(root, outputHash)) {
      root.runtime.duplicateOutputSkipped += 1;
      root.runtime.outputReplayDetected = true;
      root.runtime.lastDuplicateOutputHash = outputHash;
      WL.Runtime.recordWarning(root, 'Duplicate output replay detected; consequence processing skipped.', { hash: outputHash });
      if (WL.AuditLogger) WL.AuditLogger.duplicate(root, outputHash, 'duplicate output replay skipped before reducer');
      WL.AuditLogger.summarizeLastTurn(root, 'Duplicate output hash skipped; no consequences applied.');
      return safe;
    }
    const hits = WL.EventScannerLite.scanOutput(root, safe);
    const proposals = WL.ProposalBuilder.fromHits(root, hits, 'output');
    const validations = WL.CanonValidator.validateAll(root, proposals);
    const results = WL.ReducerRegistry.applyAcceptedBatch(root, validations);
    WL.CardSyncLite.dryRun(root);
    WL.CleanupManagerLite.run(root);
    WL.Runtime.markOutputApplied(root, outputHash);
    WL.AuditLogger.summarizeLastTurn(root, 'Output processed: hits=' + hits.length + ', proposals=' + proposals.length + ', results=' + results.length + '.');
    return safe;
  };

  WL.onInput = function (text) {
    let root;
    try {
      root = WL.StateManager.root();
      const begin = WL.Runtime.beginHook(root, 'input', text);
      WL.MigrationManager.run(root);
      const raw = begin.text;
      if (WL.U.startsCommand(raw) && root.features.commandsEnabled) {
        const out = WL.CommandRouter.route(root, raw);
        return WL.Runtime.endHook(root, 'input', out, 'completed');
      }
      root.runtime.lastInputText = raw;
      WL.Runtime.markInput(root, raw);
      root.startup.firstGameplayInputSeen = true;
      root.startup.startupLocked = true;
      root.startup.gameplayStarted = true;
      const actionCount = WL.U.getInfoActionCount();
      if (typeof actionCount !== 'number') {
        root.meta.currentTurn = (root.meta.currentTurn || 0) + 1;
        root.runtime.currentTurn = root.meta.currentTurn;
      }
      root.runtime.pendingInputIntentId = 'intent_' + WL.U.hash(raw + '|' + root.runtime.currentTurn);
      if (root.features.parserShellEnabled && WL.EventScannerLite) WL.EventScannerLite.scanInput(raw, root);
      return WL.Runtime.endHook(root, 'input', raw, 'completed');
    } catch (err) {
      return WL.Runtime.failSoft(root || { runtime: { errors: [] }, audits: { warnings: [] } }, 'input', 'onInput', err, text, false);
    }
  };

  WL.onContext = function (text) {
    let root;
    try {
      root = WL.StateManager.root();
      const begin = WL.Runtime.beginHook(root, 'context', text);
      const out = root.features.contextBuilderLiteEnabled ? WL.ContextBuilderLite.build(root, begin.text) : begin.text;
      return WL.Runtime.endHook(root, 'context', out, 'completed');
    } catch (err) {
      return WL.Runtime.failSoft(root || { runtime: { errors: [] }, audits: { warnings: [] } }, 'context', 'onContext', err, text, false);
    }
  };

  WL.onOutput = function (text) {
    let root;
    try {
      root = WL.StateManager.root();
      const begin = WL.Runtime.beginHook(root, 'output', text);
      const raw = begin.text;
      const out = WL.processOutput(root, raw);
      return WL.Runtime.endHook(root, 'output', out, 'completed');
    } catch (err) {
      return WL.Runtime.failSoft(root || { runtime: { errors: [] }, audits: { warnings: [] } }, 'output', 'onOutput', err, text, false);
    }
  };
