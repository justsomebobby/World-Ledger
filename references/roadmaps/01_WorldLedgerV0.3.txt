/*
WorldLedgerV0.3 — Core MVP Foundation Script / Code Pass 1 State Schema Lock
Made by: JustSomeBobby

WorldLedger is a freeform RPG truth engine for AI Dungeon. It is built to preserve
canon, consequences, state, and memory without turning AI Dungeon into a rigid command game.

Credit and direct inspiration:
- RPGUI by @RonenNess — state-first RPG/HUD presentation discipline and clean helper separation, reinterpreted for AI Dungeon as text-first state and command readouts. No HTML UI code is used.
- TAS / True Auto Stats by @Yi1i1i — freeform action reading, emergent growth ambition, and structured stats / inventory / abilities, rebuilt as proposal -> validation -> reducer ownership.
- Auto-Cards by @LewdLeah — object permanence, entity detection, Story Card drafting, and memory maintenance ideas, rebuilt so cards mirror canon instead of creating it.
- Inner Self by @LewdLeah — segmented long-term NPC memory and major-character persistence, rebuilt as selective reducer-owned memory rather than universal NPC simulation.

Not direct foundations for this build: LoLa, Hashtag-DnD, AID-dice-rolling, and Gnurro AIDscripts. They may remain general research references, but they are not credited here as active functional inspirations unless a future build directly borrows from them.

Design doctrine:
Input may intend. Context may remind. Output may propose. ParserHits are observations.
EventProposals are structured claims. CanonValidator decides eligibility. Reducers commit.
OperationLedger dedupes. DirtyQueue marks refresh. Context reads. Cards mirror. ReleaseCheck tells the truth.

Code Pass 1 scope:
This is the foundation build with the Build Chunk 3 state schema lock applied. It initializes, inspects, audits, validates, and commits only limited MVP-safe events while enforcing canonical state buckets, owner boundaries, schema invariants, and safe repair rules.
Deferred systems remain disabled: generated Story Cards, full economy, crafting, investigation, reputation/faction
automation, domain/labor, absorption/evolution, advanced combat, warfare, multi-protagonist, and manual mutation commands.
*/

    norm(value) {
      return String(value == null ? '' : value)
        .replace(/[\u2018\u2019]/g, "'")
        .replace(/[\u201C\u201D]/g, '"')
        .replace(/\s+/g, ' ')

    lower(value) { return this.norm(value).toLowerCase(); },
    title(value) {
      const raw = this.norm(value).replace(/^the\s+/i, '');
      if (!raw) return '';
      const small = { a: 1, an: 1, and: 1, at: 1, by: 1, for: 1, from: 1, in: 1, of: 1, on: 1, or: 1, the: 1, to: 1, with: 1 };
      return raw.split(/\s+/).map((w, i) => {
        const clean = w.toLowerCase();
        if (i > 0 && small[clean]) return clean;
        return clean.charAt(0).toUpperCase() + clean.slice(1);
      }).join(' ');
    },
    slug(value) {
      return this.lower(value).replace(/[^a-z0-9]+/g, '_').replace(/^_+|_+$/g, '').slice(0, 80) || 'unknown';
    },
    clamp(n, min, max) {
      n = Number(n);
      if (!Number.isFinite(n)) n = min;
      return Math.max(min, Math.min(max, n));
    },
    hash(value) {
      const s = String(value == null ? '' : value);
      let h = 2166136261;
      for (let i = 0; i < s.length; i++) {
        h ^= s.charCodeAt(i);
        h += (h << 1) + (h << 4) + (h << 7) + (h << 8) + (h << 24);
      }
      return ('00000000' + (h >>> 0).toString(16)).slice(-8);
    },
    capArray(arr, max) {
      if (!Array.isArray(arr)) return [];
      while (arr.length > max) arr.shift();
      return arr;
    },
    uniquePush(arr, value, max) {
      if (!Array.isArray(arr)) return false;
      const v = this.norm(value);
      if (!v) return false;
      const k = this.slug(v);
      for (let i = 0; i < arr.length; i++) if (this.slug(arr[i]) === k) return false;
      arr.push(v);
      this.capArray(arr, max || 20);
      return true;
    },
    ensureObj(parent, key, factory) {
      if (!parent || typeof parent !== 'object') return factory ? factory() : {};
      if (!parent[key] || typeof parent[key] !== 'object' || Array.isArray(parent[key])) parent[key] = factory ? factory() : {};
      return parent[key];
    },
    ensureArray(parent, key) {
      if (!parent || typeof parent !== 'object') return [];
      if (!Array.isArray(parent[key])) parent[key] = [];
      return parent[key];
    },
    safeJson(value) {
      try { return JSON.stringify(value); } catch (e) { return '[unserializable]'; }
    },
    unquoted(text) {
      return String(text || '').replace(/["“][\s\S]*?["”]/g, ' ').replace(/'[^']{1,240}'/g, ' ');
    },
    sentenceSplit(text) {
      const t = this.norm(text);
      if (!t) return [];
      return t.split(/(?<=[.!?])\s+|\n+/).map(s => s.trim()).filter(Boolean).slice(0, 16);
    },
    startsCommand(text) {
      return /^\s*\/\w+/.test(String(text || ''));
    },
    commandName(text) {
      const m = String(text || '').match(/^\s*\/([a-zA-Z0-9_-]+)/);
      return m ? m[1].toLowerCase() : '';
    },
    commandArgs(text) {
      return String(text || '').replace(/^\s*\/\S+\s*/, '').trim();
    },

    },

      return lines;
    }
  };

  WL.OperationLedger = {
    normalizeKey(value) { return WL.U.slug(value).slice(0, 140); },
    makeKey(parts) { return (parts || []).map(p => this.normalizeKey(p)).filter(Boolean).join('|').slice(0, 240); },
    bucket(root, name) {
      const led = root.operationLedger;
      if (name === 'recentKeys') return led.recentKeys;
      if (!led[name] || typeof led[name] !== 'object') led[name] = {};
      return led[name];
    },
    has(root, bucket, key) {
      key = this.normalizeKey(key);
      if (!key) return false;
      if (bucket === 'recentKeys') return root.operationLedger.recentKeys.indexOf(key) >= 0;
      return !!this.bucket(root, bucket)[key];
    },
    add(root, bucket, key, data) {
      key = this.normalizeKey(key);
      if (!key) return false;
      if (bucket === 'recentKeys') {
        if (root.operationLedger.recentKeys.indexOf(key) >= 0) return false;
        root.operationLedger.recentKeys.push(key);
        WL.U.capArray(root.operationLedger.recentKeys, WL.SCHEMA_CAPS.recentLedger);
        return true;
      }
      const b = this.bucket(root, bucket);
      if (b[key]) return false;
      b[key] = Object.assign({ turn: WL.U.nowTurn(root) }, data || {});
      return true;
    },
    tryAdd(root, bucket, key, data) {
      const existed = this.has(root, bucket, key);
      if (existed) {
        root.operationLedger.duplicateSkips.push({ turn: WL.U.nowTurn(root), bucket, key: this.normalizeKey(key) });
        WL.U.capArray(root.operationLedger.duplicateSkips, 30);
        return { applied: false, duplicate: true, key: this.normalizeKey(key) };
      }
      this.add(root, bucket, key, data);
      return { applied: true, duplicate: false, key: this.normalizeKey(key) };
    },
    summary(root) {
      const led = root.operationLedger;
      return [
        'recent=' + led.recentKeys.length,
        'permanent=' + Object.keys(led.permanentKeys).length,
        'rewards=' + Object.keys(led.rewardKeys).length,
        'cards=' + Object.keys(led.cardKeys).length,
        'sources=' + Object.keys(led.sourceUseKeys).length,
        'duplicateSkips=' + led.duplicateSkips.length
      ].join(', ');
    }
  };

  // ============================================================
  // 08. Resolver and Owner-Specific Sanitizers
  // ============================================================
  WL.Sanitizers = {
    stripUnsafeTail(value) {
      return WL.U.norm(value)
        .replace(/["“”]+/g, '')
        .replace(/\s+(?:to|by|while|when|as|because|before|after|until|with|without|near|from|inside|outside|towards?|against|for)\b[\s\S]*$/i, '')
        .replace(/[.,;:!?]+$/g, '')
        .trim();
    },
    item(value) {
      let v = WL.U.norm(value).replace(/^(?:the|a|an|some|your|my|his|her|their)\s+/i, '');
      v = v.replace(/\s+(?:from|near|on|in|inside|at|beside|under|over|within|by|with|using|for|toward|towards|against)\b[\s\S]*$/i, '');
      v = v.replace(/[.,;:!?]+$/g, '').trim();
      const l = v.toLowerCase();
      if (!l || l.length > 70) return '';
      if (/^(?:idea|concept|memory|rumor|dream|nothing|anything|something|it|them|this|that)$/i.test(l)) return '';
      if (/\b(?:idea of|concept of|thought of|rumor of|dream of)\b/i.test(l)) return '';
      return WL.U.title(v);
    },
    ability(value) {
      const raw = WL.U.norm(value);
      const m = raw.match(/\b([A-Z][A-Za-z'’-]{1,24}(?:\s+[A-Z][A-Za-z'’-]{1,24}){0,3}\s+(?:Ward|Slash|Strike|Bolt|Sense|Sight|Step|Stance|Form|Art|Barrier|Shield|Control|Pulse|Veil|Blade|Guard|Guarding|Flame|Spark|Flow|Grip))\b/);
      let v = m ? m[1] : raw;
      v = this.stripUnsafeTail(v).replace(/\bGuarding$/i, 'Guard');
      if (!/[A-Z]/.test(v)) return '';
      if (/\b(?:travel|smithy|door|key|inspect|open|road|street|inventory|quest|gold|copper)\b/i.test(v)) return '';
      if (v.split(/\s+/).length > 5) return '';
      return WL.U.title(v);
    },
    actor(value) {
      let v = WL.U.norm(value).replace(/^(?:the|a|an)\s+/i, '');
      v = v.replace(/[.,;:!?]+$/g, '').trim();
      if (!/^[A-Z][A-Za-z'’-]{1,30}(?:\s+[A-Z][A-Za-z'’-]{1,30}){0,2}$/.test(v)) return '';
      if (/^(?:The|You|Your|He|She|They|It|Guard|Veteran Guard|Innkeeper|Merchant)$/i.test(v)) return '';
      return WL.U.title(v);
    },
    place(value) {
      let v = WL.U.norm(value);
      v = v.replace(/\s+(?:while|as|where|when|with|without|near|beside|under|over)\b[\s\S]*$/i, '').trim();
      v = v.replace(/[.,;:!?]+$/g, '').trim();
      if (!v || v.length > 90) return '';
      if (/^(?:road|street|door|room|common room|wall|floor|body|hand|market lane|the road)$/i.test(v)) return '';
      const m = v.match(/\b(?:enter|arrive at|reach|inside)\s+(?:the\s+)?([A-Z][A-Za-z'’ -]{2,70})/i);
      if (m) v = m[1];
      return WL.U.title(v.replace(/^the\s+/i, ''));
    },
    quest(value) {
      let v = WL.U.norm(value).replace(/^to\s+/i, '');
      v = v.replace(/["“”]+/g, '').replace(/[.;]+$/g, '').trim();
      if (!v || v.length < 3 || v.length > 120) return '';
      return v.charAt(0).toUpperCase() + v.slice(1);
    },
    reward(value) {

        id: clean ? this.canonicalId(h.family || 'target', clean) : '',
        ambiguous: !clean,
        family: h.family || 'unknown'
      };
    }

  WL.EventScannerLite = {
    unsafeSource(kind, text) {
      const t = WL.U.norm(text);
      if (kind === 'command' || WL.U.startsCommand(t)) return true;
      if (/^\[WorldLedger\]/i.test(t) || /^\[AIDRPG\]/i.test(t)) return true;
      if (/^(?:WorldLedger|AIDRPG)\s+(?:Inventory|Abilities|Quests|Audit|Settings)/i.test(t)) return true;
      return false;
    },
    flags(text) {
      const raw = WL.U.norm(text);
      const unq = WL.U.unquoted(raw);
      const l = raw.toLowerCase();
      return {
        raw, unquoted: unq,
        quote: raw !== unq,
        dream: /\b(?:dream|dreamed|dreamt|in a dream|vision|nightmare)\b/i.test(raw),
        rumor: /\b(?:rumor|rumour|people say|they say|it is said|heard that|supposedly)\b/i.test(raw),
        pretend: /\b(?:pretend|joke|imaginary|not real|mimic|only mimicking|make believe)\b/i.test(raw),
        denied: /\b(?:do not|don't|decide not|refuse|refuses|failed|fails|cannot|can't|unable|snatches? it away|stops you|blocked|no ability is gained|no item is gained)\b/i.test(raw),
        visibleOnly: /\b(?:see|saw|visible|on the shelf|nearby|in the distance|could take|available|lying there)\b/i.test(raw) && !/\b(?:pick up|take|grab|collect|accept|hands you|gives you)\b/i.test(raw)
      };
    },
    scanOutput(root, output) {
      const hits = [];
      const input = root.runtime.lastInputText || '';
      const combined = WL.U.norm(input + '\n' + output);
      if (this.unsafeSource('output', output)) return [];
      const flags = this.flags(combined);
      const sentences = WL.U.sentenceSplit(output);
      this.scanInventory(hits, output, flags);
      this.scanAbility(hits, input, output, flags);
      this.scanBody(hits, output, flags);
      this.scanQuest(hits, input, output, flags);
      this.scanReward(hits, output, flags);
      this.scanActor(hits, output, flags);
      this.scanPlace(hits, input, output, flags);
      this.scanTime(hits, output, flags);
      root.pending.lastParserHits = hits.slice(0, 20);
      WL.AuditLogger.append(root, 'parser', { message: 'scanOutput hits=' + hits.length, families: hits.map(h => h.family).join(',') }, 30);
      return hits.slice(0, 20);
    },
    push(hits, family, type, name, text, flags, extra) {
      hits.push(Object.assign({ family, type, name: WL.U.norm(name), sourceText: WL.U.norm(text).slice(0, 240), flags: Object.assign({}, flags || {}) }, extra || {}));
    },
    scanInventory(hits, text, flags) {
      const raw = WL.U.norm(text);
      let m;
      const handoff = raw.match(/\b(?:hands you|gives you|passes you|places|puts)\s+(?:the\s+|a\s+|an\s+)?([a-z][a-z' -]{2,60}?)(?:\s+(?:in your hand|into your hand|to you))\b/i);
      if (handoff) this.push(hits, 'inventory', 'item.acquire', handoff[1], handoff[0], this.flags(handoff[0]), { mode: 'handoff' });
      const pickupRe = /\byou\s+(?:pick up|take|grab|retrieve|pocket|collect)\s+(?:the\s+|a\s+|an\s+|some\s+)?([a-z][a-z' -]{2,60}?)(?:\s+(?:from|near|on|in|and|with|before|while)\b|[.!?]|$)/gi;
      while ((m = pickupRe.exec(raw)) && hits.length < 20) this.push(hits, 'inventory', 'item.acquire', m[1], m[0], this.flags(m[0]), { mode: 'pickup' });
      const visibleRe = /\b(?:you see|you notice|there is|there are|lies|rests)\s+(?:the\s+|a\s+|an\s+|some\s+)?([a-z][a-z' -]{2,60}?)(?:\s+(?:on|near|beside|under|inside|at)\b|[.!?]|$)/gi;
      while ((m = visibleRe.exec(raw)) && hits.length < 20) this.push(hits, 'inventory', 'item.visible', m[1], m[0], this.flags(m[0]), { mode: 'visible' });
    },
    scanAbility(hits, input, output, flags) {
      const combined = WL.U.norm(input + ' ' + output);
      if (/\b(?:train|practice|attempt|try to|focus on|channel|use)\b/i.test(combined)) {
        const m = combined.match(/\b([A-Z][A-Za-z'’-]{1,24}(?:\s+[A-Z][A-Za-z'’-]{1,24}){0,3}\s+(?:Ward|Slash|Strike|Bolt|Sense|Sight|Step|Stance|Form|Art|Barrier|Shield|Control|Pulse|Veil|Blade|Guard))\b/);
        if (m) this.push(hits, 'ability', 'ability.evidence', m[1], combined, flags, { attemptedInInput: /\b(?:train|practice|attempt|try|focus|channel|use)\b/i.test(input) });
      }
      const learned = output.match(/\byou\s+(?:learn|learned|gain|gained|unlock|unlocked)\s+(?:the\s+)?([A-Z][A-Za-z'’-]{1,24}(?:\s+[A-Z][A-Za-z'’-]{1,24}){0,3}\s+(?:Ward|Slash|Strike|Bolt|Sense|Sight|Step|Stance|Form|Art|Barrier|Shield|Control|Pulse|Veil|Blade|Guard))\b/i);
      if (learned) this.push(hits, 'ability', 'ability.stabilize_claim', learned[1], learned[0], flags, { outputOnly: true });
    },
    scanBody(hits, text, flags) {
      const raw = WL.U.norm(text);
      if (/\b(?:exhausted|fatigued|tired|worn out|out of breath|muscles burn)\b/i.test(raw)) this.push(hits, 'body', 'body.fatigue', 'fatigue', raw, flags, { severity: 1 });
      if (/\b(?:wolf bites|bitten|stabbed|deep cut|bleeding|blood runs|burned|poisoned|arrow hits|sword cuts)\b/i.test(raw)) this.push(hits, 'body', 'body.injury', 'injury', raw, flags, { severity: 2 });
      if (/\b(?:you died|you die|you are dead|kills you|killed you)\b/i.test(raw)) this.push(hits, 'body', 'body.death_claim', 'death', raw, flags, { severity: 5 });
    },
    scanQuest(hits, input, output, flags) {
      const raw = WL.U.norm(input + ' ' + output);
      let m = raw.match(/\b(?:quest|task|job|mission)\s+(?:to\s+)?([^.!?;]{3,100})/i);
      if (m) this.push(hits, 'quest', 'quest.offer', m[1], m[0], flags);
      m = raw.match(/\b(?:accept|agree to|take on|undertake)\s+(?:the\s+)?(?:quest|task|job|mission)\s+(?:to\s+)?([^.!?;]{3,100})/i);
      if (m) this.push(hits, 'quest', 'quest.accept', m[1], m[0], flags);
      m = raw.match(/\b(?:complete|completed|finish|finished)\s+(?:the\s+)?(?:quest|task|job|mission)\s+(?:to\s+)?([^.!?;]{3,100})/i);
      if (m) this.push(hits, 'quest', 'quest.complete', m[1], m[0], flags);
    },
    scanReward(hits, text, flags) {
      const raw = WL.U.norm(text);
      let m = raw.match(/\b(?:promises?|vows?|offers?)\s+(?:you\s+)?(\d{1,6}\s*(?:copper|silver|gold))\b/i);
      if (m) this.push(hits, 'quest', 'reward.promise', m[1], m[0], flags, { reward: WL.Sanitizers.reward(m[1]) });
      m = raw.match(/\b(?:pays?|paid|hands you|gives you)\s+(?:(?:the\s+)?promised\s+)?(\d{1,6}\s*(?:copper|silver|gold))\b/i);
      if (m) this.push(hits, 'inventory', 'currency.gain', m[1], m[0], flags, { reward: WL.Sanitizers.reward(m[1]) });
    },
    scanActor(hits, text, flags) {
      const raw = WL.U.norm(text);
      const re = /\b([A-Z][A-Za-z'’-]{1,30}(?:\s+[A-Z][A-Za-z'’-]{1,30}){0,2})\s+(?:says|asks|replies|nods|hands|gives|joins|follows|attacks|teaches)\b/g;
      let m;
      while ((m = re.exec(raw)) && hits.length < 20) this.push(hits, 'actor', 'actor.present', m[1], m[0], flags, { companion: /\b(?:joins|follows)\b/i.test(m[0]) });
    },
    scanPlace(hits, input, output, flags) {
      const raw = WL.U.norm(input + ' ' + output);
      const weak = /\b(?:visible from|can see|could see|nearby|in the distance|mentions|talks about)\b/i.test(raw);
      const strong = /\b(?:enter|entered|step inside|walk into|move into|head into|arrive at|arrive in|reach|return to|go inside)\b/i.test(raw);
      let m = raw.match(/\b(?:enter|entered|step inside|walk into|move into|head into|arrive at|arrive in|reach|return to|go inside)\s+(?:the\s+)?([A-Z][A-Za-z'’ -]{2,80})/i);
      if (m) this.push(hits, 'place', 'place.current', m[1], m[0], flags, { strongArrival: true, weakReference: false });
      else if (weak) {
        m = raw.match(/\b([A-Z][A-Za-z'’ -]{2,80})\s+(?:is\s+)?visible\s+from\s+([A-Z][A-Za-z'’ -]{2,80})/i);
        if (m) this.push(hits, 'place', 'place.reference', m[1], m[0], flags, { weakReference: true });
      }
      if (strong && /\broad\b/i.test(raw) && !m) this.push(hits, 'place', 'place.route', 'road', raw, flags, { routeOnly: true });
    },
    scanTime(hits, text, flags) {
      const raw = WL.U.norm(text);
      const m = raw.match(/\b(\d{1,2}|one|two|three|four|five|six|seven|eight|nine|ten|twenty)\s+(minutes?|hours?|days?|weeks?|months?|years?)\s+pass\b/i);
      if (m) this.push(hits, 'time', 'time.pass', m[0], m[0], flags, { amountText: m[1], unit: m[2].toLowerCase() });
    }
  };

  WL.ProposalBuilder = {
    fromHits(root, hits, sourceKind) {
      const proposals = [];
      for (let i = 0; i < hits.length && proposals.length < 20; i++) {
        const h = hits[i];
        const binding = WL.Resolver.buildTargetBinding(h);
        const type = h.type;
        const key = WL.OperationLedger.makeKey([sourceKind || 'output', root.runtime.currentTurn, root.runtime.lastOutputHash, type, binding.id || binding.raw || h.name]);
        proposals.push({
          id: 'prop_' + proposals.length + '_' + WL.U.hash(key),
          type,
          family: h.family,
          sourceKind: sourceKind || 'output',
          sourceText: h.sourceText || '',
          flags: h.flags || {},
          target: binding,
          opKey: key,
          riskFlags: this.riskFlags(h, binding),
          data: Object.assign({}, h, { flags: undefined })
        });
      }
      root.pending.lastProposals = proposals;
      WL.AuditLogger.append(root, 'proposals', { message: 'built proposals=' + proposals.length }, 30);
      return proposals;
    },
    riskFlags(hit, binding) {
      const flags = [];
      if (!binding.display) flags.push('ambiguous_target');
      if (hit.flags && hit.flags.quote) flags.push('quote_present');
      if (hit.flags && hit.flags.dream) flags.push('dream');
      if (hit.flags && hit.flags.rumor) flags.push('rumor');
      if (hit.flags && hit.flags.pretend) flags.push('pretend');
      if (hit.flags && hit.flags.denied) flags.push('denied');
      if (hit.flags && hit.flags.visibleOnly) flags.push('visible_only');
      return flags;
    }
  };

  // ============================================================
  // 12. CanonValidator Shell
  // ============================================================
  WL.CanonValidator = {
    validateAll(root, proposals) {
      const out = [];
      for (let i = 0; i < proposals.length; i++) out.push(this.validate(root, proposals[i]));
      root.pending.lastValidations = out;
      WL.AuditLogger.append(root, 'validations', { message: 'validated=' + out.length, results: out.map(v => v.result).join(',') }, 30);
      return out;
    },
    validate(root, p) {
      const f = p.flags || {};
      const denied = p.riskFlags.indexOf('denied') >= 0;
      const fake = p.riskFlags.indexOf('dream') >= 0 || p.riskFlags.indexOf('pretend') >= 0 || p.riskFlags.indexOf('quote_present') >= 0;
      const rumor = p.riskFlags.indexOf('rumor') >= 0;
      const noTarget = !p.target || !p.target.display;
      let result = 'pending', reason = 'shell default pending';

      if (p.sourceKind === 'command' || WL.EventScannerLite.unsafeSource(p.sourceKind, p.sourceText)) return this.row(p, 'reject', 'unsafe/non-canon source');
      if (noTarget && !/^body\.|^time\./.test(p.type)) return this.row(p, 'reject', 'ambiguous or unsafe target');

      switch (p.type) {
        case 'item.acquire':
          if (fake || rumor || denied || /\b(?:idea of|concept of|not take|decide not|snatches? it away)\b/i.test(p.sourceText)) result = 'reject', reason = 'unsafe item acquisition source';
          else result = 'accept', reason = 'confirmed pickup/handoff';
          break;
        case 'item.visible':
          result = 'pending'; reason = 'visible scene item only, not owned'; break;
        case 'currency.gain':
          if (fake || rumor || denied) result = 'reject', reason = 'unsafe currency source';
          else result = 'accept', reason = 'explicit payment';
          break;
        case 'ability.evidence':
          if (fake) result = 'hard_deny', reason = 'fake/pretend/quoted ability attempt';
          else if (denied || /\b(?:fails?|nothing happens|no visible effect|cannot|can't|lack technique)\b/i.test(p.sourceText)) result = 'soft_fail', reason = 'failed training evidence';
          else result = 'accept', reason = 'ability training evidence';
          break;
        case 'ability.stabilize_claim':
          result = 'pending'; reason = 'output-only ability claim cannot stabilize in Code Pass 1'; break;
        case 'body.fatigue':
          if (fake || rumor) result = 'reject', reason = 'non-canon fatigue source';
          else result = 'accept', reason = 'confirmed fatigue/exertion';
          break;
        case 'body.injury':
          if (fake || rumor) result = 'reject', reason = 'injury blocked by quote/dream/rumor';
          else result = 'accept', reason = 'confirmed injury language';
          break;
        case 'body.death_claim':
          if (fake || rumor) result = 'reject', reason = 'death claim from rumor/dream/quote';
          else result = 'pending', reason = 'death requires stronger future combat/body gate';
          break;
        case 'quest.offer':
          if (fake || rumor) result = 'reject', reason = 'unsafe quest source';
          else result = 'accept', reason = 'quest offer detected';
          break;
        case 'quest.accept':
          if (fake || rumor || denied) result = 'reject', reason = 'unsafe quest acceptance';
          else result = 'accept', reason = 'quest acceptance detected';
          break;
        case 'quest.complete':
          if (fake || rumor || denied) result = 'reject', reason = 'unsafe quest completion';
          else result = 'accept', reason = 'quest completion detected';
          break;
        case 'reward.promise':
          if (fake || rumor) result = 'reject', reason = 'unsafe reward promise';
          else result = 'accept', reason = 'promise tracked, not paid';
          break;
        case 'actor.present':
          if (fake || rumor) result = 'reject', reason = 'unsafe actor presence';
          else result = 'accept', reason = 'actor directly present/acting';
          break;
        case 'place.current':
          if (fake || rumor || denied || p.data.weakReference) result = 'reject', reason = 'weak/unsafe place source';
          else result = 'accept', reason = 'strong arrival/current scene';
          break;
        case 'place.reference':
          result = 'pending'; reason = 'weak place reference only'; break;
        case 'time.pass':
          if (fake || rumor || /\byears?\b/i.test(p.sourceText) || /\bpracticing narration\b/i.test(p.sourceText)) result = 'reject', reason = 'unsafe/nonliteral time passage';
          else result = 'accept', reason = 'literal short time advance';
          break;
        default:
          result = 'pending'; reason = 'no active Code Pass 1 rule';
      }
      return this.row(p, result, reason);
    },
    row(p, result, reason) {
      return { proposalId: p.id, type: p.type, family: p.family, result, reason, target: p.target, opKey: p.opKey, sourceText: p.sourceText, data: p.data };
    }
  };

  // ============================================================
  // 13. ReducerRegistry and Limited Reducers
  // ============================================================
  WL.ReducerRegistry = {
    applyAcceptedBatch(root, validations) {
      const results = [];
      for (let i = 0; i < validations.length; i++) {
        const v = validations[i];
        if (['accept','soft_fail','hard_deny'].indexOf(v.result) < 0) { results.push({ type: v.type, status: 'skipped', reason: v.result }); continue; }
        const led = WL.OperationLedger.tryAdd(root, 'recentKeys', v.opKey, { type: v.type, result: v.result });
        if (!led.applied) { results.push({ type: v.type, status: 'duplicate_skip', reason: v.reason }); continue; }
        let r;
        if (v.family === 'inventory') r = WL.InventoryReducer.apply(root, v);
        else if (v.family === 'ability') r = WL.AbilityReducer.apply(root, v);
        else if (v.family === 'body') r = WL.BodyReducer.apply(root, v);
        else if (v.family === 'quest') r = WL.QuestReducerLite.apply(root, v);
        else if (v.family === 'actor') r = WL.ActorReducerLite.apply(root, v);
        else if (v.family === 'place') r = WL.PlaceReducerLite.apply(root, v);
        else if (v.family === 'time') r = WL.TimeReducerLite.apply(root, v);
        else r = { type: v.type, status: 'no_owner', reason: 'No active reducer owner.' };
        results.push(r);
      }
      root.pending.lastReducerResults = results;
      WL.AuditLogger.append(root, 'reducers', { message: 'reducer results=' + results.length, statuses: results.map(r => r.status).join(',') }, 30);
      return results;
    }
  };

  WL.InventoryReducer = {
    apply(root, v) {
      if (v.type === 'item.acquire') return this.acquire(root, v);
      if (v.type === 'item.visible') return this.visible(root, v);
      if (v.type === 'currency.gain') return this.currency(root, v);
      return { type: v.type, status: 'ignored', reason: 'Inventory reducer has no handler.' };
    },
    itemId(name) { return 'item_' + WL.U.slug(name); },
    acquire(root, v) {
      const name = v.target.display;
      const id = this.itemId(name);
      if (!root.inventory.itemsById[id]) root.inventory.itemsById[id] = { id, name, qty: 1, holder: 'player', owner: 'player', availability: 'available', createdTurn: WL.U.nowTurn(root) };
      else root.inventory.itemsById[id].qty = WL.U.clamp((root.inventory.itemsById[id].qty || 1) + 1, 1, 9999);
      if (root.inventory.ownedIds.indexOf(id) < 0) root.inventory.ownedIds.push(id);
      WL.DirtyQueue.add(root, 'inventory', 'item acquired: ' + name);
      return { type: v.type, status: 'committed', target: name };
    },
    visible(root, v) {
      const name = v.target.display;
      WL.U.uniquePush(root.inventory.visibleSceneItems, name, 20);
      WL.DirtyQueue.add(root, 'inventory', 'visible item: ' + name);
      return { type: v.type, status: 'pending_visible_only', target: name };
    },
    currency(root, v) {
      const reward = v.data && v.data.reward || WL.Sanitizers.reward(v.target.raw || v.target.display || '');
      if (!reward) return { type: v.type, status: 'rejected', reason: 'No currency parsed.' };
      const payKey = WL.OperationLedger.makeKey(['reward', v.sourceText, reward.amount, reward.coin]);
      const led = WL.OperationLedger.tryAdd(root, 'rewardKeys', payKey, { reward });
      if (!led.applied) return { type: v.type, status: 'duplicate_skip', reason: 'Reward payment already applied.' };
      root.inventory.currency[reward.coin] = WL.U.clamp((root.inventory.currency[reward.coin] || 0) + reward.amount, 0, 999999999);
      root.quests.rewardPayments.push({ turn: WL.U.nowTurn(root), amount: reward.amount, coin: reward.coin, source: v.sourceText });
      WL.U.capArray(root.quests.rewardPayments, 40);
      WL.DirtyQueue.add(root, 'inventory', 'currency gain');
      return { type: v.type, status: 'committed', target: reward.amount + ' ' + reward.coin };
    }
  };

  WL.AbilityReducer = {
    apply(root, v) {
      if (v.type === 'ability.evidence') return this.evidence(root, v);
      if (v.type === 'ability.stabilize_claim') return { type: v.type, status: 'pending', reason: v.reason };
      return { type: v.type, status: 'ignored' };
    },
    evidence(root, v) {
      const name = v.target.display;
      const id = 'ability_' + WL.U.slug(name);
      if (v.result === 'hard_deny') {
        root.abilities.denied.push({ turn: WL.U.nowTurn(root), name, reason: v.reason });
        WL.U.capArray(root.abilities.denied, 30);
        return { type: v.type, status: 'hard_denied', target: name };
      }
      let rec = root.abilities.candidatesById[id];
      if (!rec) {
        rec = { id, name, stage: 'candidate', attempts: 0, failures: 0, successes: 0, evidence: [], createdTurn: WL.U.nowTurn(root), lastTurn: WL.U.nowTurn(root) };
        root.abilities.candidatesById[id] = rec;
        if (root.abilities.candidateIds.indexOf(id) < 0) root.abilities.candidateIds.push(id);
      }
      rec.attempts += 1;
      rec.lastTurn = WL.U.nowTurn(root);
      if (v.result === 'soft_fail') rec.failures += 1;
      else rec.successes += 1;
      rec.evidence.push({ turn: WL.U.nowTurn(root), result: v.result, source: v.sourceText.slice(0, 120) });
      WL.U.capArray(rec.evidence, 12);
      WL.DirtyQueue.add(root, 'abilities', 'ability evidence: ' + name);
      return { type: v.type, status: v.result === 'soft_fail' ? 'soft_fail_committed' : 'committed', target: name };
    }
  };

  WL.BodyReducer = {
    apply(root, v) {
      if (v.type === 'body.fatigue') {
        root.body.ep = WL.U.clamp(root.body.ep - 5, 0, root.body.maxEp);
        WL.U.uniquePush(root.body.conditions, 'fatigued', 20);
        WL.DirtyQueue.add(root, 'body', 'fatigue');
        return { type: v.type, status: 'committed', target: 'fatigue' };
      }
      if (v.type === 'body.injury') {
        root.body.hp = WL.U.clamp(root.body.hp - 5, 0, root.body.maxHp);
        WL.U.uniquePush(root.body.injuries, 'minor injury', 20);
        WL.DirtyQueue.add(root, 'body', 'injury');
        return { type: v.type, status: 'committed', target: 'injury' };
      }
      return { type: v.type, status: 'pending_or_rejected', reason: v.reason };
    }
  };

  WL.QuestReducerLite = {
    apply(root, v) {
      if (v.type === 'reward.promise') return this.promise(root, v);
      const title = v.target.display || 'Untitled objective';
      const id = 'quest_' + WL.U.slug(title);
      let q = root.quests.questsById[id];
      if (!q) {
        q = { id, title, status: 'offered', createdTurn: WL.U.nowTurn(root), lastTurn: WL.U.nowTurn(root), progress: [] };
        root.quests.questsById[id] = q;
        if (root.quests.offeredIds.indexOf(id) < 0) root.quests.offeredIds.push(id);
      }
      q.lastTurn = WL.U.nowTurn(root);
      if (v.type === 'quest.accept') { q.status = 'active'; if (root.quests.activeIds.indexOf(id) < 0) root.quests.activeIds.push(id); }
      if (v.type === 'quest.complete') { q.status = 'completed'; WL.U.uniquePush(root.quests.completedIds, id, 80); root.quests.activeIds = root.quests.activeIds.filter(x => x !== id); }
      WL.DirtyQueue.add(root, 'quests', v.type + ': ' + title);
      return { type: v.type, status: 'committed', target: title };
    },
    promise(root, v) {
      const reward = v.data && v.data.reward;
      const text = reward ? (reward.amount + ' ' + reward.coin + ' promised') : WL.U.norm(v.target.raw || v.target.display) + ' promised';
      root.quests.rewardPromises.push({ turn: WL.U.nowTurn(root), text, paid: false, source: v.sourceText });
      WL.U.capArray(root.quests.rewardPromises, 40);
      WL.DirtyQueue.add(root, 'quests', 'reward promise');
      return { type: v.type, status: 'committed_promise_only', target: text };
    }
  };

  WL.ActorReducerLite = {
    apply(root, v) {
      const name = v.target.display;
      const id = 'actor_' + WL.U.slug(name);
      let a = root.actors.actorsById[id];
      if (!a) a = root.actors.actorsById[id] = { id, name, stage: 'present', mentions: 0, present: true, createdTurn: WL.U.nowTurn(root), lastTurn: WL.U.nowTurn(root), notes: [] };
      a.mentions += 1; a.present = true; a.lastTurn = WL.U.nowTurn(root);
      if (root.actors.presentIds.indexOf(id) < 0) root.actors.presentIds.push(id);
      if (v.data && v.data.companion && root.actors.companionIds.indexOf(id) < 0) root.actors.companionIds.push(id);
      WL.DirtyQueue.add(root, 'actors', 'actor present: ' + name);
      return { type: v.type, status: 'committed', target: name };
    }
  };

  WL.PlaceReducerLite = {
    apply(root, v) {
      if (v.type === 'place.reference') {
        const name = v.target.display; const id = 'place_' + WL.U.slug(name);
        if (name && root.places.referencedIds.indexOf(id) < 0) root.places.referencedIds.push(id);
        return { type: v.type, status: 'reference_only', target: name };
      }
      if (v.type === 'place.route') { root.places.routeId = 'route_' + WL.U.slug(v.target.display || 'route'); return { type: v.type, status: 'route_only' }; }
      const name = v.target.display;
      const id = 'place_' + WL.U.slug(name);
      let p = root.places.placesById[id];
      if (!p) p = root.places.placesById[id] = { id, name, visits: 0, createdTurn: WL.U.nowTurn(root), lastTurn: WL.U.nowTurn(root), notes: [] };
      p.visits += 1; p.lastTurn = WL.U.nowTurn(root);
      root.places.currentId = id;
      root.player.locationId = id;
      WL.DirtyQueue.add(root, 'places', 'current place: ' + name);
      return { type: v.type, status: 'committed', target: name };
    }
  };

  WL.TimeReducerLite = {
    textNum(value) {
      const m = { one:1, two:2, three:3, four:4, five:5, six:6, seven:7, eight:8, nine:9, ten:10, twenty:20 };
      const l = WL.U.lower(value);
      return m[l] || Number(value) || 0;
    },
    apply(root, v) {
      const n = this.textNum(v.data && v.data.amountText);
      const unit = String(v.data && v.data.unit || 'minutes');
      if (!n) return { type: v.type, status: 'rejected', reason: 'No amount.' };
      let minutes = n;
      if (/hour/.test(unit)) minutes = n * 60;
      if (/day/.test(unit)) minutes = n * 1440;
      root.time.minute += minutes;
      while (root.time.minute >= 60) { root.time.minute -= 60; root.time.hour += 1; }
      while (root.time.hour >= 24) { root.time.hour -= 24; root.time.day += 1; }
      root.time.notes.push({ turn: WL.U.nowTurn(root), text: v.sourceText });
      WL.U.capArray(root.time.notes, 20);
      WL.DirtyQueue.add(root, 'time', 'time pass');
      return { type: v.type, status: 'committed', target: n + ' ' + unit };
    }
  };


  WL.Reducers = {
    inventory: 'InventoryReducer',
    ability: 'AbilityReducer',
    body: 'BodyReducer',
    quest: 'QuestReducerLite',
    actor: 'ActorReducerLite',
    place: 'PlaceReducerLite',
    time: 'TimeReducerLite'
  };

    add(root, target, reason) {
      const k = WL.U.slug(target);
      if (root.dirty.targets.indexOf(k) < 0) root.dirty.targets.push(k);
      root.dirty.lastReason = WL.U.norm(reason);
      WL.U.capArray(root.dirty.targets, WL.SCHEMA_CAPS.dirtyTargets);
    }
  };

  WL.ContextBuilderLite = {
    build(root, baseText) {
      const parts = [];
      const b = root.body;
      parts.push('[WorldLedger Current Truth]');
      parts.push('Player: ' + root.player.name + ' | Level ' + root.player.level + ' | HP ' + b.hp + '/' + b.maxHp + ' MP ' + b.mp + '/' + b.maxMp + ' EP ' + b.ep + '/' + b.maxEp);
      if (root.places.currentId && root.places.placesById[root.places.currentId]) parts.push('Current place: ' + root.places.placesById[root.places.currentId].name);
      const items = root.inventory.ownedIds.map(id => root.inventory.itemsById[id] && root.inventory.itemsById[id].name).filter(Boolean).slice(0, 8);
      if (items.length) parts.push('Inventory: ' + items.join(', '));
      const cands = root.abilities.candidateIds.map(id => root.abilities.candidatesById[id] && root.abilities.candidatesById[id].name + ' candidate').filter(Boolean).slice(0, 6);
      if (cands.length) parts.push('Ability development: ' + cands.join(', ') + '. Candidates are not stable abilities yet.');
      const quests = root.quests.activeIds.map(id => root.quests.questsById[id] && root.quests.questsById[id].title).filter(Boolean).slice(0, 5);
      if (quests.length) parts.push('Active quests: ' + quests.join('; '));
      if (root.body.conditions.length) parts.push('Conditions: ' + root.body.conditions.join(', '));
      if (root.body.injuries.length) parts.push('Injuries: ' + root.body.injuries.join(', '));
      parts.push('Rules: player input is intent; output confirms consequences; rumors, dreams, quotes, commands, and Story Card text are not canon by themselves.');
      let packet = parts.join('\n');
      const cap = WL.U.clamp(root.features.contextBudget || 1800, 400, 5000);
      if (packet.length > cap) packet = packet.slice(0, cap - 20) + '\n[trimmed]';
      WL.AuditLogger.append(root, 'context', { message: 'Context packet built, chars=' + packet.length }, 30);
      if (!baseText) return packet;
      return String(baseText) + '\n\n' + packet;
    }
  };

  // ============================================================

  WL.CardApiAdapter = {
    available() { return typeof globalThis.storyCards !== 'undefined' || typeof globalThis.addStoryCard === 'function' || typeof globalThis.updateStoryCard === 'function'; },
    safeFindByTitle(title) {
      const wanted = WL.U.lower(title);
      const cards = WL.U.getStoryCards();
      for (let i = 0; i < cards.length; i++) {
        const c = cards[i] || {};
        if (WL.U.lower(c.title || c.name || '') === wanted) return { ok: true, card: c, index: i };
      }
      return { ok: false, card: null, index: -1 };
    },
    dryRunOnly() { return true; }
  };

    protectedRegistry(root) { return root.cards.protectedTitles || []; },
    audit(root) {
      const lines = [];
      const cards = WL.U.getStoryCards();
      lines.push('Mode: dry-run=' + !!root.features.cardSyncDryRun + ', protectedWrites=' + !!root.features.protectedCardWritesEnabled + ', generated=false');
      lines.push('Visible Story Cards detected: ' + cards.length);
      const titles = this.protectedRegistry(root);
      for (let i = 0; i < titles.length; i++) lines.push('Protected: ' + titles[i]);
      lines.push('Generated CardSync is disabled in Code Pass 1. Cards never create canon.');
      return lines;
    },
    dryRun(root) {
      if (!root.features.cardSyncLiteEnabled) return 0;
      const targets = root.dirty.targets.slice(0, 4);
      for (let i = 0; i < targets.length; i++) root.cards.dryRunWrites.push({ turn: WL.U.nowTurn(root), target: targets[i], mode: 'dry-run' });
      WL.U.capArray(root.cards.dryRunWrites, WL.SCHEMA_CAPS.dryRunWrites);
      if (targets.length) WL.AuditLogger.append(root, 'cards', { message: 'CardSync dry-run targets=' + targets.join(',') }, 30);
      return targets.length;
    }
  };

  // ============================================================
  // 16. ReleaseCheckLite Foundation
  // ============================================================
  WL.ReleaseCheckLite = {
    pass(label) { return 'PASS: ' + label; },
    warn(label) { return 'WARN: ' + label; },
    fail(label) { return 'FAIL: ' + label; },
    run(root) {
      const out = [];
      out.push(this.pass('build identity present: ' + WL.BUILD_VERSION));
      out.push(root.schemaVersion === WL.SCHEMA_VERSION ? this.pass('schema version matches') : this.fail('schema version mismatch'));
      out.push(root.features.generatedCardsEnabled === false ? this.pass('generated cards disabled') : this.fail('generated cards enabled too early'));
      out.push(root.features.protectedCardWritesEnabled === false ? this.pass('live protected card writes disabled by default') : this.warn('protected card writes enabled'));
      out.push(root.features.manualMutationCommandsEnabled === false ? this.pass('manual mutation commands disabled') : this.fail('manual mutation commands enabled'));
      const deferred = ['economyEnabled','craftingEnabled','investigationEnabled','reputationEnabled','domainEnabled','absorptionEnabled','advancedCombatEnabled','warfareEnabled','multiProtagonistEnabled'];
      for (let i = 0; i < deferred.length; i++) out.push(root.features[deferred[i]] === false ? this.pass(deferred[i] + ' disabled') : this.fail(deferred[i] + ' enabled too early'));
      out.push(root.operationLedger && Array.isArray(root.operationLedger.recentKeys) ? this.pass('operation ledger buckets present') : this.fail('operation ledger missing'));
      out.push(WL.EventScannerLite && WL.ProposalBuilder && WL.CanonValidator ? this.pass('parser/proposal/validator shells present') : this.fail('pipeline shell missing'));
      out.push(WL.ReducerRegistry ? this.pass('reducer registry present') : this.fail('reducer registry missing'));
      out.push(root.dirty && Array.isArray(root.dirty.targets) ? this.pass('dirty queue present') : this.fail('dirty queue missing'));
      out.push(root.features.contextBuilderLiteEnabled ? this.pass('context builder lite enabled') : this.warn('context builder disabled'));
      out.push(root.features.cardSyncDryRun ? this.pass('CardSync dry-run mode active') : this.fail('CardSync dry-run not active'));
      out.push(root.features.layoutLockEnabled ? this.pass('layout lock enabled') : this.fail('layout lock disabled'));
      out.push(WL.SECTION_ORDER && WL.SECTION_ORDER.length === 25 ? this.pass('25-section layout order registered') : this.fail('layout order not registered'));
      out.push(WL.LayoutLock ? this.pass('LayoutLock inspector present') : this.fail('LayoutLock inspector missing'));
      out.push(WL.StateSchemaLock ? this.pass('StateSchemaLock inspector present') : this.fail('StateSchemaLock inspector missing'));
      out.push(root.meta && root.meta.schemaId === WL.SCHEMA_ID ? this.pass('locked schema metadata present') : this.fail('locked schema metadata missing'));
      out.push(WL.CANONICAL_ROOT_BUCKETS && WL.CANONICAL_ROOT_BUCKETS.length === 20 ? this.pass('20 canonical state buckets registered') : this.fail('canonical bucket registry mismatch'));
      out.push(WL.MVP_ACTIVE_COMMANDS.indexOf('schema') >= 0 ? this.pass('/schema command registered as inspect-only') : this.fail('/schema command missing'));
      const layoutRows = WL.LayoutLock ? WL.LayoutLock.inspect(root) : [];
      const layoutFails = layoutRows.filter(r => r.status === 'FAIL').length;
      out.push(layoutFails === 0 ? this.pass('module boundary/layout audit clean') : this.fail('module boundary/layout audit failures: ' + layoutFails));
      const schemaRows = WL.StateSchemaLock ? WL.StateSchemaLock.inspect(root) : [];
      const schemaFails = schemaRows.filter(r => r.status === 'FAIL').length;
      out.push(schemaFails === 0 ? this.pass('state schema audit clean') : this.fail('state schema audit failures: ' + schemaFails));
      out.push(this.warn('Code Pass 1 is foundation-limited; advanced systems are intentionally deferred.'));
      root.audits.releasecheck = out.map(x => ({ turn: WL.U.nowTurn(root), text: x }));
      return out;
    },
    stateAudit(root) {
      const out = [];
      out.push(root && typeof root === 'object' ? this.pass('root exists') : this.fail('root missing'));
      const buckets = WL.ROOT_BUCKETS_ALL || ['meta','features','layout','runtime','startup','player','body','progression','inventory','abilities','quests','actors','places','time','cards','operationLedger','audits','migration','quarantine','dirty','cleanup','pending'];
      for (let i = 0; i < buckets.length; i++) out.push(root[buckets[i]] ? this.pass('bucket ' + buckets[i]) : this.fail('missing bucket ' + buckets[i]));
      if (WL.StateSchemaLock) WL.StateSchemaLock.inspect(root).forEach(r => out.push(r.status + ': ' + r.label));
      const b = root.body;
      out.push(b.hp >= 0 && b.hp <= b.maxHp ? this.pass('HP invariant') : this.fail('HP invariant broken'));
      out.push(b.mp >= 0 && b.mp <= b.maxMp ? this.pass('MP invariant') : this.fail('MP invariant broken'));
      out.push(b.ep >= 0 && b.ep <= b.maxEp ? this.pass('EP invariant') : this.fail('EP invariant broken'));
      return out;
    }
  };

  // ============================================================
  // 17. Hook Orchestration
  // ============================================================
  WL.processOutput = function (root, output) {
    if (root.runtime.commandSuppressed || root.runtime.commandActive) return output;
    if (WL.Runtime.isDuplicateOutput(root, output)) {
      root.runtime.duplicateOutputSkipped += 1;
      WL.AuditLogger.summarizeLastTurn(root, 'Duplicate output hash skipped; no consequences applied.');
      return output;
    }
    const hits = WL.EventScannerLite.scanOutput(root, output);
    const proposals = WL.ProposalBuilder.fromHits(root, hits, 'output');
    const validations = WL.CanonValidator.validateAll(root, proposals);
    const results = WL.ReducerRegistry.applyAcceptedBatch(root, validations);
    WL.CardSyncLite.dryRun(root);
    WL.CleanupManagerLite.run(root);
    WL.Runtime.markOutputApplied(root, output);
    WL.AuditLogger.summarizeLastTurn(root, 'Output processed: hits=' + hits.length + ', proposals=' + proposals.length + ', results=' + results.length + '.');
    return output;
  };

  WL.onInput = function (text) {
    let root;
    try {
      root = WL.StateManager.root();
      WL.Runtime.begin(root, 'input', text);
      WL.MigrationManager.run(root);
      const raw = String(text == null ? '' : text);
      if (WL.U.startsCommand(raw) && root.features.commandsEnabled) {
        const out = WL.CommandRouter.route(root, raw);
        WL.Runtime.end(root);
        return out || ' ';
      }
      root.runtime.lastInputText = raw;
      root.startup.firstGameplayInputSeen = true;
      root.startup.startupLocked = true;
      const actionCount = WL.U.getInfoActionCount();
      if (typeof actionCount !== 'number') {
        root.meta.currentTurn = (root.meta.currentTurn || 0) + 1;
        root.runtime.currentTurn = root.meta.currentTurn;
      }
      WL.Runtime.end(root);
      return raw || ' ';
    } catch (err) {
      return WL.Runtime.failSoft(root || { runtime: { errors: [] }, audits: { warnings: [] } }, 'input', err, text);
    }
  };

  WL.onContext = function (text) {
    let root;
    try {
      root = WL.StateManager.root();
      WL.Runtime.begin(root, 'context', text);
      const out = root.features.contextBuilderLiteEnabled ? WL.ContextBuilderLite.build(root, text) : String(text || '');
      WL.Runtime.end(root);
      return out || ' ';
    } catch (err) {
      return WL.Runtime.failSoft(root || { runtime: { errors: [] }, audits: { warnings: [] } }, 'context', err, text);
    }
  };

  WL.onOutput = function (text) {
    let root;
    try {
      root = WL.StateManager.root();
      WL.Runtime.begin(root, 'output', text);
      const raw = String(text == null ? '' : text);
      if (root.runtime.commandSuppressed || /^\[WorldLedger\]/i.test(raw)) {
        WL.Runtime.end(root);
        return raw || ' ';
      }
      const out = WL.processOutput(root, raw);
      WL.Runtime.end(root);
      return out || ' ';
    } catch (err) {
      return WL.Runtime.failSoft(root || { runtime: { errors: [] }, audits: { warnings: [] } }, 'output', err, text);
    }
  };

  // Minimal bundled self-test for local/dev use. It is inspect-only and never runs automatically.
  WL.devSelfTest = function () {
    globalThis.state = {};
    let r;
    WL.onInput('/releasecheck');
    WL.onInput('/layout');
    WL.onInput('/schema');
    WL.onInput('I train Ember Ward to protect allies.');
    WL.onOutput('Nothing happens, but you feel the shape of mana resist you.');
    r = WL.StateManager.root();
    const hasEmber = !!r.abilities.candidatesById.ability_ember_ward;
    WL.onOutput('You decide not to take the rusted iron key.');
    const noKey = Object.keys(WL.StateManager.root().inventory.itemsById).length === 0;
    WL.onOutput('Mira refuses the crown but hands you a potion in your hand.');
    const hasPotion = !!WL.StateManager.root().inventory.itemsById.item_potion;
    return { hasEmber, noKey, hasPotion, releasecheck: WL.ReleaseCheckLite.run(WL.StateManager.root()) };
  };

  globalThis.WorldLedger = WL;
})();
